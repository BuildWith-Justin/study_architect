const _months = [
  'Jan',
  'Feb',
  'Mar',
  'Apr',
  'May',
  'Jun',
  'Jul',
  'Aug',
  'Sep',
  'Oct',
  'Nov',
  'Dec',
];

String _dateLabel(DateTime d) {
  final base = '${_months[d.month - 1]} ${d.day}';
  return d.year == DateTime.now().year ? base : '$base, ${d.year}';
}

/// Everything the screen shows, loaded in one pass.
class _AchievementsData {
  final GamificationSnapshot snapshot;
  final GamificationStats stats;
  final StreakInfo streak;
  final Map<String, DateTime> unlocked;
  final List<XpEntry> recentXp;

  const _AchievementsData({
    required this.snapshot,
    required this.stats,
    required this.streak,
    required this.unlocked,
    required this.recentXp,
  });

  static Future<_AchievementsData> load() async {
    final engine = GamificationEngine.instance;
    final snapshot = await engine.getSnapshot();
    final stats = await engine.getStats();
    final streak = await engine.getStreakInfo();
    final badges = await AchievementRepository().getAll();
    final xp = await XpRepository().getAll();

    return _AchievementsData(
      snapshot: snapshot,
      stats: stats,
      streak: streak,
      unlocked: {for (final a in badges) a.id: a.unlockedAt},
      recentXp: xp.take(8).toList(),
    );
  }
}

class _AchievementsScreenState extends State<AchievementsScreen>

  void initState() {
    super.initState(); // RefreshableScreenMixin subscribes to the bus here.
    _future = _AchievementsData.load();
  }

  @override
  Future<void> onRefresh() async {
    final next = _AchievementsData.load();

    if (!mounted) {
      next.ignore();
      return;
    }

    setState(() {
      _future = next;
    });

    try {
      await next;
    } catch (_) {
      // The FutureBuilder shows the error state.
    }
  }

  Future<void> _recalculate() async {
    final confirmed = await showDialog<bool>(
      context: context,
      builder: (dialogContext) => AlertDialog(
        title: const Text('Recalculate XP?'),
        content: const Text(
          'This re-checks every session, task and exam topic: it adds XP '
          'that is missing and removes XP whose session, task or topic no '
          'longer exists. Your badges are kept.',
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(dialogContext, false),
            child: const Text('Cancel'),
          ),
          TextButton(
            onPressed: () => Navigator.pop(dialogContext, true),
            child: const Text('Recalculate'),
          ),
        ],
      ),
    );

    if (confirmed != true || !mounted) return;

    try {
      final report = await GamificationEngine.instance.rebuildFromHistory(
        removeOrphans: true,
      );
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text(
            'Done: ${report.written} added, ${report.removed} removed.',
          ),
        ),
      );
    } catch (_) {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Could not recalculate XP.')),
      );
    }

    await onRefresh();
  }

  List<Widget> _badgeRows(_AchievementsData d) {
    // Unlocked badges first (newest first), then the locked ones.
    final unlocked = kBadges.where((b) => d.unlocked.containsKey(b.id)).toList()
      ..sort((a, b) => d.unlocked[b.id]!.compareTo(d.unlocked[a.id]!));
    final locked = kBadges.where((b) => !d.unlocked.containsKey(b.id)).toList();
    final ordered = [...unlocked, ...locked];

    Widget card(BadgeDef b) => _BadgeCard(
          badge: b,
          unlockedAt: d.unlocked[b.id],
          progress: b.progress(d.stats),
        );

    final rows = <Widget>[];
    for (var i = 0; i < ordered.length; i += 2) {
      final right = i + 1 < ordered.length ? ordered[i + 1] : null;
      rows.add(
        Padding(
          padding: const EdgeInsets.only(bottom: AppSpacing.md),
          child: IntrinsicHeight(
            child: Row(
              crossAxisAlignment: CrossAxisAlignment.stretch,
              children: [
                Expanded(child: card(ordered[i])),
                const SizedBox(width: AppSpacing.md),
                Expanded(
                  child: right == null ? const SizedBox() : card(right),
                ),
              ],
            ),
          ),
        ),
      );
    }
    return rows;
  }

/// One badge: colored when unlocked, grey with a progress bar when not.
class _BadgeCard extends StatelessWidget {
  final BadgeDef badge;
  final DateTime? unlockedAt;
  final int progress;

  const _BadgeCard({
    required this.badge,
    required this.unlockedAt,
    required this.progress,
  });

  bool get _unlocked => unlockedAt != null;

  /// Minutes are shown as hours and minutes; everything else as a count.
  String _progressLabel() {
    final isMinutes = badge.id.startsWith('focus_') || badge.id == 'deep_focus';
    final shown = progress > badge.target ? badge.target : progress;
    if (isMinutes) {
      return '${formatDuration(shown)} / ${formatDuration(badge.target)}';
    }
    return '$shown / ${badge.target}';
  }

  @override
  Widget build(BuildContext context) {
    final theme = Theme.of(context);

    final neutralFill = context.isDarkMode
        ? Colors.white.withValues(alpha: 0.08)
        : AppColors.primary.withValues(alpha: 0.08);

    final fraction = badge.target == 0
        ? 0.0
        : (progress / badge.target).clamp(0.0, 1.0).toDouble();

    return Container(
      padding: const EdgeInsets.all(AppSpacing.md),
      decoration: BoxDecoration(
        color: context.cardSurfaceAlt,
        borderRadius: BorderRadius.circular(AppRadii.card),
      ),
      child: Column(
        mainAxisSize: MainAxisSize.min,
        children: [
          Container(
            width: 52,
            height: 52,
            decoration: BoxDecoration(
              shape: BoxShape.circle,
              color: _unlocked ? AppColors.primary : neutralFill,
            ),
            child: Icon(
              badge.icon,
              size: 26,
              color: _unlocked
                  ? Colors.white
                  : context.secondaryText.withValues(alpha: 0.7),
            ),
          ),
          const SizedBox(height: AppSpacing.sm),
          Text(
            badge.title,
            textAlign: TextAlign.center,
            maxLines: 1,
            overflow: TextOverflow.ellipsis,
            style: theme.textTheme.titleSmall?.copyWith(
              color: _unlocked ? context.primaryText : context.secondaryText,
              fontWeight: FontWeight.w700,
            ),
          ),
          const SizedBox(height: 2),
          Text(
            badge.description,
            textAlign: TextAlign.center,
            maxLines: 3,
            overflow: TextOverflow.ellipsis,
            style: theme.textTheme.bodySmall?.copyWith(
              color: context.secondaryText,
            ),
          ),
          const SizedBox(height: AppSpacing.sm),
          if (_unlocked)
            Text(
              'Unlocked ${_dateLabel(unlockedAt!)}',
              textAlign: TextAlign.center,
              style: theme.textTheme.bodySmall?.copyWith(
                color: AppColors.primary,
                fontWeight: FontWeight.w600,
              ),
            )
          else ...[
            ClipRRect(
              borderRadius: BorderRadius.circular(4),
              child: LinearProgressIndicator(
                value: fraction,
                minHeight: 5,
                backgroundColor: neutralFill,
                valueColor: const AlwaysStoppedAnimation(AppColors.primary),
              ),
            ),
            const SizedBox(height: 4),
            Text(
              _progressLabel(),
              textAlign: TextAlign.center,
              style: theme.textTheme.bodySmall?.copyWith(
                color: context.secondaryText,
              ),
            ),
          ],
        ],
      ),
    );
  }
}

/// V2 Step 8a: the screen behind the "+" nav item (route /plan).
///
/// Pick a subject, an optional topic and a focus length, then either:
///  - "Start focus now"  -> opens the Focus Timer straight away, or
///  - "Schedule for later" -> opens the Step 7b session form (unchanged).
class CreateSessionScreen extends StatefulWidget {
  const CreateSessionScreen({super.key});

  @override
  State<CreateSessionScreen> createState() => _CreateSessionScreenState();
}

class _CreateSessionScreenState extends State<CreateSessionScreen> {
  static const _lengthOptions = [25, 45, 60, 90, 120];

  final _subjectRepo = SubjectRepository();
  final _topicRepo = TopicRepository();

  List<Subject> _subjects = [];
  List<Topic> _topics = [];

  String? _subjectId;
  String? _topicId;
  int _minutes = 60;

  bool _loading = true;
  String? _error;

  @override
  void initState() {
    super.initState();
    _loadSubjects();
  }

  Subject? get _selectedSubject {
    for (final s in _subjects) {
      if (s.id == _subjectId) return s;
    }
    return null;
  }

  Future<void> _loadSubjects() async {
    try {
      final subjects = await _subjectRepo.getAll();

      if (!mounted) return;

      setState(() {
        _subjects = subjects;
        _subjectId = subjects.isNotEmpty ? subjects.first.id : null;
        _loading = false;
        _error = null;
      });

      if (_subjectId != null) {
        await _loadTopics(_subjectId!);
      }
    } catch (_) {
      if (!mounted) return;

      setState(() {
        _loading = false;
        _error = 'Could not load your subjects.';
      });
    }
  }

  Future<void> _loadTopics(String subjectId) async {
    try {
      final topics = await _topicRepo.getForSubject(subjectId);

      if (!mounted) return;

      setState(() {
        _topics = topics;
        if (!_topics.any((t) => t.id == _topicId)) {
          _topicId = null;
        }
      });
    } catch (_) {
      if (!mounted) return;

      setState(() {
        _topics = [];
        _topicId = null;
      });
    }
  }

  Future<void> _startNow() async {
    final subject = _selectedSubject;

    if (subject == null) {
      setState(() {
        _error = 'Choose a subject first.';
      });
      return;
    }

    String? topicName;
    for (final t in _topics) {
      if (t.id == _topicId) topicName = t.name;
    }

    final navigator = Navigator.of(context);

    // pushReplacement: closing the timer returns straight to the tab
    // the user came from, not back to this form.
    await navigator.pushReplacement(
      MaterialPageRoute<bool>(
        builder: (_) => FocusTimerScreen(
          subjectId: subject.id,
          subjectName: subject.name,
          subjectColorValue: subject.colorValue,
          topicId: _topicId,
          topicName: topicName,
          goal: Duration(minutes: _minutes),
        ),
      ),
    );

    // Tell Home / Timetable that sessions may have changed.
    DataRefreshBus.instance.notify(RefreshScope.sessions);
  }

  void _scheduleLater() {
    Navigator.of(context).pushReplacement(
      MaterialPageRoute<bool>(
        builder: (_) => AddEditSessionScreen(initialDate: DateTime.now()),
      ),
    );
  }

  InputDecoration get _fieldDecoration => InputDecoration(
        filled: true,
        fillColor: AppColors.surfaceLight,
        contentPadding: const EdgeInsets.symmetric(
          horizontal: AppSpacing.lg,
          vertical: AppSpacing.md,
        ),
        border: OutlineInputBorder(
          borderRadius: BorderRadius.circular(AppRadii.control),
          borderSide: BorderSide.none,
        ),
      );

  Widget _label(String text) {
    return Padding(
      padding: const EdgeInsets.only(bottom: AppSpacing.sm),
      child: Text(
        text,
        style: Theme.of(context).textTheme.labelLarge?.copyWith(
              color: context.primaryText,
              fontWeight: FontWeight.w600,
            ),
      ),
    );
  }

  Widget _subjectDot(int colorValue) {
    return Container(
      width: 10,
      height: 10,
      decoration: BoxDecoration(
        color: Color(colorValue),
        shape: BoxShape.circle,
      ),
    );
  }

  Widget _buildForm() {
    return ListView(
      padding: const EdgeInsets.all(AppSpacing.xl),
      children: [
        _label('Subject'),
        DropdownButtonFormField<String>(
          // ignore: deprecated_member_use
          value: _subjectId,
          isExpanded: true,
          dropdownColor: AppColors.surfaceLight,
          style: const TextStyle(color: AppColors.textPrimary, fontSize: 16),
          decoration: _fieldDecoration,
          items: _subjects
              .map(
                (s) => DropdownMenuItem<String>(
                  value: s.id,
                  child: Row(
                    children: [
                      _subjectDot(s.colorValue),
                      const SizedBox(width: AppSpacing.sm),
                      Expanded(
                        child: Text(s.name, overflow: TextOverflow.ellipsis),
                      ),
                    ],
                  ),
                ),
              )
              .toList(),
          onChanged: (value) {
            if (value == null) return;
            setState(() {
              _subjectId = value;
              _topicId = null;
              _topics = [];
              _error = null;
            });
            _loadTopics(value);
          },
        ),
        const SizedBox(height: AppSpacing.lg),
        _label('Topic (optional)'),

/// Handles both creating a session (pass no [existing]) and editing
/// one. [initialDate] pre-selects a date for new sessions (e.g. the
/// day currently selected on the Timetable screen). Pops `true` on

/// V2 Step 7: a new session (or an existing one-off session) can be
/// repeated weekly. Every weekly occurrence is saved as its own row,
/// all sharing one seriesId.
class AddEditSessionScreen extends StatefulWidget {

  DateTime _combine(

    final sessionsLabel = count == 1 ? '1 session' : '$count sessions';
    return '$sessionsLabel · every ${_weekdayName(_date)} until ${_formatDate(_repeatUntil)}';

  Future<void> _pickDate() async {
    final now = DateTime.now();
    final oneYearAgo = startOfDay(now.subtract(const Duration(days: 365)));
    final oneYearAhead = startOfDay(now.add(const Duration(days: 365)));
    final current = startOfDay(_date);

    final picked = await showDatePicker(
      context: context,
      initialDate: current,
      firstDate: current.isBefore(oneYearAgo) ? current : oneYearAgo,
      lastDate: current.isAfter(oneYearAhead) ? current : oneYearAhead,
    );

    if (picked == null || !mounted) return;

    setState(() {
      _date = picked;

      // Keep "repeat until" valid for the new start date.
      if (_repeatUntil.isBefore(picked)) {
        _repeatUntil = DateTime(picked.year, picked.month, picked.day + 77);
      }
      final maxUntil = _maxRepeatUntil;
      if (_repeatUntil.isAfter(maxUntil)) {
        _repeatUntil = maxUntil;
      }
    });
  }

  Future<void> _pickRepeatUntil() async {
    final first = startOfDay(_date);
    final last = _maxRepeatUntil;
    var initial = startOfDay(_repeatUntil);
    if (initial.isBefore(first)) initial = first;
    if (initial.isAfter(last)) initial = last;

    final picked = await showDatePicker(
      context: context,
      initialDate: initial,
      firstDate: first,
      lastDate: last,
      helpText: 'Repeat until',
    );

    if (picked == null || !mounted) return;

    setState(() {
      _repeatUntil = picked;
    });
  }

  Future<void> _pickTime({
    required bool isStart,
  }) async {
    final picked = await showTimePicker(
      context: context,
      initialTime: isStart ? _startTime : _endTime,
    );

    if (picked == null || !mounted) return;

    setState(() {
      if (isStart) {
        _startTime = picked;
      } else {
        _endTime = picked;
      }
    });
  }

  /// Reminders are a nice-to-have: a failure here must never make a
  /// successful save look like a failed one.
  Future<void> _scheduleReminders(List<StudySession> sessions) async {
    try {
      if (_subjects.isEmpty) return;
      final subject = _subjects.firstWhere(
        (s) => s.id == sessions.first.subjectId,
        orElse: () => _subjects.first,
      );

      final now = DateTime.now();
      var scheduled = 0;
      for (final s in sessions) {
        if (s.startTime.isBefore(now)) continue;
        if (scheduled >= _maxReminders) break;
        await NotificationService.instance.scheduleSessionReminder(
          sessionId: s.id,
          subjectName: subject.name,
          sessionStart: s.startTime,
        );
        scheduled++;
      }
    } catch (_) {
      // Ignored on purpose, see above.
    }
  }

  Future<void> _cancelReminders(List<String> sessionIds) async {
    try {
      for (final id in sessionIds) {
        await NotificationService.instance.cancelSessionReminder(id);
      }
    } catch (_) {
      // Ignored on purpose: the session itself is already deleted.
    }
  }

  Future<void> _save() async {
    if (_subjectId == null) {
      setState(() {
        _error = 'Choose a subject first.';
      });
      return;
    }

    final start = _combine(
      _date,
      _startTime,
    );

    final end = _combine(
      _date,
      _endTime,
    );

    final existing = widget.existing;

    final base = StudySession(
      id: existing?.id ?? const Uuid().v4(),
      subjectId: _subjectId!,
      topicId: _topicId,
      scheduledDate: startOfDay(_date),
      startTime: start,
      endTime: end,
      notes: _notesController.text.trim(),
      status: existing?.status ?? SessionStatus.planned,
      complete: existing?.complete ?? false,
      actualDuration: existing?.actualDuration,
      rating: existing?.rating,
      recurrence: existing?.recurrence ?? SessionRecurrence.none,
      recurrenceUntil: existing?.recurrenceUntil,
      seriesId: existing?.seriesId,
    );

    if (!base.isTimeRangeValid) {
      setState(() {
        _error = 'End time must be after start time.';
      });
      return;
    }

    setState(() {
      _saving = true;
      _error = null;
    });

    try {
      final makeSeries = _repeatWeekly && !_isRecurringExisting;
      List<StudySession> toRemind;

      if (makeSeries) {
        final series = buildWeeklySeries(first: base, until: _repeatUntil);

        // Keep every field of the session being edited (status, rating,
        // notes...) on the first occurrence; the others start fresh.
        final firstSaved = base.copyWith(
          recurrence: SessionRecurrence.weekly,
          recurrenceUntil: startOfDay(_repeatUntil),
          seriesId: series.first.seriesId,
        );
        final all = [firstSaved, ...series.skip(1)];

        if (_isEditing) {
          await _sessionRepo.update(firstSaved);
          if (all.length > 1) {
            await _sessionRepo.createMany(all.sublist(1));
          }
        } else {
          await _sessionRepo.createMany(all);
        }
        toRemind = all;
      } else {
        if (_isEditing) {
          await _sessionRepo.update(base);
        } else {
          await _sessionRepo.create(base);
        }
        toRemind = [base];
      }

      await _scheduleReminders(toRemind);

      // V2 Step 13: a completed session that was edited (for example moved
      // to another day) keeps its XP in line.
      if (_isEditing && base.complete) {
        await GamificationHooks.sessionCompleted(base);
      }

      DataRefreshBus.instance.notify(RefreshScope.sessions);

      if (!mounted) return;

      Navigator.of(context).pop(true);
    } catch (e) {
      if (!mounted) return;

      setState(() {
        _saving = false;
        _error = 'Could not save the session. Please try again.';
      });
    }
  }

  Future<_DeleteChoice> _askDeleteChoice() async {
    final existing = widget.existing!;

    if (!existing.isRecurring) {

                  color: Colors.redAccent,
                ),
              ),
            ),
          ],
        ),
      );
      return confirmed == true ? _DeleteChoice.onlyThis : _DeleteChoice.cancel;
    }

    final choice = await showDialog<_DeleteChoice>(
      context: context,
      builder: (context) => AlertDialog(
        title: const Text('Delete recurring session?'),
        content: const Text(
          'This session is part of a weekly series. This cannot be undone.',
        ),
        actions: [

            child: const Text(
              'Only this session',
              style: TextStyle(color: Colors.redAccent),
            ),
          ),
          TextButton(
            onPressed: () => Navigator.pop(
              context,
              _DeleteChoice.thisAndFollowing,
            ),
            child: const Text(
              'This and following',
              style: TextStyle(color: Colors.redAccent),
            ),
          ),
        ],
      ),
    );
    return choice ?? _DeleteChoice.cancel;
  }

  Future<void> _delete() async {
    final existing = widget.existing!;
    final choice = await _askDeleteChoice();

    if (choice == _DeleteChoice.cancel) return;

    try {
      if (choice == _DeleteChoice.thisAndFollowing &&
          existing.seriesId != null) {
        final removed = await _sessionRepo.deleteSeries(
          existing.seriesId!,
          fromDate: existing.scheduledDate,
        );
        await _cancelReminders(removed.map((s) => s.id).toList());
        await GamificationHooks.sessionsRemoved(
          removed.map((s) => s.id).toList(),
        );
      } else {
        await _sessionRepo.delete(existing.id);
        await _cancelReminders([existing.id]);
        await GamificationHooks.sessionRemoved(existing.id);
      }

      DataRefreshBus.instance.notify(RefreshScope.sessions);

      if (!mounted) return;

      Navigator.of(context).pop(true);
    } catch (e) {
      if (!mounted) return;

      setState(() {
        _error = 'Could not delete the session. Please try again.';
      });
    }
  }

  DateTime? _loadedWeekStart;

  /// Data scopes that make this screen reload itself automatically.

  List<RefreshScope> get watchedScopes => [
        RefreshScope.sessions,

    super.initState(); // RefreshableScreenMixin subscribes to the bus here.

  /// Loads the week that contains [_selectedDay] and remembers it for
  /// the day dots, but only if that week is still the one on screen.
  Future<TimetableWeekData> _load() {
    final weekStart = startOfWeek(_selectedDay);
    return TimetableWeekData.load(_selectedDay).then((data) {
      if (mounted && startOfWeek(_selectedDay) == weekStart) {
        setState(() {
          _loadedData = data;
          _loadedWeekStart = weekStart;
        });
      }
      return data;

  /// (a session was saved, deleted or synced) and "Today".

    if (!mounted) return;

    setState(() => _future = next);
    try {
      await next;

      // AsyncBuilder shows the error state with a retry button.
    }

  String _weekRangeLabel(DateTime weekStart) {

    const months = [

      'Jan',
      'Feb',
      'Mar',
      'Apr',
      'May',
      'Jun',
      'Jul',
      'Aug',
      'Sep',
      'Oct',
      'Nov',
      'Dec',
    ];

    if (weekStart.month == weekEnd.month) {
      final month = months[weekStart.month - 1];
      return 'Week • $month ${weekStart.day}–${weekEnd.day}';
    }

    // The week crosses two months, e.g. "Sep 28 – Oct 4".
    final startMonth = shortMonths[weekStart.month - 1];
    final endMonth = shortMonths[weekEnd.month - 1];
    return 'Week • $startMonth ${weekStart.day} – $endMonth ${weekEnd.day}';
  }

import 'package:flutter/material.dart';
import 'app_theme.dart';

/// Shorthand for the isDark-branch pattern already used in
/// StatCard/ProgressCard since Step 2 — lets every other screen pick
/// up the same "dedicated dark scheme, not just inverted" behavior
/// without repeating `Theme.of(context).brightness == Brightness.dark`
/// everywhere.
///
/// Deliberately NOT used inside form input fields (TextField /
/// DropdownButtonFormField fill colors and their typed-text color) —
/// those stay a fixed light surface with fixed dark text in both
/// themes, since that was a specific, already-debugged fix and
/// switching it per-theme risks reintroducing invisible typed text.
extension ThemeAwareColors on BuildContext {
  bool get isDarkMode => Theme.of(this).brightness == Brightness.dark;

  /// Plain info/stat card background — e.g. a settings row, a summary
  /// box. Not for pastel accent cards, which stay the same hue in
  /// both themes by design.
  Color get cardSurface =>
      isDarkMode ? AppColors.surfaceDark : AppColors.surfaceLight;

  Color get cardSurfaceAlt =>
      isDarkMode ? AppColors.surfaceDarkAlt : AppColors.surfaceLightAlt;

  /// Body text sitting on a [cardSurface] (not on a pastel card, and
  /// not typed text inside a form field).
  Color get primaryText =>
      isDarkMode ? AppColors.textPrimaryDark : AppColors.textPrimary;

  Color get secondaryText =>
      isDarkMode ? AppColors.textSecondaryDark : AppColors.textSecondary;
}

class _QuickAddSheetState extends State<_QuickAddSheet> {
  static const _uuid = Uuid();
  static const _weekdayNames = [
    'Mon',
    'Tue',
    'Wed',
    'Thu',
    'Fri',
    'Sat',
    'Sun'
  ];
  static const _monthNames = [
    'Jan',
    'Feb',
    'Mar',
    'Apr',
    'May',
    'Jun',
    'Jul',
    'Aug',
    'Sep',
    'Oct',
    'Nov',
    'Dec',
  ];
  static const _durationChoices = [15, 30, 45, 60, 90, 120, 180, 240];

  final _inputCtrl = TextEditingController();
  final _titleCtrl = TextEditingController();

  List<Subject> _subjects = const [];
  bool _loadingSubjects = true;
  QuickAddResult? _parsed;

  QuickAddKind _kind = QuickAddKind.session;
  String? _subjectId;
  DateTime _date = startOfDay(DateTime.now());
  int _startMinutes = 17 * 60;
  int _duration = 60;

  String? _error;
  bool _saving = false;

  @override
  void initState() {
    super.initState();
    _loadSubjects();
  }

  @override
  void dispose() {
    _inputCtrl.dispose();
    _titleCtrl.dispose();
    super.dispose();
  }

  Future<void> _loadSubjects() async {
    try {
      final list = await SubjectRepository().getAll();
      if (!mounted) return;
      setState(() {
        _subjects = list;
        _loadingSubjects = false;
      });
      _onInputChanged(_inputCtrl.text);
    } catch (_) {
      if (!mounted) return;
      setState(() {
        _loadingSubjects = false;
        _error = 'Could not load your subjects.';
      });
    }
  }

  void _onInputChanged(String text) {
    final result = QuickAddParser.parse(text, subjects: _subjects);
    setState(() {
      if (!result.hasContent) {
        _parsed = null;
        return;
      }
      _parsed = result;
      _kind = result.kind;
      if (result.subject != null) {
        _subjectId = result.subject!.id;
      } else if (_subjectId != null &&
          !_subjects.any((s) => s.id == _subjectId)) {
        _subjectId = null;
      }
      if (_subjectId == null && _subjects.length == 1) {
        _subjectId = _subjects.first.id;
      }
      _date = result.date;
      _startMinutes = result.startMinutes;
      _duration = result.durationMinutes;
      _titleCtrl.text = result.title;
      _error = null;
    });
  }

  String _formatDate(DateTime d) =>
      '${_weekdayNames[d.weekday - 1]} ${d.day} ${_monthNames[d.month - 1]} ${d.year}';

  String _formatTime(int minutes) =>
      formatTimeOfDay(DateTime(2000, 1, 1, minutes ~/ 60, minutes % 60));

  Future<void> _pickDate() async {
    final picked = await showDatePicker(
      context: context,
      initialDate: _date,
      firstDate: DateTime(2020),
      lastDate: DateTime(2100),
    );
    if (picked != null && mounted) {
      setState(() => _date = startOfDay(picked));
    }
  }

  Future<void> _pickTime() async {
    final picked = await showTimePicker(
      context: context,
      initialTime: TimeOfDay(
        hour: _startMinutes ~/ 60,
        minute: _startMinutes % 60,
      ),
    );
    if (picked != null && mounted) {
      setState(() => _startMinutes = picked.hour * 60 + picked.minute);
    }
  }

  Future<void> _save() async {
    final parsed = _parsed;
    if (parsed == null) {
      setState(
          () => _error = 'Type something first, e.g. "Maths tomorrow 4pm 1h".');
      return;
    }
    final subjectId = _subjectId;
    if (subjectId == null) {
      setState(() => _error = 'Choose a subject.');
      return;
    }

    setState(() {
      _saving = true;
      _error = null;
    });

    try {
      if (_kind == QuickAddKind.session) {
        final window =
            QuickAddParser.sessionWindow(_date, _startMinutes, _duration);
        if (!window.valid) {
          setState(() {
            _saving = false;
            _error = 'The end time must be after the start time.';
          });
          return;
        }
        final session = StudySession(
          id: _uuid.v4(),
          subjectId: subjectId,
          scheduledDate: startOfDay(_date),
          startTime: window.start,
          endTime: window.end,
          notes: _titleCtrl.text.trim(),
        );
        await SessionRepository().create(session);
        DataRefreshBus.instance.notify(RefreshScope.sessions);
      } else {
        final title = _titleCtrl.text.trim();
        if (title.isEmpty) {
          setState(() {
            _saving = false;
            _error = 'Give the task a title.';
          });
          return;
        }
        final task = Task(
          id: _uuid.v4(),
          title: title,
          subjectId: subjectId,
          dueDate: startOfDay(_date),
          type: parsed.taskType,
          createdAt: DateTime.now(),
        );
        await TaskRepository().create(task);
        DataRefreshBus.instance.notify(RefreshScope.tasks);
      }
      if (!mounted) return;
      Navigator.of(context).pop(true);
    } catch (e) {
      if (!mounted) return;
      setState(() {
        _saving = false;
        _error = 'Could not save: $e';
      });
    }
  }

  List<String> _hints() {
    final parsed = _parsed;
    if (parsed == null) return const [];
    final hints = <String>[];
    if (parsed.dateAssumed) hints.add('No date found, using today.');
    if (_kind == QuickAddKind.session) {
      if (parsed.timeAssumed) {
        hints.add('No time found, using ${_formatTime(parsed.startMinutes)}.');
      }
      if (parsed.durationAssumed) {
        hints.add(
            'No duration found, using ${formatDuration(parsed.durationMinutes)}.');
      }
      final window =
          QuickAddParser.sessionWindow(_date, _startMinutes, _duration);
      if (window.clamped) {
        hints.add('This would pass midnight, so it ends at 11:59 PM.');
      }
      if (window.start.isBefore(DateTime.now())) {
        hints.add('This time has already passed.');
      }
    }
    hints.addAll(parsed.warnings);
    return hints;
  }
