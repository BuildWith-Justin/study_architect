import 'package:flutter/material.dart';
import '../repositories/settings_repository.dart';

/// Single source of truth for the app's current ThemeMode. MaterialApp
/// listens to [mode] directly, so calling [setMode] anywhere in the
/// app (from the More/Settings screen) updates every screen
/// immediately — no restart needed.
class ThemeController {
  ThemeController._internal();
  static final ThemeController instance = ThemeController._internal();

  final ValueNotifier<ThemeMode> mode = ValueNotifier(ThemeMode.system);
  final _settingsRepo = SettingsRepository();

  Future<void> loadFromSettings() async {
    final settings = await _settingsRepo.get();
    mode.value = settings.themeMode;
  }

  Future<void> setMode(ThemeMode newMode) async {
    mode.value = newMode;
    final current = await _settingsRepo.get();
    await _settingsRepo.save(current.copyWith(themeMode: newMode));
  }
}

enum RefreshScope {
  subjects,
  sessions,

  exams,

class DataRefreshBus {
  DataRefreshBus._internal();
  static final DataRefreshBus instance = DataRefreshBus._internal();

  final StreamController<RefreshScope> _controller =
      StreamController<RefreshScope>.broadcast();

  Stream<RefreshScope> get stream => _controller.stream;

  void notify(RefreshScope scope) {
    _controller.add(scope);
  }

  void dispose() {
    _controller.close();
  }
}

import 'dart:async';
import 'package:flutter/material.dart';
import 'data_refresh_bus.dart';

/// Mixin any StatefulWidget's State with this to get:
/// - auto-refresh when relevant data changes elsewhere in the app
/// - pull-to-refresh wired to the same reload function
mixin RefreshableScreenMixin<T extends StatefulWidget> on State<T> {
  StreamSubscription<RefreshScope>? _refreshSub;

  /// Override: which scopes should trigger a refresh on this screen
  List<RefreshScope> get watchedScopes => [RefreshScope.all];

  /// Override: how this screen reloads its data
  Future<void> onRefresh();

  @override
  void initState() {
    super.initState();
    _refreshSub = DataRefreshBus.instance.stream.listen((scope) {
      if (watchedScopes.contains(scope) || scope == RefreshScope.all) {
        onRefresh();
      }
    });
  }

  @override
  void dispose() {
    _refreshSub?.cancel();
    super.dispose();
  }

  Widget buildRefreshable({required Widget child}) {
    return RefreshIndicator(
      onRefresh: onRefresh,
      child: child,
    );
  }
}

/// Sync state of a locally stored row.
enum SyncStatus {
  synced(0),
  pendingUpsert(1),
  pendingDelete(2);

  const SyncStatus(this.value);
  final int value;

  static SyncStatus fromValue(int v) => SyncStatus.values
      .firstWhere((s) => s.value == v, orElse: () => SyncStatus.synced);
}

/// Every syncable local table carries these columns.
/// - id: client-generated UUID (works offline, same id on server)
/// - updatedAt: last modification time (server value after a sync)
/// - deletedAt: soft-delete marker (row is kept until the delete is pushed)
/// - syncStatus: what still needs to be pushed
mixin Syncable {
  String get id;
  DateTime get updatedAt;
  DateTime? get deletedAt;
  SyncStatus get syncStatus;

  bool get isDeleted => deletedAt != null;
  bool get needsPush => syncStatus != SyncStatus.synced;
}

/// Names of tables in the exact order they must be pushed
/// (parents before children, so foreign keys never fail).
const List<String> kSyncPushOrder = [
  'subjects',

  'study_sessions',
  'study_logs',
  'tasks',
  'exams',
  'exam_topics',
  'achievements',
  'xp_log',
];

/// Local key that stores the last successful pull time per table.
String lastPulledKey(String table) => 'last_pulled_$table';

void main() async {
  WidgetsFlutterBinding.ensureInitialized();

  // Plain sqflite only ships a native database engine for Android/iOS.
  // On Windows and Linux desktop there is no such engine bundled, so
  // sqflite_common_ffi's databaseFactory must be installed explicitly
  // before DatabaseService (or anything else) opens a database —
  // otherwise every DB call throws "databaseFactory not initialized."
  if (!kIsWeb && (Platform.isWindows || Platform.isLinux)) {
    sqfliteFfiInit();
    databaseFactory = databaseFactoryFfi;
  }

  // V2: load Supabase credentials from .env and start the client.
  // Wrapped in try/catch so a missing .env or a Supabase problem never
  // stops the local-first app from launching.
  try {
    await dotenv.load(fileName: '.env');
    await Supabase.initialize(
      url: dotenv.env['SUPABASE_URL']!,
      publishableKey: dotenv.env['SUPABASE_ANON_KEY']!,
    );
    // Only flips to true when init fully succeeded. AuthGate reads this
    // so it never touches Supabase.instance when the client isn't ready.
    AuthService.isAvailable = true;
    debugPrint('Supabase initialized OK');

    // V2 Step 4: start local <-> cloud sync (does nothing until signed in).
    SyncService.instance.start();
  } catch (e) {
    AuthService.isAvailable = false;
    debugPrint('Supabase init failed: $e');
  }

  // Safe no-op on Web — see notification_service.dart for why.
  await NotificationService.instance.init();
  runApp(const StudyArchitectApp());
}

/// An exam the student is preparing for (V2 Step 11).
class Exam {
  final String id;
  final String subjectId;
  final String title;

  /// Date and time of the exam, in the device's local time.
  final DateTime examDate;
  final String location;
  final String notes;
  final DateTime createdAt;

  const Exam({
    required this.id,
    required this.subjectId,
    required this.title,
    required this.examDate,
    this.location = '',
    this.notes = '',
    required this.createdAt,
  });

  Exam copyWith({
    String? id,
    String? subjectId,
    String? title,
    DateTime? examDate,
    String? location,
    String? notes,
    DateTime? createdAt,
  }) {
    return Exam(
      id: id ?? this.id,
      subjectId: subjectId ?? this.subjectId,
      title: title ?? this.title,
      examDate: examDate ?? this.examDate,
      location: location ?? this.location,
      notes: notes ?? this.notes,
      createdAt: createdAt ?? this.createdAt,
    );
  }

  /// Whole calendar days from today to the exam day (0 = today,
  /// negative = already passed). The time of day is ignored.
  int daysLeft([DateTime? now]) {
    final today = now ?? DateTime.now();
    final todayDay = DateTime.utc(today.year, today.month, today.day);
    final examDay = DateTime.utc(examDate.year, examDate.month, examDate.day);
    return examDay.difference(todayDay).inDays;
  }

  /// True once the exam day is behind us.
  bool isPast([DateTime? now]) => daysLeft(now) < 0;

  Map<String, dynamic> toMap() => {
        'id': id,
        'subjectId': subjectId,
        'title': title,
        'examDate': examDate.toLocal().toIso8601String(),
        'location': location,
        'notes': notes,
        'createdAt': createdAt.toIso8601String(),
      };

  factory Exam.fromMap(Map<String, dynamic> map) => Exam(
        id: map['id'] as String,
        subjectId: map['subjectId'] as String,
        title: map['title'] as String,
        examDate: DateTime.parse(map['examDate'] as String),
        location: (map['location'] as String?) ?? '',
        notes: (map['notes'] as String?) ?? '',
        createdAt: DateTime.parse(map['createdAt'] as String),
      );
}

/// How far along the preparation checklist of one exam is.
class ExamProgress {
  final int prepared;
  final int total;

  const ExamProgress({required this.prepared, required this.total});

  const ExamProgress.empty()
      : prepared = 0,
        total = 0;

  /// 0.0 to 1.0 (0 when the exam has no topics yet).
  double get fraction => total == 0 ? 0 : prepared / total;

  int get percent => (fraction * 100).round();
}

import 'package:sqflite/sqflite.dart';
import 'package:uuid/uuid.dart';

import '../models/exam.dart';
import '../models/exam_topic.dart';
import '../services/database_service.dart';

/// All reads and writes for exams and their topic checklist.
/// Like the other repositories, it does not notify DataRefreshBus —
/// the screens that change data call
/// DataRefreshBus.instance.notify(RefreshScope.exams) afterwards.
class ExamRepository {
  final _db = DatabaseService.instance;
  static const _uuid = Uuid();

  /// Creates the exam and, in the same transaction, its topic checklist.
  Future<void> create(Exam exam, {List<String> topicIds = const []}) async {
    final db = await _db.database;
    await db.transaction((txn) async {
      await txn.insert('exams', exam.toMap());
      for (final topicId in topicIds.toSet()) {
        await txn.insert(
          'exam_topics',
          ExamTopic(id: _uuid.v4(), examId: exam.id, topicId: topicId).toMap(),
        );
      }
    });
  }

  /// Updates the exam. When [topicIds] is given, the checklist is made to
  /// match it: removed topics are deleted, new ones are added, and topics
  /// that stay keep their "prepared" state.
  Future<void> update(Exam exam, {List<String>? topicIds}) async {
    final db = await _db.database;
    await db.transaction((txn) async {
      await txn.update(
        'exams',
        exam.toMap(),
        where: 'id = ?',
        whereArgs: [exam.id],
      );
      if (topicIds != null) {
        await _syncTopicLinks(txn, exam.id, topicIds);
      }
    });
  }

  Future<void> _syncTopicLinks(
    DatabaseExecutor txn,
    String examId,
    List<String> topicIds,
  ) async {
    final wanted = topicIds.toSet();
    final rows = await txn.query(
      'exam_topics',
      where: 'examId = ?',
      whereArgs: [examId],
    );
    final existing = <String, String>{
      for (final r in rows) r['topicId'] as String: r['id'] as String,
    };

    for (final entry in existing.entries) {
      if (!wanted.contains(entry.key)) {
        await txn.delete(
          'exam_topics',
          where: 'id = ?',
          whereArgs: [entry.value],
        );
      }
    }
    for (final topicId in wanted) {
      if (!existing.containsKey(topicId)) {
        await txn.insert(
          'exam_topics',
          ExamTopic(id: _uuid.v4(), examId: examId, topicId: topicId).toMap(),
        );
      }
    }
  }

  /// Deletes the exam. Its checklist rows cascade.
  Future<void> delete(String id) async {
    final db = await _db.database;
    await db.delete('exams', where: 'id = ?', whereArgs: [id]);
  }

class SessionRepository {

  /// Throws if the session's end time is not after its start time -

      throw ArgumentError('Session end time must be after start time.');
    }
    final db = await _db.database;
    await db.insert('study_sessions', session.toMap());
  }

  /// V2 Step 7: saves a whole weekly series in one transaction, so a
  /// failure can never leave half a series behind.
  Future<void> createMany(List<StudySession> sessions) async {
    for (final s in sessions) {
      if (!s.isTimeRangeValid) {
        throw ArgumentError('Session end time must be after start time.');
      }
    }
    final db = await _db.database;
    await db.transaction((txn) async {
      for (final s in sessions) {
        await txn.insert('study_sessions', s.toMap());
      }
    });
  }

  Future<void> update(StudySession session) async {
    if (!session.isTimeRangeValid) {
      throw ArgumentError('Session end time must be after start time.');
    }
    final db = await _db.database;
    await db.update(
      'study_sessions',
      session.toMap(),
      where: 'id = ?',
      whereArgs: [session.id],
    );
  }

  Future<void> delete(String id) async {
    final db = await _db.database;
    await db.delete('study_sessions', where: 'id = ?', whereArgs: [id]);
  }

  /// V2 Step 7: deletes the occurrences of one series. With [fromDate],
  /// only occurrences on or after that day are removed. Returns the
  /// deleted sessions so the caller can cancel their reminders.
  Future<List<StudySession>> deleteSeries(
    String seriesId, {
    DateTime? fromDate,
  }) async {
    final db = await _db.database;
    final String where;
    final List<Object?> args;
    if (fromDate == null) {
      where = 'seriesId = ?';
      args = [seriesId];
    } else {
      final day = DateTime(fromDate.year, fromDate.month, fromDate.day);
      where = 'seriesId = ? AND scheduledDate >= ?';
      args = [seriesId, day.toIso8601String()];
    }
    final rows = await db.query(
      'study_sessions',
      where: where,
      whereArgs: args,
    );
    await db.delete('study_sessions', where: where, whereArgs: args);
    return rows.map(StudySession.fromMap).toList();
  }

  /// Every occurrence of one series, oldest first.
  Future<List<StudySession>> getBySeries(String seriesId) async {

class TopicRepository {
  final _db = DatabaseService.instance;

  /// Always appends the topic at the end of its subject's roadmap,
  /// whatever sortOrder the caller passed in.
  Future<void> create(Topic topic) async {
    final db = await _db.database;
    final rows = await db.rawQuery(
      'SELECT COALESCE(MAX(sortOrder), -1) + 1 AS next '
      'FROM topics WHERE subjectId = ?',
      [topic.subjectId],
    );
    final next = (rows.first['next'] as int?) ?? 0;
    await db.insert('topics', topic.copyWith(sortOrder: next).toMap());
  }

  Future<void> update(Topic topic) async {
    final db = await _db.database;
    await db.update('topics', topic.toMap(),
        where: 'id = ?', whereArgs: [topic.id]);
  }

  /// Changes only the name, leaving position and completion untouched.
  Future<void> rename(String id, String name) async {
    final db = await _db.database;
    await db.update(
      'topics',
      {'name': name},
      where: 'id = ?',
      whereArgs: [id],
    );
  }

  Future<void> delete(String id) async {
    final db = await _db.database;
    await db.delete('topics', where: 'id = ?', whereArgs: [id]);
  }

  Future<void> setCompleted(String id, bool completed) async {
    final db = await _db.database;
    await db.update(
      'topics',
      {'completed': completed ? 1 : 0},
      where: 'id = ?',
      whereArgs: [id],
    );
  }

  /// Topics of one subject in roadmap order (ties fall back to the
  /// order they were created in).
  Future<List<Topic>> getForSubject(String subjectId) async {
    final db = await _db.database;
    final rows = await db.query(
      'topics',
      where: 'subjectId = ?',
      whereArgs: [subjectId],
      orderBy: 'sortOrder ASC, rowid ASC',
    );
    return rows.map(Topic.fromMap).toList();
  }

  /// Saves a new roadmap order. [orderedIds] is the full list of topic
  /// ids of one subject, first topic first. Only rows whose position
  /// actually changed are written, so unchanged topics are not re-synced.
  Future<void> reorder(List<String> orderedIds) async {
    final db = await _db.database;
    await db.transaction((txn) async {
      for (var i = 0; i < orderedIds.length; i++) {
        await txn.update(
          'topics',
          {'sortOrder': i},
          where: 'id = ? AND sortOrder != ?',
          whereArgs: [orderedIds[i], i],
        );

                        accent: subject != null

                            : AppColors.primary,
                      ),

                  }),

            ),
          ),
        );
      },

  /// The session currently running, or the next upcoming one today.
  _SessionAndSubject? _currentOrNextSession(HomeViewData data) {
    final now = DateTime.now();

    final sorted = [...data.todaySessions]
      ..sort((a, b) => a.startTime.compareTo(b.startTime));

    for (final session in sorted) {
      if (session.endTime.isAfter(now)) {
        return _SessionAndSubject(
          session,
          data.subjectsById[session.subjectId],
        );
      }
    }

    return null;
  }

  Widget _buildClassCard(HomeViewData data) {
    final match = _currentOrNextSession(data)!;
    final session = match.session;
    final subject = match.subject;
    final now = DateTime.now();

    final isLive =
        session.startTime.isBefore(now) && session.endTime.isAfter(now);

    return ClassCard(
      timeRange:
          '${formatTimeOfDay(session.startTime)} - ${formatTimeOfDay(session.endTime)}',
      title: subject?.name ?? 'Unknown subject',
      hue: _hueForSubject(subject),
      actionLabel: isLive ? 'Start session' : 'View details',
      onAction: isLive

                  subjectColorValue:
                      subject?.colorValue ?? AppColors.accentPurpleStrong.value,
                  topicId: session.topicId,
                  existingSessionId: session.id,
                  goal: session.endTime.difference(session.startTime),
                ),
              )
          : () {},
    );
  }

  PastelHue _hueForSubject(Subject? subject) {
    if (subject == null) {
      return PastelHue.blue;

    final color = subject.colorValue;

    if (color == AppColors.accentGreenStrong.value) {
      return PastelHue.green;
    }

    if (color == AppColors.accentBlueStrong.value) {
      return PastelHue.blue;
    }

    return PastelHue.purple;
  }

/// Everything the Home screen needs, loaded in one pass so the UI
/// never has to juggle five separate FutureBuilders.
class HomeViewData {
  final Student student;
  final int todayMinutes;
  final int dailyGoalMinutes;
  final int weeklyMinutes;
  final int weeklyGoalMinutes;
  final int subjectsCount;
  final int tasksTodayCount;
  final int streak;
  final List<StudySession> todaySessions;
  final Map<String, Subject> subjectsById;
  final List<Task> todayTasks;

  /// Days (yyyy-MM-dd) with at least one completed session. Powers the
  /// 7-day streak strip on Home (V2 step 6).
  final Set<String> completedDates;

  const HomeViewData({
    required this.student,
    required this.todayMinutes,
    required this.dailyGoalMinutes,
    required this.weeklyMinutes,
    required this.weeklyGoalMinutes,
    required this.subjectsCount,
    required this.tasksTodayCount,
    required this.streak,
    required this.todaySessions,
    required this.subjectsById,
    required this.todayTasks,
    this.completedDates = const <String>{},
  });

  double get todayProgress => dailyGoalMinutes == 0
      ? 0
      : (todayMinutes / dailyGoalMinutes).clamp(0.0, 1.0);

  int get weeklyGoalPercent => weeklyGoalMinutes == 0
      ? 0
      : ((weeklyMinutes / weeklyGoalMinutes) * 100).round();

  /// True when [day] has at least one completed study session.
  bool studiedOn(DateTime day) => completedDates.contains(_dateKey(day));

  static String _dateKey(DateTime date) {
    final y = date.year.toString().padLeft(4, '0');
    final m = date.month.toString().padLeft(2, '0');
    final d = date.day.toString().padLeft(2, '0');
    return '$y-$m-$d';
  }

  static Future<HomeViewData> load() async {
    final studentRepo = StudentRepository();
    final subjectRepo = SubjectRepository();
    final sessionRepo = SessionRepository();
    final taskRepo = TaskRepository();

    final now = DateTime.now();
    final weekStart = startOfWeek(now);
    final weekEnd = endOfWeek(now);

    final student = await studentRepo.getCurrent();
    if (student == null) {
      throw StateError(
          'HomeViewData.load() called before onboarding completed.');
    }

    final subjects = await subjectRepo.getAll();
    final subjectsById = {for (final s in subjects) s.id: s};

    final todaySessions = await sessionRepo.getForDate(now);
    final weekSessions = await sessionRepo.getForRange(weekStart, weekEnd);
    final todayTasks = await taskRepo.getForDate(now);
    final completedDates = await sessionRepo.getCompletedDates();

    return HomeViewData(
      student: student,
      todayMinutes: completedMinutesForDate(todaySessions, now),
      dailyGoalMinutes: student.dailyGoalMinutes,
      weeklyMinutes: completedMinutesTotal(weekSessions),
      weeklyGoalMinutes: student.weeklyGoalMinutes,
      subjectsCount: subjects.length,
      tasksTodayCount: todayTasks.where((t) => !t.completed).length,
      streak: currentStreak(completedDates, today: now),
      todaySessions: todaySessions,
      subjectsById: subjectsById,
      todayTasks: todayTasks,
      completedDates: completedDates,
    );
  }
}

  Future<void> _exportData() async {
    if (_isLoading) return;

    setState(() {
      _isLoading = true;
    });

    try {
      await _backupService.shareBackup();

      if (!mounted) return;

      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text('Backup created successfully.'),
        ),
      );
    } catch (e) {
      if (!mounted) return;

      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(

        ),
      );
    } finally {
      if (!mounted) return;

      setState(() {
        _isLoading = false;
      });
    }
  }

  Future<void> _importData() async {
    if (_isLoading) return;

    try {
      final result = await FilePicker.pickFiles(
        type: FileType.custom,
        allowedExtensions: ['json'],
      );

      if (result.isEmpty) {
        return;
      }

      if (!mounted) return;

      final file = result.first;

      if (file.path == null) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(
            content: Text('Could not access the selected file.'),
          ),
        );
        return;
      }

      setState(() {
        _isLoading = true;
      });

      final fileBytes = await File(file.path!).readAsBytes();
      final jsonString = utf8.decode(fileBytes);
      final parsed = jsonDecode(jsonString);

      if (!mounted) return;

      final shouldRestore = await showDialog<bool>(
        context: context,
        builder: (context) {
          return AlertDialog(
            title: const Text('Restore backup?'),
            content: const Text(
              'Restoring this backup may replace your current '
              'Study Architect data.',
            ),
            actions: [
              TextButton(
                onPressed: () => Navigator.pop(context, false),
                child: const Text('Cancel'),
              ),
              FilledButton(
                onPressed: () => Navigator.pop(context, true),
                child: const Text('Restore'),
              ),
            ],
          );
        },
      );

      if (shouldRestore != true) {
        return;
      }

      await _backupService.restore(parsed);

      if (!mounted) return;

      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text('Backup restored successfully.'),
        ),
      );

      _goToStartup();
    } catch (e) {
      if (!mounted) return;

      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text('Restore failed: $e'),
        ),
      );
    } finally {
      if (!mounted) return;

      setState(() {
        _isLoading = false;
      });
    }
  }

  Future<void> _clearAllData() async {
    if (_isLoading) return;

    final shouldClear = await showDialog<bool>(
      context: context,

        } catch (_) {
          // AsyncBuilder shows the error state.
        }
      } while (_refreshQueued && mounted);
    } finally {
      _isRefreshing = false;
    }

import '../../models/study_session.dart';
import '../../models/subject.dart';
import '../../repositories/session_repository.dart';
import '../../repositories/student_repository.dart';
import '../../repositories/subject_repository.dart';
import '../../utils/date_utils.dart';
import '../../utils/study_stats.dart';
import '../../utils/study_stats.dart' as study_stats;
import '../../utils/study_insights.dart';
import '../../widgets/simple_bar_chart.dart';

enum ProgressPeriod { week, month, allTime }

class ProgressViewData {
  final ProgressPeriod period;
  final int totalMinutes;
  final int targetMinutes;
  final List<BarChartPoint> chartPoints;
  final Map<Subject, int> subjectPerformancePercent;
  final int streak;
  final int longestStreak;
  final List<StudyInsight> insights;

  const ProgressViewData({
    required this.period,
    required this.totalMinutes,
    required this.targetMinutes,
    required this.chartPoints,
    required this.subjectPerformancePercent,
    required this.streak,
    required this.longestStreak,
    required this.insights,
  });

  static Future<ProgressViewData> load(ProgressPeriod period) async {
    final studentRepo = StudentRepository();
    final subjectRepo = SubjectRepository();
    final sessionRepo = SessionRepository();

    final student = await studentRepo.getCurrent();
    final subjects = await subjectRepo.getAll();
    final completedDates = await sessionRepo.getCompletedDates();
    final now = DateTime.now();

    List<StudySession> periodSessions;
    List<BarChartPoint> chartPoints;
    int targetMinutes;
    double periodWeeks;

    switch (period) {
      case ProgressPeriod.week:
        final weekStart = startOfWeek(now);
        final weekEnd = endOfWeek(now);
        periodSessions = await sessionRepo.getForRange(weekStart, weekEnd);
        const labels = ['M', 'T', 'W', 'T', 'F', 'S', 'S'];
        chartPoints = List.generate(7, (i) {
          final day = weekStart.add(Duration(days: i));
          final minutes = completedMinutesForDate(periodSessions, day);
          return BarChartPoint(label: labels[i], value: minutes);
        });
        targetMinutes = student?.weeklyGoalMinutes ?? 0;
        periodWeeks = 1;
        break;

      case ProgressPeriod.month:
        final monthStart = DateTime(now.year, now.month, 1);
        final monthEnd = DateTime(now.year, now.month + 1, 1);
        periodSessions = await sessionRepo.getForRange(monthStart, monthEnd);
        // Bucket into calendar weeks within the month.
        final weekBuckets = <int, int>{};
        for (final session in periodSessions.where((s) => s.complete)) {
          final weekIndex = ((session.scheduledDate.day - 1) / 7).floor();
          weekBuckets[weekIndex] =
              (weekBuckets[weekIndex] ?? 0) + (session.actualDuration ?? 0);
        }
        final weekCount = (monthEnd.difference(monthStart).inDays / 7).ceil();
        chartPoints = List.generate(weekCount, (i) {
          return BarChartPoint(label: 'W${i + 1}', value: weekBuckets[i] ?? 0);
        });
        final daysInMonth = monthEnd.difference(monthStart).inDays;
        targetMinutes = (student?.dailyGoalMinutes ?? 0) * daysInMonth;
        periodWeeks = daysInMonth / 7;
        break;

      case ProgressPeriod.allTime:
        periodSessions = await sessionRepo.getAll();
        final completedSessions =
            periodSessions.where((s) => s.complete).toList();
        if (completedSessions.isEmpty) {
          chartPoints = const [];
          targetMinutes = 0;
          periodWeeks = 1;
        } else {
          final monthBuckets = <String, int>{};
          for (final session in completedSessions) {
            final key =
                '${session.scheduledDate.year}-${session.scheduledDate.month}';
            monthBuckets[key] =
                (monthBuckets[key] ?? 0) + (session.actualDuration ?? 0);
          }
          final sortedKeys = monthBuckets.keys.toList()
            ..sort((a, b) {
              final ap = a.split('-').map(int.parse).toList();
              final bp = b.split('-').map(int.parse).toList();
              return DateTime(ap[0], ap[1]).compareTo(DateTime(bp[0], bp[1]));
            });
          // Cap to the most recent 6 months so the chart stays readable.
          final shown = sortedKeys.length > 6
              ? sortedKeys.sublist(sortedKeys.length - 6)
              : sortedKeys;
          const monthNames = [
            'Jan',
            'Feb',
            'Mar',
            'Apr',
            'May',
            'Jun',
            'Jul',
            'Aug',
            'Sep',
            'Oct',
            'Nov',
            'Dec',
          ];
          chartPoints = shown.map((key) {
            final month = int.parse(key.split('-')[1]);
            return BarChartPoint(
                label: monthNames[month - 1], value: monthBuckets[key]!);
          }).toList();

          final firstDate = completedSessions
              .map((s) => s.scheduledDate)
              .reduce((a, b) => a.isBefore(b) ? a : b);
          final daysTracked = now.difference(firstDate).inDays.clamp(1, 100000);
          targetMinutes = (student?.dailyGoalMinutes ?? 0) * daysTracked;
          periodWeeks = daysTracked / 7;
        }
        break;
    }

    if (periodWeeks < 1) periodWeeks = 1;

    final totalMinutes = completedMinutesTotal(periodSessions);

    final subjectPerformance = <Subject, int>{};
    for (final subject in subjects) {
      final subjectMinutes = completedMinutesTotal(
        periodSessions.where((s) => s.subjectId == subject.id).toList(),
      );
      final subjectTarget = subject.weeklyTargetMinutes * periodWeeks;
      final percent = subjectTarget == 0
          ? 0
          : ((subjectMinutes / subjectTarget) * 100).round().clamp(0, 100);
      subjectPerformance[subject] = percent;
    }

    final streak = currentStreak(completedDates, today: now);
    final longest = study_stats.longestStreak(completedDates);

    final insights = generateInsights(
      todayMinutes:
          completedMinutesForDate(await sessionRepo.getForDate(now), now),
      weeklyMinutes: period == ProgressPeriod.week
          ? totalMinutes
          : completedMinutesTotal(
              await sessionRepo.getForRange(startOfWeek(now), endOfWeek(now))),
      weeklyGoalMinutes: student?.weeklyGoalMinutes ?? 0,
      streak: streak,
      longestStreak: longest,
      subjectPerformancePercent: {
        for (final entry in subjectPerformance.entries)
          entry.key.name: entry.value,
      },
    );

    return ProgressViewData(
      period: period,
      totalMinutes: totalMinutes,
      targetMinutes: targetMinutes,
      chartPoints: chartPoints,
      subjectPerformancePercent: subjectPerformance,
      streak: streak,
      longestStreak: longest,
      insights: insights,
    );
  }
}

        RefreshScope.sessions,
        RefreshScope.subjects,
      ];

  void initState() {
    super.initState(); // RefreshableScreenMixin subscribes to the bus here.
    _future = WeeklyReportViewData.load();
  }

  /// Used by pull-to-refresh and the DataRefreshBus. A refresh asked for
  /// while one is already running is queued and runs right after.
  @override
  Future<void> onRefresh() async {
    if (!mounted) return;

    if (_isRefreshing) {
      _refreshQueued = true;
      return;
    }

    _isRefreshing = true;

    try {
      do {
        _refreshQueued = false;

        final next = WeeklyReportViewData.load();

        if (!mounted) {
          next.ignore();
          return;
        }

        setState(() => _future = next);

        try {
          await next;
        } catch (_) {
          // AsyncBuilder shows the error state.
        }
      } while (_refreshQueued && mounted);
    } finally {
      _isRefreshing = false;
    }
  }

/// Minutes studied for one subject during the report week.
class SubjectMinutes {
  final Subject subject;
  final int minutes;

  const SubjectMinutes({required this.subject, required this.minutes});
}

/// Everything the Weekly Report screen needs, loaded in one pass.
class WeeklyReportViewData {
  final DateTime weekStart;
  final int totalMinutes;
  final int lastWeekMinutes;
  final int goalMinutes;
  final int completedSessions;
  final int totalSessions;
  final int lastWeekCompletedSessions;
  final List<BarChartPoint> dayPoints;

  /// 0 = Monday ... 6 = Sunday, or -1 when nothing was studied.
  final int bestDayIndex;
  final SubjectMinutes? topSubject;
  final List<SubjectMinutes> subjectBreakdown;
  final double? averageRating;
  final int streak;

  const WeeklyReportViewData({
    required this.weekStart,
    required this.totalMinutes,
    required this.lastWeekMinutes,
    required this.goalMinutes,
    required this.completedSessions,
    required this.totalSessions,
    required this.lastWeekCompletedSessions,
    required this.dayPoints,
    required this.bestDayIndex,
    required this.topSubject,
    required this.subjectBreakdown,
    required this.averageRating,
    required this.streak,
  });

  static const _months = [
    'Jan',
    'Feb',
    'Mar',
    'Apr',
    'May',
    'Jun',
    'Jul',
    'Aug',
    'Sep',
    'Oct',
    'Nov',
    'Dec',
  ];
  static const _dayNames = [
    'Mon',
    'Tue',
    'Wed',
    'Thu',
    'Fri',
    'Sat',
    'Sun',
  ];

  int get deltaMinutes => totalMinutes - lastWeekMinutes;

  bool get goalReached => goalMinutes > 0 && totalMinutes >= goalMinutes;

  int get goalPercent =>
      goalMinutes == 0 ? 0 : ((totalMinutes / goalMinutes) * 100).round();

  String get bestDayLabel => bestDayIndex < 0 ? '-' : _dayNames[bestDayIndex];

  String get weekRangeLabel {
    final end = DateTime(weekStart.year, weekStart.month, weekStart.day + 6);
    final startMonth = _months[weekStart.month - 1];
    if (weekStart.month == end.month) {
      return '$startMonth ${weekStart.day} - ${end.day}';
    }
    return '$startMonth ${weekStart.day} - ${_months[end.month - 1]} ${end.day}';
  }

  String get summaryLine {
    if (totalMinutes == 0 && lastWeekMinutes == 0) {
      return 'No study time logged yet. Complete a session to start your report.';
    }
    final diff = deltaMinutes;
    final String compare;
    if (diff > 0) {
      compare = 'You studied ${formatDuration(diff)} more than last week.';
    } else if (diff < 0) {
      compare = 'You studied ${formatDuration(-diff)} less than last week.';
    } else {
      compare = 'Same study time as last week.';
    }
    return goalReached ? '$compare Weekly goal reached.' : compare;
  }

  static Future<WeeklyReportViewData> load() async {
    final sessionRepo = SessionRepository();
    final subjectRepo = SubjectRepository();
    final studentRepo = StudentRepository();

    final now = DateTime.now();
    final weekStart = startOfWeek(now);
    final weekEnd = endOfWeek(now);
    final lastWeekStart = DateTime(
      weekStart.year,
      weekStart.month,
      weekStart.day - 7,
    );

    final student = await studentRepo.getCurrent();
    final subjects = await subjectRepo.getAll();
    final subjectsById = {for (final s in subjects) s.id: s};
    final all = await sessionRepo.getForRange(lastWeekStart, weekEnd);
    final completedDates = await sessionRepo.getCompletedDates();

    // The repository range is inclusive at both ends, so filter again
    // here to keep each week strictly [start, nextStart).
    bool inRange(StudySession s, DateTime a, DateTime b) =>
        !s.scheduledDate.isBefore(a) && s.scheduledDate.isBefore(b);

    final thisWeek = all.where((s) => inRange(s, weekStart, weekEnd)).toList();
    final lastWeek =
        all.where((s) => inRange(s, lastWeekStart, weekStart)).toList();

    const labels = ['M', 'T', 'W', 'T', 'F', 'S', 'S'];
    final dayPoints = List.generate(7, (i) {
      final day = DateTime(weekStart.year, weekStart.month, weekStart.day + i);
      return BarChartPoint(
        label: labels[i],
        value: completedMinutesForDate(thisWeek, day),
      );
    });

    var bestDayIndex = -1;
    var bestValue = 0;
    for (var i = 0; i < dayPoints.length; i++) {
      if (dayPoints[i].value > bestValue) {
        bestValue = dayPoints[i].value;
        bestDayIndex = i;
      }
    }

    final minutesBySubject = <String, int>{};
    final ratings = <int>[];
    for (final s in thisWeek.where((s) => s.complete)) {
      minutesBySubject[s.subjectId] =
          (minutesBySubject[s.subjectId] ?? 0) + (s.actualDuration ?? 0);
      if (s.rating != null) ratings.add(s.rating!);
    }

    final breakdown = <SubjectMinutes>[];
    for (final entry in minutesBySubject.entries) {
      final subject = subjectsById[entry.key];
      if (subject != null && entry.value > 0) {
        breakdown.add(SubjectMinutes(subject: subject, minutes: entry.value));
      }
    }
    breakdown.sort((a, b) => b.minutes.compareTo(a.minutes));

    final averageRating = ratings.isEmpty
        ? null
        : ratings.fold<int>(0, (sum, r) => sum + r) / ratings.length;

    return WeeklyReportViewData(
      weekStart: weekStart,
      totalMinutes: completedMinutesTotal(thisWeek),
      lastWeekMinutes: completedMinutesTotal(lastWeek),
      goalMinutes: student?.weeklyGoalMinutes ?? 0,
      completedSessions: thisWeek.where((s) => s.complete).length,
      totalSessions: thisWeek.length,
      lastWeekCompletedSessions: lastWeek.where((s) => s.complete).length,
      dayPoints: dayPoints,
      bestDayIndex: bestDayIndex,
      topSubject: breakdown.isEmpty ? null : breakdown.first,
      subjectBreakdown: breakdown,
      averageRating: averageRating,
      streak: currentStreak(completedDates, today: now),
    );
  }
}

class _ConfettiPiece {
  final double x; // 0..1 across the screen
  final double startY; // starts slightly above the screen
  final double speed;
  final double sway;
  final double phase;
  final double spin;
  final double width;
  final double height;
  final Color color;

  const _ConfettiPiece({
    required this.x,
    required this.startY,
    required this.speed,
    required this.sway,
    required this.phase,
    required this.spin,
    required this.width,
    required this.height,
    required this.color,
  });
}

class _ConfettiPainter extends CustomPainter {
  final double progress; // 0..1
  final List<_ConfettiPiece> pieces;

  const _ConfettiPainter({
    required this.progress,
    required this.pieces,
  });

  @override
  void paint(Canvas canvas, Size size) {
    if (progress >= 1) return;

    // Fade everything out over the last 30% of the animation.
    final fade = progress < 0.7 ? 1.0 : (1 - (progress - 0.7) / 0.3);

    for (final p in pieces) {
      final y = (p.startY + p.speed * progress * 1.25) * size.height;
      if (y > size.height || y < -20) continue;

      final x = (p.x + math.sin(p.phase + progress * 6) * p.sway) * size.width;

      final paint = Paint()
        ..color = p.color.withValues(alpha: fade.clamp(0.0, 1.0));

      canvas.save();
      canvas.translate(x, y);
      canvas.rotate(p.phase + progress * p.spin);
      canvas.drawRRect(
        RRect.fromRectAndRadius(
          Rect.fromCenter(
            center: Offset.zero,
            width: p.width,
            height: p.height,
          ),
          const Radius.circular(2),
        ),
        paint,
      );
      canvas.restore();
    }
  }

  @override
  bool shouldRepaint(covariant _ConfettiPainter old) {
    return old.progress != progress;
  }
}

/// The export/import format is just each model's existing toMap()
/// shape (the same one the repositories use for sqflite), wrapped
/// with a version tag — Step 3 built the models with this reuse in
/// mind from the start.
class BackupService {
  static const _currentVersion = 1;

  /// Gathers every table into one JSON-serializable map.
  Future<Map<String, dynamic>> buildBackupJson() async {
    final student = await StudentRepository().getCurrent();
    final settings = await SettingsRepository().get();
    final subjects = await SubjectRepository().getAll();

    final topics = <Map<String, dynamic>>[];
    final sessions = <Map<String, dynamic>>[];
    for (final subject in subjects) {
      final subjectTopics = await TopicRepository().getForSubject(subject.id);
      topics.addAll(subjectTopics.map((t) => t.toMap()));
      final subjectSessions =
          await SessionRepository().getForSubject(subject.id);
      sessions.addAll(subjectSessions.map((s) => s.toMap()));
    }

    final tasks = await TaskRepository().getAll();

    // V2 Step 11: exams and their topic checklists. Older backups simply
    // do not have these two keys.
    final examRepo = ExamRepository();
    final exams = await examRepo.getAll();
    final examTopics = await examRepo.getAllTopicLinks();

    return {
      'version': _currentVersion,
      'exportedAt': DateTime.now().toIso8601String(),
      'student': student?.toMap(),
      'settings': settings.toMap(),
      'subjects': subjects.map((s) => s.toMap()).toList(),
      'topics': topics,
      'sessions': sessions,
      'tasks': tasks.map((t) => t.toMap()).toList(),
      'exams': exams.map((e) => e.toMap()).toList(),
      'examTopics': examTopics.map((l) => l.toMap()).toList(),
    };
  }

  /// Writes the backup JSON to a temp file and returns its path, so
  /// callers can share it or otherwise hand it off to the OS.
  Future<String> exportToFile() async {
    final json = await buildBackupJson();
    final dir = await getTemporaryDirectory();
    final timestamp =
        DateTime.now().toIso8601String().replaceAll(RegExp(r'[:.]'), '-');
    final file = File('${dir.path}/study_architect_backup_$timestamp.json');
    await file.writeAsString(const JsonEncoder.withIndent('  ').convert(json));
    return file.path;
  }

  /// Writes the backup and opens the OS share sheet so the person can
  /// save it to Drive, Files, email it to themselves, etc. — there's
  /// no "export folder" concept needed since the share sheet handles
  /// wherever they actually want to put it.
  Future<void> shareBackup() async {
    final path = await exportToFile();
    await SharePlus.instance.share(
      ShareParams(files: [XFile(path)], text: 'Study Architect backup'),
    );
  }

  /// Validates and parses backup JSON text. Throws FormatException
  /// with a human-readable message if it doesn't look like a Study
  /// Architect backup.
  Map<String, dynamic> parseBackup(String jsonText) {
    final decoded = jsonDecode(jsonText);
    if (decoded is! Map<String, dynamic>) {
      throw const FormatException(
          'This file is not a valid Study Architect backup.');
    }
    if (decoded['version'] == null) {
      throw const FormatException('This file is missing a version marker.');
    }
    if (decoded['version'] > _currentVersion) {
      throw const FormatException(
          'This backup was made by a newer version of the app.');
    }
    return decoded;
  }

  /// Fully replaces local data with the contents of [backup].
  Future<void> restore(Map<String, dynamic> backup) async {
    await DatabaseService.instance.restoreFromBackup(
      student: (backup['student'] as Map?)?.cast<String, dynamic>(),
      settings: (backup['settings'] as Map?)?.cast<String, dynamic>(),
      subjects: _asMapList(backup['subjects']),
      topics: _asMapList(backup['topics']),
      sessions: _asMapList(backup['sessions']),
      tasks: _asMapList(backup['tasks']),
      exams: _asMapList(backup['exams']),
      examTopics: _asMapList(backup['examTopics']),
    );
  }

  List<Map<String, dynamic>> _asMapList(dynamic value) {
    if (value is! List) return [];
    return value.map((e) => (e as Map).cast<String, dynamic>()).toList();
  }
}

class DatabaseService {
  DatabaseService._internal();

  Database? _db;

  Future<Database> get database async {

    return _db!;
  }

  Future<Database> _open() async {
    final dbPath = await getDatabasesPath();
    final path = join(dbPath, _dbName);
    return openDatabase(
      path,
      version: _dbVersion,
      onCreate: (db, version) async {
        await _createSchema(db, version);
        await applySyncSchema(db);
        await _addExams(db);
      },
      onUpgrade: (db, oldVersion, newVersion) async {
        if (oldVersion < 2) await applySyncSchema(db);
        if (oldVersion < 3) await _addSessionRecurrence(db);
        if (oldVersion < 4) await _addTopicSortOrder(db);
        if (oldVersion < 5) await _addExams(db);
      },
      // sqflite does not enforce foreign keys by default — turn them on
      // so the ON DELETE CASCADE / SET NULL rules above actually run.
      onConfigure: (db) => db.execute('PRAGMA foreign_keys = ON'),
    );
  }

  /// V2 Step 7: recurring sessions. Every weekly occurrence is stored as
  /// its own row; these columns link the rows of one series together.
  Future<void> _addSessionRecurrence(Database db) async {
    final info = await db.rawQuery('PRAGMA table_info(study_sessions)');
    final existing = info.map((c) => c['name'] as String).toSet();
    if (!existing.contains('recurrence')) {
      await db.execute(
        "ALTER TABLE study_sessions ADD COLUMN recurrence TEXT NOT NULL DEFAULT 'none'",
      );
    }
    if (!existing.contains('recurrenceUntil')) {
      await db.execute(
        'ALTER TABLE study_sessions ADD COLUMN recurrenceUntil TEXT',
      );
    }
    if (!existing.contains('seriesId')) {
      await db.execute('ALTER TABLE study_sessions ADD COLUMN seriesId TEXT');
    }
  }

  /// V2 Step 9: ordered study roadmap. Adds topics.sortOrder and gives
  /// every existing topic a position that keeps the order it was created
  /// in (per subject). The update triggers stamp updatedAt, so the new
  /// positions upload to the cloud on the next sync.
  Future<void> _addTopicSortOrder(Database db) async {
    final info = await db.rawQuery('PRAGMA table_info(topics)');
    final existing = info.map((c) => c['name'] as String).toSet();
    if (existing.contains('sortOrder')) return;

    await db.execute(
      'ALTER TABLE topics ADD COLUMN sortOrder INTEGER NOT NULL DEFAULT 0',
    );
    await db.execute('''
      UPDATE topics
      SET sortOrder = (
        SELECT COUNT(*) FROM topics t2
        WHERE t2.subjectId = topics.subjectId AND t2.rowid < topics.rowid
      )
    ''');
  }

  Future<void> _addExams(Database db) async {

        subjectId TEXT NOT NULL,
        title TEXT NOT NULL,
        examDate TEXT NOT NULL,

      for (final row in topics) {
        await txn.insert('topics', row);
      }
      for (final row in sessions) {

      }
      for (final row in tasks) {

      }
    });

/// Stamps existing rows and creates the three bookkeeping triggers for
/// one table. Safe to run twice (IF NOT EXISTS).
Future<void> _stampAndCreateTriggers(Database db, String t) async {
  // Existing rows get a timestamp so they upload on the first sync.
  await db.execute('UPDATE $t SET updatedAt = $_now WHERE updatedAt IS NULL');

  // New row: stamp it, and cancel any pending delete for the same id
  // (happens when a backup restore re-inserts a row).
  await db.execute('''
    CREATE TRIGGER IF NOT EXISTS trg_${t}_ai AFTER INSERT ON $t
    BEGIN
      DELETE FROM sync_tombstones
        WHERE tableName = '$t' AND rowId = CAST(NEW.id AS TEXT);
      UPDATE $t SET updatedAt = $_now
        WHERE rowid = NEW.rowid AND updatedAt IS NULL;
    END
  ''');

  // Edited row: repositories never set updatedAt, so it is unchanged
  // here and we stamp it. Rows written by the sync itself carry a
  // different updatedAt, so they are left alone.
  await db.execute('''
    CREATE TRIGGER IF NOT EXISTS trg_${t}_au AFTER UPDATE ON $t
    WHEN NEW.updatedAt IS OLD.updatedAt
    BEGIN
      UPDATE $t SET updatedAt = $_now WHERE rowid = NEW.rowid;
    END
  ''');

  // Deleted row: remember it so the delete can be sent to the cloud.
  await db.execute('''
    CREATE TRIGGER IF NOT EXISTS trg_${t}_ad AFTER DELETE ON $t
    BEGIN
      INSERT OR REPLACE INTO sync_tombstones (tableName, rowId, deletedAt)
      VALUES ('$t', CAST(OLD.id AS TEXT), $_now);
    END
  ''');
}

/// Adds sync bookkeeping to the local database (schema v2).
/// - updatedAt column on every synced table
/// - triggers that stamp updatedAt on insert/update
/// - triggers that record deletes in sync_tombstones
Future<void> applySyncSchema(Database db) async {
  await _ensureMetaTables(db);

  for (final t in syncedTables) {
    await db.execute('ALTER TABLE $t ADD COLUMN updatedAt TEXT');
    await _stampAndCreateTriggers(db, t);
  }
}

/// Same bookkeeping for the exam tables (schema v5). Can be run more than
/// once: it only adds what is missing.
Future<void> applyExamSyncSchema(Database db) async {
  await _ensureMetaTables(db);

  for (final t in examSyncedTables) {
    final info = await db.rawQuery('PRAGMA table_info($t)');
    final hasUpdatedAt = info.any((c) => c['name'] == 'updatedAt');
    if (!hasUpdatedAt) {
      await db.execute('ALTER TABLE $t ADD COLUMN updatedAt TEXT');
    }
    await _stampAndCreateTriggers(db, t);
  }
}

import 'dart:async';

import 'package:flutter/foundation.dart';

class _SyncTable {
  final String name;

  /// settings uses an INTEGER id locally; the cloud stores ids as text.
  final bool intId;
  const _SyncTable(this.name, {this.intId = false});
}

// Parents first, so children never arrive before the row they reference.
const _tables = [
  _SyncTable('students'),
  _SyncTable('settings', intId: true),
  _SyncTable('subjects'),
  _SyncTable('topics'),
  _SyncTable('study_sessions'),
  _SyncTable('tasks'),
  _SyncTable('exams'),
  _SyncTable('exam_topics'),

/// UTC ISO-8601 with millisecond precision — the same shape SQLite writes,
/// so plain string comparison orders timestamps correctly.
String _iso(DateTime d) => DateTime.fromMillisecondsSinceEpoch(
      d.millisecondsSinceEpoch,
      isUtc: true,
    ).toIso8601String();

class SyncService {
  SyncService._();
  static final SyncService instance = SyncService._();

  static const _pageSize = 500;

  SupabaseClient get _client => Supabase.instance.client;

  bool _running = false;
  bool _runAgain = false;
  Timer? _debounce;
  Timer? _poller;
  RealtimeChannel? _channel;
  StreamSubscription<AuthState>? _authSub;

  /// Call once after Supabase.initialize().
  Future<void> start() async {
    await _authSub?.cancel();
    _authSub = _client.auth.onAuthStateChange.listen((state) {
      if (state.session != null) {
        _subscribeRealtime();
        scheduleSync();
      } else {
        _unsubscribeRealtime();
      }
    });

    // Picks up local edits (repositories don't notify the sync layer).
    _poller ??= Timer.periodic(
      const Duration(seconds: 30),
      (_) => scheduleSync(),
    );

    if (_client.auth.currentUser != null) {
      _subscribeRealtime();
      scheduleSync();
    }
  }

  void stop() {
    _poller?.cancel();
    _poller = null;
    _debounce?.cancel();
    _authSub?.cancel();
    _unsubscribeRealtime();
  }

  /// Debounced: many quick calls collapse into one sync.
  void scheduleSync() {
    _debounce?.cancel();
    _debounce = Timer(const Duration(seconds: 2), syncNow);
  }

  Future<void> syncNow() async {
    final user = _client.auth.currentUser;
    if (user == null) return;
    if (_running) {
      _runAgain = true;
      return;
    }
    _running = true;
    try {
      final changed = await _pull(user.id); // cloud -> local
      await _push(user.id); // local -> cloud
      // Only when the pull actually changed local data, refresh every screen.
      if (changed.isNotEmpty) {
        DataRefreshBus.instance.notify(RefreshScope.all);
      }
    } catch (e) {
      // Offline or server error: nothing is lost, the next run retries.
      debugPrint('Sync failed: $e');
    } finally {
      _running = false;
      if (_runAgain) {
        _runAgain = false;
        scheduleSync();
      }
    }
  }

  // ---------------------------------------------------------------- realtime

  void _subscribeRealtime() {
    _unsubscribeRealtime();
    final channel = _client.channel('study-architect-sync');
    for (final t in _tables) {
      channel.onPostgresChanges(
        event: PostgresChangeEvent.all,
        schema: 'public',
        table: t.name,
        callback: (_) => scheduleSync(),
      );
    }
    channel.subscribe();
    _channel = channel;
  }

  void _unsubscribeRealtime() {
    final channel = _channel;
    if (channel != null) _client.removeChannel(channel);
    _channel = null;
  }

  // -------------------------------------------------------------------- meta

  Future<String?> _getMeta(Database db, String key) async {
    final rows =
        await db.query('sync_meta', where: 'key = ?', whereArgs: [key]);
    return rows.isEmpty ? null : rows.first['value'] as String;
  }

  Future<void> _setMeta(Database db, String key, String value) {
    return db.insert(
      'sync_meta',
      {'key': key, 'value': value},
      conflictAlgorithm: ConflictAlgorithm.replace,
    );
  }

  // -------------------------------------------------------------------- pull

  /// Downloads rows changed in the cloud since the last pull.
  /// Returns the names of local tables that were modified.
  Future<Set<String>> _pull(String uid) async {
    final db = await DatabaseService.instance.database;
    final touched = <String>{};

    for (final t in _tables) {
      final key = 'pull:$uid:${t.name}';
      var cursor = await _getMeta(db, key);

      while (true) {
        final base = _client.from(t.name).select();
        final rows =
            await (cursor == null ? base : base.gt('updatedAt', cursor))
                .order('updatedAt', ascending: true)
                .limit(_pageSize);

        for (final raw in rows) {
          final row = Map<String, dynamic>.from(raw);
          try {
            if (await _applyRemote(db, t, row)) touched.add(t.name);
          } catch (e) {
            debugPrint('Skipped ${t.name} row ${row['id']}: $e');
          }
          cursor = _iso(DateTime.parse(row['updatedAt'] as String));
        }

        if (cursor != null) await _setMeta(db, key, cursor);
        if (rows.length < _pageSize) break;
      }
    }
    return touched;
  }

  /// Applies one cloud row with last-write-wins. Returns true if the
  /// local table changed.
  Future<bool> _applyRemote(
    Database db,
    _SyncTable t,
    Map<String, dynamic> remote,
  ) async {
    final id = remote['id'] as String;
    final localId = t.intId ? int.parse(id) : id;
    final remoteAt = _iso(DateTime.parse(remote['updatedAt'] as String));

    final local = await db.query(
      t.name,
      columns: ['updatedAt'],
      where: 'id = ?',
      whereArgs: [localId],
    );
    final localAt = local.isEmpty ? null : local.first['updatedAt'] as String?;

    // Local copy is newer or identical: keep it.
    if (localAt != null && localAt.compareTo(remoteAt) >= 0) return false;

    // Cloud row was deleted.
    if (remote['deletedAt'] != null) {
      await db.delete(t.name, where: 'id = ?', whereArgs: [localId]);
      await db.delete(
        'sync_tombstones',
        where: 'tableName = ? AND rowId = ?',
        whereArgs: [t.name, id],
      );
      return local.isNotEmpty;
    }

    // Deleted locally after the cloud version was written: keep it deleted.
    final tomb = await db.query(
      'sync_tombstones',
      where: 'tableName = ? AND rowId = ?',
      whereArgs: [t.name, id],
    );
    if (tomb.isNotEmpty &&
        (tomb.first['deletedAt'] as String).compareTo(remoteAt) >= 0) {
      return false;
    }

    final values = Map<String, dynamic>.from(remote)
      ..remove('user_id')
      ..remove('deletedAt')
      ..['id'] = localId
      ..['updatedAt'] = remoteAt;

    if (local.isEmpty) {
      await db.insert(t.name, values);
    } else {
      // update, not "insert or replace": REPLACE would delete the row first
      // and cascade-delete its children.
      await db.update(t.name, values, where: 'id = ?', whereArgs: [localId]);
    }
    return true;
  }

  // -------------------------------------------------------------------- push

  Future<void> _push(String uid) async {
    final db = await DatabaseService.instance.database;

    // 1) New and edited rows.
    for (final t in _tables) {
      final key = 'push:$uid:${t.name}';
      final since = await _getMeta(db, key);

      final rows = await db.query(
        t.name,
        where: since == null ? null : 'updatedAt > ?',
        whereArgs: since == null ? null : [since],
        orderBy: 'updatedAt ASC',
      );
      if (rows.isEmpty) continue;

      final payload = rows
          .map((r) => <String, dynamic>{
                ...r,
                'id': r['id'].toString(),
                'user_id': uid,
                'deletedAt': null,
              })
          .toList();

      for (var i = 0; i < payload.length; i += 200) {
        final end = i + 200 > payload.length ? payload.length : i + 200;
        await _client
            .from(t.name)
            .upsert(payload.sublist(i, end), onConflict: 'user_id,id');
      }
      await _setMeta(db, key, rows.last['updatedAt'] as String);
    }

    // 2) Deletes.
    final tombstones = await db.query('sync_tombstones');
    for (final tomb in tombstones) {
      final table = tomb['tableName'] as String;
      final rowId = tomb['rowId'] as String;
      final at = tomb['deletedAt'] as String;

      await _client
          .from(table)
          .update({'deletedAt': at, 'updatedAt': at})
          .eq('user_id', uid)
          .eq('id', rowId);

      await db.delete(
        'sync_tombstones',
        where: 'tableName = ? AND rowId = ?',
        whereArgs: [table, rowId],
      );
    }
  }
}

/// Small date helpers shared by Home, Timetable, and Progress —
/// keeps "start of day" / "start of week" logic in one place.
DateTime startOfDay(DateTime date) => DateTime(date.year, date.month, date.day);

/// Monday-start week, matching the spec's "Week • September 8–14" style.
DateTime startOfWeek(DateTime date) {
  final day = startOfDay(date);
  return day.subtract(Duration(days: day.weekday - DateTime.monday));
}

DateTime endOfWeek(DateTime date) =>
    startOfWeek(date).add(const Duration(days: 7));

bool isSameDay(DateTime a, DateTime b) =>
    a.year == b.year && a.month == b.month && a.day == b.day;

String formatDuration(int minutes) {
  final h = minutes ~/ 60;
  final m = minutes % 60;
  if (h == 0) return '${m}m';
  if (m == 0) return '${h}h';
  return '${h}h ${m}m';
}

String formatTimeOfDay(DateTime time) {
  final hour = time.hour % 12 == 0 ? 12 : time.hour % 12;
  final minute = time.minute.toString().padLeft(2, '0');
  final period = time.hour >= 12 ? 'PM' : 'AM';
  return '$hour:$minute $period';
}

import 'package:uuid/uuid.dart';
import '../models/study_session.dart';

/// Hard cap so a weekly repeat can never create an unbounded number of rows.
const int kMaxWeeklyOccurrences = 52;

DateTime _dayOnly(DateTime d) => DateTime(d.year, d.month, d.day);

/// How many weekly occurrences fall between [first] and [until] (inclusive).
/// Always at least 1 and never more than [maxOccurrences].
int countWeeklyOccurrences(
  DateTime first,
  DateTime until, {
  int maxOccurrences = kMaxWeeklyOccurrences,
}) {
  final start = _dayOnly(first);
  final end = _dayOnly(until);
  var count = 0;
  for (var week = 0; week < maxOccurrences; week++) {
    final date = DateTime(start.year, start.month, start.day + 7 * week);
    if (week > 0 && date.isAfter(end)) break;
    count++;
  }
  return count;
}

/// Expands one session into a weekly series: the same weekday and the
/// same start/end time, every 7 days, up to and including [until].
/// The first element keeps [first]'s id; the others get new ids.
/// All of them share one seriesId.
List<StudySession> buildWeeklySeries({
  required StudySession first,
  required DateTime until,
  int maxOccurrences = kMaxWeeklyOccurrences,
}) {
  final seriesId = first.seriesId ?? const Uuid().v4();
  final start = _dayOnly(first.scheduledDate);
  final end = _dayOnly(until);
  final result = <StudySession>[];

  for (var week = 0; week < maxOccurrences; week++) {
    final date = DateTime(start.year, start.month, start.day + 7 * week);
    if (week > 0 && date.isAfter(end)) break;

    result.add(
      StudySession(
        id: week == 0 ? first.id : const Uuid().v4(),
        subjectId: first.subjectId,
        topicId: first.topicId,
        scheduledDate: date,
        startTime: DateTime(
          date.year,
          date.month,
          date.day,
          first.startTime.hour,
          first.startTime.minute,
        ),
        endTime: DateTime(
          date.year,
          date.month,
          date.day,
          first.endTime.hour,
          first.endTime.minute,
        ),
        status: SessionStatus.planned,
        notes: first.notes,
        recurrence: SessionRecurrence.weekly,
        recurrenceUntil: end,
        seriesId: seriesId,
      ),
    );
  }
  return result;

import 'package:flutter/material.dart';

class StudyInsight {
  final IconData icon;
  final String text;

  const StudyInsight({required this.icon, required this.text});
}

/// Deterministic, rule-based insights (no ML) — the V1 spec calls for
/// "rule-based study insights" specifically. Each rule is independent
/// and only fires when its condition is clearly true, so the list can
/// safely be empty on a quiet day.
List<StudyInsight> generateInsights({
  required int todayMinutes,
  required int weeklyMinutes,
  required int weeklyGoalMinutes,
  required int streak,
  required int longestStreak,
  required Map<String, int> subjectPerformancePercent,
}) {
  final insights = <StudyInsight>[];
  final now = DateTime.now();

  // Hasn't studied yet today, and the day is more than half over.
  if (todayMinutes == 0 && now.hour >= 15) {
    insights.add(const StudyInsight(
      icon: Icons.schedule_rounded,
      text:
          "You haven't studied yet today — even 20 minutes keeps your streak alive.",
    ));
  }

  // Behind pace on the weekly goal once the week is more than half done.
  if (weeklyGoalMinutes > 0 && now.weekday >= DateTime.thursday) {
    final progress = weeklyMinutes / weeklyGoalMinutes;
    if (progress < 0.5) {
      insights.add(StudyInsight(
        icon: Icons.trending_down_rounded,
        text:
            "You're at ${(progress * 100).round()}% of this week's goal with the week more than half over.",
      ));
    }
  }

  // Subjects clearly falling behind their own weekly target — collapsed
  // into a single line instead of one card per subject. Previously
  // this looped and added one insight *per* behind-target subject,
  // which meant a normal early-week state (nothing logged yet, so
  // every subject reads 0%) produced a card for every single subject
  // — pure noise, not an insight.
  final behindSubjects = subjectPerformancePercent.entries
      .where((e) => e.value < 40)
      .toList()
    ..sort((a, b) => a.value.compareTo(b.value));

  if (behindSubjects.isNotEmpty) {
    if (weeklyMinutes == 0 &&
        behindSubjects.length == subjectPerformancePercent.length) {
      // Nothing logged at all yet this period — a single general
      // nudge is more useful than naming every subject.
      insights.add(const StudyInsight(
        icon: Icons.flag_outlined,
        text: "You haven't logged any study time yet this period.",
      ));
    } else {
      final shown = behindSubjects.take(2).map((e) => e.key).join(', ');
      final remaining = behindSubjects.length - 2;
      final verb = behindSubjects.length == 1 ? 'is' : 'are';
      final suffix = remaining > 0 ? ', and $remaining more,' : '';
      insights.add(StudyInsight(
        icon: Icons.flag_outlined,
        text: '$shown$suffix $verb behind on weekly targets.',
      ));
    }
  }

  // Reward consistency.
  if (streak >= 7) {
    insights.add(StudyInsight(
      icon: Icons.local_fire_department_rounded,
      text:
          "You're on a $streak-day streak — your longest is $longestStreak days.",
    ));
  } else if (streak == 0 && longestStreak > 0) {
    insights.add(StudyInsight(
      icon: Icons.replay_rounded,
      text:
          'Your streak reset. Your best was $longestStreak days — start a new one today.',
    ));
  }

  return insights;
}

import '../models/study_session.dart';
import 'date_utils.dart';

/// Sum of actualDuration (minutes) across completed sessions whose
/// scheduledDate falls on [date].
int completedMinutesForDate(List<StudySession> sessions, DateTime date) {
  return sessions
      .where((s) => s.complete && isSameDay(s.scheduledDate, date))
      .fold(0, (sum, s) => sum + (s.actualDuration ?? 0));
}

/// Sum of actualDuration across completed sessions within [sessions],
/// with no date filtering — pass in a week's worth of sessions to get
/// a weekly total.
int completedMinutesTotal(List<StudySession> sessions) {
  return sessions
      .where((s) => s.complete)
      .fold(0, (sum, s) => sum + (s.actualDuration ?? 0));
}

/// Current study streak, per the V1 business rule: a day counts only
/// if it has at least one completed session.
///
/// [completedDates] are yyyy-MM-dd strings from
/// SessionRepository.getCompletedDates(). Today not yet having a
/// completed session does not break the streak — it simply isn't
/// counted yet, since the day isn't over.
int currentStreak(Set<String> completedDates, {DateTime? today}) {
  final now = today ?? DateTime.now();
  var cursor = startOfDay(now);
  var streak = 0;

  bool hasSessionOn(DateTime day) => completedDates.contains(_dateKey(day));

  // If today already has a completed session, count it and step back
  // from yesterday. If not, today is skipped (not yet failed) and we
  // start checking from yesterday.
  if (hasSessionOn(cursor)) {
    streak++;
  }
  cursor = cursor.subtract(const Duration(days: 1));

  while (hasSessionOn(cursor)) {
    streak++;
    cursor = cursor.subtract(const Duration(days: 1));
  }

  return streak;
}

String _dateKey(DateTime date) {
  final y = date.year.toString().padLeft(4, '0');
  final m = date.month.toString().padLeft(2, '0');
  final d = date.day.toString().padLeft(2, '0');
  return '$y-$m-$d';
}

/// Longest run of consecutive completed-session days on record, e.g.
/// for the "Longest: 18 days" line on the Progress screen.
int longestStreak(Set<String> completedDates) {
  if (completedDates.isEmpty) return 0;

  final sortedDates = completedDates.map((key) {
    final parts = key.split('-');
    return DateTime(
        int.parse(parts[0]), int.parse(parts[1]), int.parse(parts[2]));
  }).toList()
    ..sort();

  var longest = 1;
  var current = 1;
  for (var i = 1; i < sortedDates.length; i++) {
    final gap = sortedDates[i].difference(sortedDates[i - 1]).inDays;
    if (gap == 1) {
      current++;
      longest = current > longest ? current : longest;
    } else if (gap > 1) {
      current = 1;
    }
  }
  return longest;
}
