/// Result of looking for the person's profile in the cloud.
enum _CloudCheck { restored, noProfile, unreachable }

/// The app's actual root widget.
///
/// Checks whether a Student profile already exists in local storage.
/// If no profile exists, onboarding is shown.
/// If a profile exists, the main tab shell is shown.
///
/// V2: if the local database is empty but the user is signed in, the
/// profile may already be in the cloud (new phone, reinstall, cleared
/// data). The cloud is asked directly:
/// - active profile found -> it is pulled (with retries) before the app opens
/// - no active profile    -> a genuinely new user, onboarding is shown
///   (a profile that was deleted is kept in the cloud with a deletedAt
///   date and is NOT restored)
/// - cloud unreachable    -> a "could not reach your account" screen, so a
///   bad network never sends a returning user into profile building
///   (which would create a second profile).
class StartupDecider extends StatefulWidget {
  const StartupDecider({super.key});

  @override
  State<StartupDecider> createState() => _StartupDeciderState();
}

class _StartupDeciderState extends State<StartupDecider> {
  final _studentRepository = StudentRepository();

  bool? _onboardingComplete;
  bool _cloudCheckFailed = false;
  Object? _error;

  @override
  void initState() {
    super.initState();
    _checkOnboardingStatus();
  }

  bool get _signedIn =>
      AuthService.isAvailable &&
      Supabase.instance.client.auth.currentUser != null;

  /// Asks the cloud whether this account already has an ACTIVE profile
  /// and, if so, pulls it onto this phone.
  Future<_CloudCheck> _restoreFromCloud() async {
    final client = Supabase.instance.client;
    final user = client.auth.currentUser;
    if (user == null) return _CloudCheck.unreachable;

    try {
      final rows = await client
          .from('students')
          .select('id, deletedAt')
          .eq('user_id', user.id)
          .timeout(const Duration(seconds: 10));
      final active = rows.where((r) => r['deletedAt'] == null).length;
      debugPrint(
        'StartupDecider: cloud profiles total=${rows.length} active=$active',
      );
      if (active == 0) return _CloudCheck.noProfile;
    } catch (e) {
      debugPrint('StartupDecider cloud check failed: $e');
      return _CloudCheck.unreachable;
    }

    // The cloud has an active profile: pull it. syncNow() never throws and
    // returns at once if another sync is already running, so keep
    // checking the local database until the profile arrives (up to 60 s,
    // because the first full sync can be slow on mobile data).
    final deadline = DateTime.now().add(const Duration(seconds: 60));
    var attempt = 0;
    while (DateTime.now().isBefore(deadline)) {
      attempt++;
      try {
        await SyncService.instance
            .syncNow()
            .timeout(const Duration(seconds: 10));
      } catch (e) {
        debugPrint('StartupDecider sync attempt $attempt skipped: $e');
      }
      final local = await _studentRepository.getCurrent();
      debugPrint(
        'StartupDecider: attempt $attempt, local profile present: ${local != null}',
      );
      if (local != null) return _CloudCheck.restored;
      await Future<void>.delayed(const Duration(seconds: 2));
    }
    debugPrint('StartupDecider: profile did not arrive locally after retries');
    return _CloudCheck.unreachable;
  }

  Future<void> _checkOnboardingStatus() async {
    if (!mounted) return;

    setState(() {
      _onboardingComplete = null;
      _cloudCheckFailed = false;
      _error = null;
    });

    try {
      var student = await _studentRepository.getCurrent();

      if (student == null && _signedIn) {
        final result = await _restoreFromCloud();
        student = await _studentRepository.getCurrent();

        if (student == null && result == _CloudCheck.unreachable) {
          if (!mounted) return;
          setState(() {
            _cloudCheckFailed = true;
          });
          return;
        }
      }

      if (!mounted) return;

      setState(() {
        _onboardingComplete = student != null;
      });
    } catch (e, stackTrace) {
      debugPrint(
        'StartupDecider failed to load student: $e\n$stackTrace',
      );

      if (!mounted) return;

      setState(() {
        _error = e;
      });
    }
  }

  Future<void> _handleOnboardingComplete() async {
    await _checkOnboardingStatus();
  }

  @override
  Widget build(BuildContext context) {
    if (_error != null) {
      return GradientBackground(
        child: SafeArea(
          child: Padding(
            padding: const EdgeInsets.all(AppSpacing.lg),
            child: Center(
              child: SingleChildScrollView(
                child: Column(
                  mainAxisSize: MainAxisSize.min,
                  crossAxisAlignment: CrossAxisAlignment.center,
                  children: [
                    const Icon(
                      Icons.error_outline_rounded,
                      size: 48,
                    ),
                    const SizedBox(height: AppSpacing.md),
                    Text(
                      'Could not start Study Architect.',
                      style: Theme.of(context).textTheme.titleLarge,
                      textAlign: TextAlign.center,
                    ),
                    const SizedBox(height: AppSpacing.sm),
                    Text(
                      'We could not load your local profile. '
                      'Please try again.',
                      textAlign: TextAlign.center,
                      style: Theme.of(context).textTheme.bodyMedium?.copyWith(
                            color: context.secondaryText,
                          ),
                    ),
                    const SizedBox(height: AppSpacing.lg),
                    FilledButton.icon(
                      onPressed: _checkOnboardingStatus,
                      icon: const Icon(Icons.refresh_rounded),
                      label: const Text('Try Again'),
                    ),
                  ],
                ),
              ),
            ),
          ),
        ),
      );
    }

    if (_cloudCheckFailed) {
      return GradientBackground(
        child: SafeArea(
          child: Padding(
            padding: const EdgeInsets.all(AppSpacing.lg),
            child: Center(
              child: SingleChildScrollView(
                child: Column(
                  mainAxisSize: MainAxisSize.min,
                  crossAxisAlignment: CrossAxisAlignment.center,
                  children: [
                    const Icon(
                      Icons.cloud_off_rounded,
                      size: 48,
                    ),
                    const SizedBox(height: AppSpacing.md),
                    Text(
                      'Could not reach your account.',
                      style: Theme.of(context).textTheme.titleLarge,
                      textAlign: TextAlign.center,
                    ),
                    const SizedBox(height: AppSpacing.sm),
                    Text(
                      'Your profile may already be saved in your account. '
                      'Check your connection and try again.',
                      textAlign: TextAlign.center,
                      style: Theme.of(context).textTheme.bodyMedium?.copyWith(
                            color: context.secondaryText,
                          ),
                    ),
                    const SizedBox(height: AppSpacing.lg),
                    FilledButton.icon(
                      onPressed: _checkOnboardingStatus,
                      icon: const Icon(Icons.refresh_rounded),
                      label: const Text('Try Again'),
                    ),
                    const SizedBox(height: AppSpacing.sm),
                    TextButton(
                      onPressed: () {
                        setState(() {
                          _cloudCheckFailed = false;
                          _onboardingComplete = false;
                        });
                      },
                      child: const Text('Set up a new profile instead'),
                    ),
                  ],
                ),
              ),
            ),
          ),
        ),
      );
    }

    if (_onboardingComplete == null) {
      return const SplashView();
    }

    if (_onboardingComplete == false) {
      return OnboardingFlow(
        onComplete: _handleOnboardingComplete,
      );
    }

    return const MainScaffold();
  }
}

/// Debug-only screen that tests the auth system beyond "did login work?".
/// Never prints tokens. Safe to keep in the project; it is only reachable
/// from the More screen when running in debug mode.
class AuthDiagnosticsScreen extends StatefulWidget {
  const AuthDiagnosticsScreen({super.key});

  @override
  State<AuthDiagnosticsScreen> createState() => _AuthDiagnosticsScreenState();
}

class _Check {
  final String name;
  final bool ok;
  final String detail;
  const _Check(this.name, this.ok, this.detail);
}

class _AuthDiagnosticsScreenState extends State<AuthDiagnosticsScreen> {
  final List<_Check> _checks = [];
  final List<String> _events = [];
  StreamSubscription<AuthState>? _sub;
  bool _running = false;

  @override
  void initState() {
    super.initState();
    if (AuthService.isAvailable) {
      _sub = AuthService.instance.authStateChanges.listen((state) {
        if (!mounted) return;
        setState(() => _events.insert(0, '${_now()}  ${state.event.name}'));
      });
    }
    _runChecks();
  }

  @override
  void dispose() {
    _sub?.cancel();
    super.dispose();
  }

  String _now() {
    final t = DateTime.now();
    String two(int n) => n.toString().padLeft(2, '0');
    return '${two(t.hour)}:${two(t.minute)}:${two(t.second)}';
  }

  String _timeLeft(Duration d) {
    if (d.isNegative) return 'already expired';
    if (d.inMinutes < 1) return 'in under a minute';
    if (d.inHours < 1) return 'in ${d.inMinutes} min';
    return 'in ${d.inHours} h ${d.inMinutes % 60} min';
  }

  Future<void> _runChecks() async {
    setState(() {
      _running = true;
      _checks.clear();
    });

    final results = <_Check>[];

    void finish() {
      if (!mounted) return;
      setState(() {
        _checks
          ..clear()
          ..addAll(results);
        _running = false;
      });
    }

    // 1. Client initialized
    results.add(_Check(
      'Supabase initialized',
      AuthService.isAvailable,
      AuthService.isAvailable
          ? 'Client is ready'
          : 'main() could not start Supabase. Check the .env file.',
    ));
    if (!AuthService.isAvailable) {
      finish();
      return;
    }

    final client = Supabase.instance.client;
    final session = client.auth.currentSession;
    final user = client.auth.currentUser;

    // 2. Session exists
    results.add(_Check(
      'Session present',
      session != null && user != null,
      session != null
          ? 'A session is stored on this device'
          : 'No one is signed in',
    ));
    if (session == null || user == null) {
      finish();
      return;
    }

    // 3. Account info
    results.add(_Check(
      'Signed-in account',
      true,
      '${user.email ?? 'no email'}\nid: ${user.id}',
    ));

    // 4. Token expiry
    final expiry = DateTime.fromMillisecondsSinceEpoch(
      (session.expiresAt ?? 0) * 1000,
    );
    results.add(_Check(
      'Access token valid',
      !session.isExpired,
      'Expires ${_timeLeft(expiry.difference(DateTime.now()))}',
    ));

    // 5. Server accepts the token
    try {
      final res = await client.auth.getUser();
      final same = res.user?.id == user.id;
      results.add(_Check(
        'Server accepts the token',
        same,
        same
            ? 'Supabase verified this user'
            : 'Server returned a different user',
      ));
    } catch (e) {
      results.add(_Check(
        'Server accepts the token',
        false,
        AuthService.friendlyError(e),
      ));
    }

    // 6. Token refresh works
    try {
      final res = await client.auth.refreshSession();
      results.add(_Check(
        'Token refresh works',
        res.session != null,
        res.session != null
            ? 'New token issued, expires ${_timeLeft(DateTime.fromMillisecondsSinceEpoch((res.session!.expiresAt ?? 0) * 1000).difference(DateTime.now()))}'
            : 'Refresh returned no session',
      ));
    } catch (e) {
      results.add(_Check(
        'Token refresh works',
        false,
        AuthService.friendlyError(e),
      ));
    }

    // 7. Linked sign-in methods
    final providers = (user.identities ?? []).map((i) => i.provider).toSet();
    results.add(_Check(
      'Linked sign-in methods',
      providers.isNotEmpty,
      providers.isEmpty ? 'None reported' : providers.join(', '),
    ));

    // 8. Email confirmed
    results.add(_Check(
      'Email confirmed',
      user.emailConfirmedAt != null,
      user.emailConfirmedAt != null
          ? 'Confirmed'
          : 'Not confirmed (expected only if email confirmation is off)',
    ));

    finish();
  }

  Future<void> _copyReport() async {
    final buffer = StringBuffer('Auth diagnostics ${DateTime.now()}\n');
    for (final c in _checks) {
      buffer.writeln('[${c.ok ? 'OK' : 'FAIL'}] ${c.name}: ${c.detail}');
    }
    await Clipboard.setData(ClipboardData(text: buffer.toString()));
    if (!mounted) return;
    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(content: Text('Report copied to clipboard')),
    );
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Auth diagnostics'),
        actions: [
          IconButton(
            tooltip: 'Copy report',
            onPressed: _checks.isEmpty ? null : _copyReport,
            icon: const Icon(Icons.copy_rounded),
          ),
        ],
      ),
      body: ListView(
        padding: const EdgeInsets.all(16),
        children: [
          FilledButton.icon(
            onPressed: _running ? null : _runChecks,
            icon: _running
                ? const SizedBox(
                    width: 18,
                    height: 18,
                    child: CircularProgressIndicator(strokeWidth: 2),
                  )
                : const Icon(Icons.play_arrow_rounded),
            label: Text(_running ? 'Running checks...' : 'Run checks again'),
          ),
          const SizedBox(height: 16),
          for (final c in _checks)
            Card(
              child: ListTile(
                leading: Icon(
                  c.ok ? Icons.check_circle_rounded : Icons.cancel_rounded,
                  color: c.ok ? Colors.green : Colors.red,
                ),
                title: Text(c.name),
                subtitle: Text(c.detail),
              ),
            ),
          const SizedBox(height: 16),
          Text(
            'Auth events since this screen opened',
            style: Theme.of(context).textTheme.titleSmall,
          ),
          const SizedBox(height: 8),
          if (_events.isEmpty)
            const Text('No events yet.')
          else
            for (final e in _events) Text(e),
        ],
      ),
    );
  }
}

/// Decides what the user sees:
/// - Supabase not ready         -> your app shell directly (local-first fallback)
/// - password-reset link opened -> Set password screen
/// - session exists             -> your signed-in app shell, wrapped in the
///                                 password prompt (bottom sheet on entry)
/// - no session, but a profile is saved on this phone and the person did
///   not tap Sign out          -> the app shell in offline mode (an expired
///                                 login cannot refresh without internet)
/// - no session otherwise       -> Sign In screen
/// It listens to Supabase, so sign-in, sign-out and expired sessions
/// switch screens automatically with no manual navigation.
class AuthGate extends StatefulWidget {
  /// Your existing V1 shell (the screen with Home â€¢ Timetable â€¢ + â€¢ Subjects â€¢ More).
  final WidgetBuilder signedInBuilder;

  const AuthGate({super.key, required this.signedInBuilder});

  @override
  State<AuthGate> createState() => _AuthGateState();
}

class _AuthGateState extends State<AuthGate> {
  bool _recovering = false;
  bool _offlineNoticeShown = false;
  Future<bool>? _localAccessCheck;
  StreamSubscription<AuthState>? _sub;

  @override
  void initState() {
    super.initState();
    if (AuthService.isAvailable) {
      if (AuthService.instance.currentSession != null) {
        unawaited(OfflineAccess.clearSignedOut());
      }
      _sub = AuthService.instance.authStateChanges.listen((state) {
        if (!mounted) return;
        debugPrint(
          'AuthGate event: ${state.event.name}, session: ${state.session != null}',
        );
        if (state.session != null) {
          _offlineNoticeShown = false;
          unawaited(OfflineAccess.clearSignedOut());
        } else {
          // Re-check the saved-profile rule on the next build.
          _localAccessCheck = null;
        }
        if (state.event == AuthChangeEvent.passwordRecovery) {
          setState(() => _recovering = true);
        } else if (state.event == AuthChangeEvent.signedOut && _recovering) {
          setState(() => _recovering = false);
        }
      });
    }
  }

  @override
  void dispose() {
    _sub?.cancel();
    super.dispose();
  }

  void _showOfflineNotice() {
    if (_offlineNoticeShown) return;
    _offlineNoticeShown = true;
    WidgetsBinding.instance.addPostFrameCallback((_) {
      if (!mounted) return;
      ScaffoldMessenger.maybeOf(context)?.showSnackBar(
        const SnackBar(
          duration: Duration(seconds: 6),
          content: Text(
            'Working offline. Your data is saved on this phone. '
            'Syncing resumes when you are back online and signed in.',
          ),
        ),
      );
    });
  }

  @override
  Widget build(BuildContext context) {
    // Supabase failed to start (missing .env, bad key...): keep the
    // local-first app usable instead of trapping the user on a login
    // screen that can never succeed.
    if (!AuthService.isAvailable) {
      return widget.signedInBuilder(context);
    }

    return StreamBuilder<AuthState>(
      stream: AuthService.instance.authStateChanges,
      initialData: AuthState(
        AuthChangeEvent.initialSession,
        AuthService.instance.currentSession,
      ),
      builder: (context, snapshot) {
        final session = snapshot.data?.session;
        if (session == null) {
          _localAccessCheck ??= OfflineAccess.canUseWithoutSession();
          return FutureBuilder<bool>(
            future: _localAccessCheck,
            builder: (context, local) {
              if (local.connectionState != ConnectionState.done) {
                return const Scaffold(
                  body: Center(child: CircularProgressIndicator()),
                );
              }
              if (local.data == true) {
                _showOfflineNotice();
                return widget.signedInBuilder(context);
              }
              return const SignInScreen();
            },
          );
        }
        if (_recovering) {
          return SetPasswordScreen(
            recoveryMode: true,
            onDone: () {
              if (mounted) setState(() => _recovering = false);
            },
          );
        }
        return PasswordPromptGate(
          child: widget.signedInBuilder(context),
        );
      },
    );
  }
}

import 'package:flutter/foundation.dart' show kIsWeb;
import 'package:supabase_flutter/supabase_flutter.dart';
import 'package:url_launcher/url_launcher.dart' show LaunchMode;

import 'offline_access.dart';

/// Thin wrapper around Supabase Auth so no screen talks to Supabase directly.
class AuthService {
  AuthService._();
  static final AuthService instance = AuthService._();

  /// Set to true in main() only after Supabase.initialize succeeds.
  static bool isAvailable = false;

  /// True after the user taps "Not now" on the password banner.
  /// Lives in memory only, so the banner returns on the next app launch.
  static bool passwordPromptDismissed = false;

  SupabaseClient get _client => Supabase.instance.client;

  /// Deep link used to come back into the app after Google sign-in, after
  /// tapping the confirmation link in a sign-up email, or after tapping a
  /// password-reset link (Android/iOS/Windows).
  static const String mobileRedirect =
      'io.supabase.studyarchitect://login-callback/';

  User? get currentUser => _client.auth.currentUser;
  Session? get currentSession => _client.auth.currentSession;
  Stream<AuthState> get authStateChanges => _client.auth.onAuthStateChange;

  /// Whether the signed-in user can sign in with email + password.
  /// True when the account has an email identity (email sign-up) or when a
  /// password was set through the app (has_password marker in metadata).
  /// A Google-only account returns false.
  bool get hasPassword {
    final user = currentUser;
    if (user == null) return true;
    if (user.userMetadata?['has_password'] == true) return true;
    final identities = user.identities ?? [];
    return identities.any((i) => i.provider == 'email');
  }

  // ---------------------------------------------------------------------------
  // Email + password
  // ---------------------------------------------------------------------------

  /// Creates an account with email and password.
  ///
  /// - emailRedirectTo sends the confirmation link back into the app
  ///   (without it the link opens the Site URL, which is localhost).
  /// - When "Confirm email" is ON, Supabase answers a sign-up for an address
  ///   that already exists with a fake success (no error, empty identities)
  ///   so attackers cannot discover registered emails. We detect that case and
  ///   raise a clear error instead of showing "check your inbox" for nothing.
  Future<AuthResponse> signUpWithEmail({
    required String fullName,
    required String email,
    required String password,
  }) async {
    final res = await _client.auth.signUp(
      email: email.trim(),
      password: password,
      data: {'full_name': fullName.trim(), 'has_password': true},
      emailRedirectTo: kIsWeb ? null : mobileRedirect,
    );

    final identities = res.user?.identities;
    if (res.user != null && identities != null && identities.isEmpty) {
      throw AuthException('User already registered');
    }
    return res;
  }

  Future<AuthResponse> signInWithEmail({
    required String email,
    required String password,
  }) {
    return _client.auth.signInWithPassword(
      email: email.trim(),
      password: password,
    );
  }

  Future<void> sendPasswordReset(String email) {
    return _client.auth.resetPasswordForEmail(
      email.trim(),
      redirectTo: kIsWeb ? null : mobileRedirect,
    );
  }

  /// Sets a new password for the account that is signed in right now.
  /// Used both after a reset link and from More > Password.
  /// For a Google-only account this adds a password, so the same email
  /// can also be used in the email sign-in form. The has_password marker
  /// stops the "Add a password" banner from showing again.
  Future<UserResponse> updatePassword(String newPassword) {
    return _client.auth.updateUser(
      UserAttributes(
        password: newPassword,
        data: {'has_password': true},
      ),
    );
  }

  // ---------------------------------------------------------------------------
  // OAuth providers (Google now, others later)
  // ---------------------------------------------------------------------------

  /// Shared launcher so every provider behaves the same way.
  /// In-app browser (Chrome Custom Tab on Android) returns to the app
  /// reliably; a normal browser tab often stalls on a blank page.
  Future<bool> _signInWithOAuth(OAuthProvider provider) {
    return _client.auth.signInWithOAuth(
      provider,
      redirectTo: kIsWeb ? null : mobileRedirect,
      authScreenLaunchMode:
          kIsWeb ? LaunchMode.platformDefault : LaunchMode.inAppBrowserView,
    );
  }

  Future<bool> signInWithGoogle() => _signInWithOAuth(OAuthProvider.google);

  // Later, adding a provider is one line once it is enabled in Supabase:
  // Future<bool> signInWithGitHub() => _signInWithOAuth(OAuthProvider.github);

  /// Attaches another sign-in method to the account that is signed in now.
  /// Requires "Allow manual linking" to be enabled in Supabase.
  Future<bool> linkProvider(OAuthProvider provider) {
    return _client.auth.linkIdentity(
      provider,
      redirectTo: kIsWeb ? null : mobileRedirect,
      authScreenLaunchMode:
          kIsWeb ? LaunchMode.platformDefault : LaunchMode.inAppBrowserView,
    );
  }

  // ---------------------------------------------------------------------------
  // Session
  // ---------------------------------------------------------------------------

  Future<void> signOut() async {
    passwordPromptDismissed = false;
    // Remember that this sign-out was the user's choice, so the app does
    // not treat them as "offline" and skip the sign-in screen.
    await OfflineAccess.markSignedOut();
    await _client.auth.signOut();
  }

  // ---------------------------------------------------------------------------
  // Errors
  // ---------------------------------------------------------------------------

  /// Turns Supabase errors into short, human messages.
  static String friendlyError(Object error) {
    if (error is AuthException) {
      final m = error.message.toLowerCase();
      if (m.contains('invalid login credentials')) {
        return 'Incorrect email or password. If you created your account with Google, tap "Continue with Google" instead.';
      }
      if (m.contains('email not confirmed')) {
        return 'Please confirm your email first. Check your inbox.';
      }
      if (m.contains('already registered') ||
          m.contains('already been registered')) {
        return 'An account with this email already exists. Try signing in, or tap "Continue with Google" if you signed up with Google.';
      }
      if (m.contains('different from the old password') ||
          m.contains('same as the old password')) {
        return 'Choose a password that is different from your current one.';
      }
      if (m.contains('reauthentication')) {
        return 'For security, please sign out, sign in again, then change your password.';
      }
      if (m.contains('auth session missing')) {
        return 'Your session expired. Please sign in again.';
      }
      if (m.contains('password should be at least')) {
        return 'Password is too short. Use at least 6 characters.';
      }
      if (m.contains('unable to validate email') ||
          m.contains('invalid email')) {
        return 'That email address does not look valid.';
      }
      if (m.contains('signups not allowed') ||
          m.contains('signup is disabled')) {
        return 'Sign-ups are currently turned off.';
      }
      if (m.contains('provider is not enabled') ||
          m.contains('unsupported provider')) {
        return 'This sign-in method is not enabled yet.';
      }
      if (m.contains('rate limit') || m.contains('too many requests')) {
        return 'Too many attempts. Please wait a minute and try again.';
      }
      return error.message;
    }
    final text = error.toString().toLowerCase();
    if (text.contains('socket') ||
        text.contains('failed host lookup') ||
        text.contains('network')) {
      return 'No internet connection. Please check your network and retry.';
    }
    return 'Something went wrong. Please try again.';
  }

/// Wraps the signed-in app. A few seconds after entry it slides up a bottom
/// sheet inviting the user to add a password, when the account has none
/// (for example a Google-only account).
///
/// Rules:
/// - Shown at most once per app launch for each user.
/// - Skipped when the account already has a password.
/// - Skipped for accounts created in the last few minutes, so it does not
///   land on top of onboarding. Those users still get the banner on More and
///   see this sheet on their next launch.
/// - "Not now" only closes the sheet. The More banner stays as a reminder.
class PasswordPromptGate extends StatefulWidget {
  final Widget child;

  const PasswordPromptGate({super.key, required this.child});

  @override
  State<PasswordPromptGate> createState() => _PasswordPromptGateState();
}

class _PasswordPromptGateState extends State<PasswordPromptGate> {
  /// Id of the user who has already seen the sheet in this launch.
  static String? _promptedUserId;

  /// How long to wait after entering the app before showing the sheet.
  static const Duration _delay = Duration(milliseconds: 2500);

  /// Accounts younger than this are treated as brand new (likely onboarding).
  /// Set to Duration.zero while testing with a fresh account.
  static const Duration _newAccountGrace = Duration(minutes: 10);

  Timer? _timer;

  @override
  void initState() {
    super.initState();
    if (_eligible()) {
      _timer = Timer(_delay, _maybeShow);
    }
  }

  @override
  void dispose() {
    _timer?.cancel();
    super.dispose();
  }

  bool _eligible() {
    if (!AuthService.isAvailable) return false;
    final user = AuthService.instance.currentUser;
    if (user == null) return false;
    if (AuthService.instance.hasPassword) return false;
    if (_promptedUserId == user.id) return false;

    final created = DateTime.tryParse(user.createdAt);
    if (created != null &&
        DateTime.now().toUtc().difference(created.toUtc()) < _newAccountGrace) {
      return false;
    }
    return true;
  }

  Future<void> _maybeShow() async {
    if (!mounted || !_eligible()) return;

    // Do not cover another screen (a pushed page or a dialog).
    final route = ModalRoute.of(context);
    if (route != null && !route.isCurrent) return;

    _promptedUserId = AuthService.instance.currentUser?.id;

    final wantsToSet = await showModalBottomSheet<bool>(
      context: context,
      backgroundColor: context.cardSurface,
      isScrollControlled: true,
      showDragHandle: true,
      shape: const RoundedRectangleBorder(
        borderRadius: BorderRadius.vertical(top: Radius.circular(24)),
      ),
      builder: (sheetContext) => const _PasswordSheet(),
    );

    if (wantsToSet == true && mounted) {
      await Navigator.of(context).push(
        MaterialPageRoute(builder: (_) => const SetPasswordScreen()),
      );
    }
  }

  @override
  Widget build(BuildContext context) => widget.child;
}

void main() async {
  WidgetsFlutterBinding.ensureInitialized();

  // Plain sqflite only ships a native database engine for Android/iOS.
  // On Windows and Linux desktop there is no such engine bundled, so
  // sqflite_common_ffi's databaseFactory must be installed explicitly
  // before DatabaseService (or anything else) opens a database â€”
  // otherwise every DB call throws "databaseFactory not initialized."
  if (!kIsWeb && (Platform.isWindows || Platform.isLinux)) {
    sqfliteFfiInit();
    databaseFactory = databaseFactoryFfi;
  }

  // V2: load Supabase credentials from .env and start the client.
  // Wrapped in try/catch so a missing .env or a Supabase problem never
  // stops the local-first app from launching.
  try {
    await dotenv.load(fileName: '.env');
    await Supabase.initialize(
      url: dotenv.env['SUPABASE_URL']!,
      publishableKey: dotenv.env['SUPABASE_ANON_KEY']!,
    );
    // Only flips to true when init fully succeeded. AuthGate reads this
    // so it never touches Supabase.instance when the client isn't ready.
    AuthService.isAvailable = true;
    debugPrint('Supabase initialized OK');

    // V2 Step 4: start local <-> cloud sync (does nothing until signed in).
    SyncService.instance.start();
  } catch (e) {
    AuthService.isAvailable = false;
    debugPrint('Supabase init failed: $e');
  }

  // Safe no-op on Web â€” see notification_service.dart for why.
  await NotificationService.instance.init();

  // V2 Step 16: re-apply reminders from the saved settings (daily goal,
  // exams). Never throws, runs in the background.
  NotificationService.instance.applyAllSafe();

  // V2 Step 13: gives XP to existing sessions and tasks the first time,
  // then keeps the ledger in line with the data. Runs in the background.
  GamificationHooks.startupRebuild();
  await AvatarService.init();
  runApp(const StudyArchitectApp());
}

/// Reads and writes unlocked badges. A badge is never removed once
/// unlocked (only "Clear All Data" wipes the table).
class AchievementRepository {
  final _db = DatabaseService.instance;

  /// Unlocks a badge. Returns true only the first time, so the caller
  /// knows when to show a celebration.
  Future<bool> unlock(String id, {DateTime? at}) async {
    final db = await _db.database;
    return db.transaction((txn) async {
      final existing = await txn.query(
        'achievements',
        columns: ['id'],
        where: 'id = ?',
        whereArgs: [id],
        limit: 1,
      );
      if (existing.isNotEmpty) return false;
      final badge = Achievement(id: id, unlockedAt: at ?? DateTime.now());
      await txn.insert('achievements', badge.toMap());
      return true;
    });
  }

  Future<bool> isUnlocked(String id) async {
    final db = await _db.database;
    final rows = await db.query(
      'achievements',
      columns: ['id'],
      where: 'id = ?',
      whereArgs: [id],
      limit: 1,
    );
    return rows.isNotEmpty;
  }

  Future<Set<String>> getUnlockedIds() async {
    final db = await _db.database;
    final rows = await db.query('achievements', columns: ['id']);
    return rows.map((r) => r['id'] as String).toSet();
  }

  /// Every unlocked badge, newest first.
  Future<List<Achievement>> getAll() async {
    final db = await _db.database;
    final rows = await db.query('achievements', orderBy: 'unlockedAt DESC');
    return rows.map(Achievement.fromMap).toList();
  }

  Future<List<Achievement>> getRecent({int limit = 3}) async {
    final db = await _db.database;
    final rows = await db.query(
      'achievements',
      orderBy: 'unlockedAt DESC',
      limit: limit,
    );
    return rows.map(Achievement.fromMap).toList();
  }
}

  }

/// "Level 300 â€¢ Computer Engineering". The level is saved as "Level 300"
/// (or just "300" when typed by hand), so "Level" is only added if missing.
String _profileSubtitle(String level, String program) {
  final lvl = level.trim();
  final prog = program.trim();
  final levelText = lvl.isEmpty
      ? ''
      : (lvl.toLowerCase().startsWith('level') ? lvl : 'Level $lvl');
  if (levelText.isEmpty) return prog;
  if (prog.isEmpty) return levelText;
  return '$levelText â€¢ $prog';
}

/// Drives welcome -> profile -> daily goal (dial) -> subjects, then saves a
/// Student and its Subjects locally. SyncService uploads them to Supabase
/// (the database triggers stamp updatedAt), so there is no cloud code here.
class OnboardingFlow extends StatefulWidget {
  final VoidCallback onComplete;

  const OnboardingFlow({
    super.key,
    required this.onComplete,
  });

  @override
  State<OnboardingFlow> createState() => _OnboardingFlowState();
}

class _OnboardingFlowState extends State<OnboardingFlow> {
  static const _pageCount = 4;

  final _pageController = PageController();
  final _studentRepository = StudentRepository();
  final _subjectRepository = SubjectRepository();
  final _uuid = const Uuid();

  int _page = 0;

  String _name = '';
  String _school = '';
  String _program = '';
  String _level = '';
  int _dailyGoalMinutes = 180;

  bool _saving = false;
  String? _error;

  // Kept across retries so a failed attempt never creates duplicates.
  String? _studentId;
  bool _studentSaved = false;
  final Set<String> _savedSubjectIds = {};

  void _goTo(int page) {
    if (_saving) return;
    FocusScope.of(context).unfocus();
    setState(() => _page = page);
    _pageController.animateToPage(
      page,
      duration: const Duration(milliseconds: 250),
      curve: Curves.easeOut,
    );
  }

  void _next() => _goTo(_page + 1);

  Future<void> _finish(List<SubjectDraft> drafts) async {
    if (_saving) return;

    FocusScope.of(context).unfocus();

    setState(() {
      _saving = true;
      _error = null;
    });

    try {
      _studentId ??= _uuid.v4();

      final student = Student(
        id: _studentId!,
        name: _name.trim(),
        school: _school.trim(),
        program: _program.trim(),
        level: _level.trim(),
        dailyGoalMinutes: _dailyGoalMinutes,
        // Weekly goal follows the daily goal until the user changes it
        // from More/Settings.
        weeklyGoalMinutes: _dailyGoalMinutes * 7,
      );

      if (_studentSaved) {
        await _studentRepository.update(student);
      } else {
        await _studentRepository.save(student);
        _studentSaved = true;
      }

      final now = DateTime.now();
      for (final d in drafts) {
        if (_savedSubjectIds.contains(d.id)) continue;

        await _subjectRepository.create(
          Subject(
            id: d.id,
            name: d.name.trim(),
            difficulty: d.difficulty,
            colorValue: d.colorValue,
            weeklyTargetMinutes: d.weeklyTargetHours * 60,
            createdAt: now,
          ),
        );
        _savedSubjectIds.add(d.id);
      }

      // Notification setup should not prevent onboarding from
      // completing if the platform rejects permission or scheduling.
      try {
        await NotificationService.instance.requestPermissions();

        await NotificationService.instance.scheduleDailyGoalReminder(
          const TimeOfDay(hour: 19, minute: 0),
        );
      } catch (e, stackTrace) {
        debugPrint(
          'Onboarding notification setup failed: $e\n$stackTrace',
        );
      }

      // Ask the sync layer to upload the new rows soon.
      if (AuthService.isAvailable) {
        SyncService.instance.scheduleSync();
      }

      if (!mounted) return;

      setState(() => _saving = false);

      widget.onComplete();
    } catch (e, stackTrace) {
      debugPrint('Onboarding save failed: $e\n$stackTrace');

      if (!mounted) return;

      setState(() {
        _saving = false;
        _error = 'Could not finish setup. Please try again.';
      });
    }
  }

                                    ),
                                    onPressed: () => _goTo(_page - 1),
                                  )
                                : null,
                          ),
                          Expanded(
                            child: Row(
                              mainAxisAlignment: MainAxisAlignment.center,
                              children: [
                                for (var i = 0; i < _pageCount; i++)
                                  AnimatedContainer(
                                    duration: const Duration(milliseconds: 250),
                                    margin: const EdgeInsets.symmetric(
                                      horizontal: 4,
                                    ),
                                    width: i == _page ? 26 : 8,
                                    height: 8,
                                    decoration: BoxDecoration(

                                          ? AppColors.primary
                                          : onSurface.withValues(alpha: 0.2),

                              ],

/// The export/import format is each model's existing toMap() shape,
/// wrapped with a version tag. V2 Step 16 adds three optional keys
/// (avatarPath, preferences, appVersion); older backups simply do not
/// have them and older apps ignore them, so the version stays 1.
class BackupService {
  static const _currentVersion = 1;

  static const _countTables = {
    'subjects': 'subjects',
    'topics': 'topics',
    'sessions': 'study_sessions',
    'tasks': 'tasks',
    'exams': 'exams',
    'xpLog': 'xp_log',
    'achievements': 'achievements',
  };

  /// Gathers every table into one JSON-serializable map.
  Future<Map<String, dynamic>> buildBackupJson() async {
    final student = await StudentRepository().getCurrent();
    final settings = await SettingsRepository().get();
    final subjects = await SubjectRepository().getAll();

    final topics = <Map<String, dynamic>>[];
    final sessions = <Map<String, dynamic>>[];
    for (final subject in subjects) {
      final subjectTopics = await TopicRepository().getForSubject(subject.id);
      topics.addAll(subjectTopics.map((t) => t.toMap()));
      final subjectSessions =
          await SessionRepository().getForSubject(subject.id);
      sessions.addAll(subjectSessions.map((s) => s.toMap()));
    }

    final tasks = await TaskRepository().getAll();

    // V2 Step 11: exams and their topic checklists.
    final examRepo = ExamRepository();
    final exams = await examRepo.getAll();
    final examTopics = await examRepo.getAllTopicLinks();

    // V2 Step 13: XP ledger and unlocked badges.
    final xpLog = await XpRepository().getAll();
    final achievements = await AchievementRepository().getAll();

    return {
      'version': _currentVersion,
      'appVersion': 'V2',
      'exportedAt': DateTime.now().toIso8601String(),
      'student': student?.toMap(),
      'settings': settings.toMap(),
      'subjects': subjects.map((s) => s.toMap()).toList(),
      'topics': topics,
      'sessions': sessions,
      'tasks': tasks.map((t) => t.toMap()).toList(),
      'exams': exams.map((e) => e.toMap()).toList(),
      'examTopics': examTopics.map((l) => l.toMap()).toList(),
      'xpLog': xpLog.map((x) => x.toMap()).toList(),
      'achievements': achievements.map((a) => a.toMap()).toList(),
      // V2 Step 16: the profile photo link and the reminder preferences.
      'avatarPath': await _readAvatarPath(),
      'preferences': await AppPrefs.exportForBackup(),
    };
  }

  /// Reads the avatar path straight from the table, so it is saved
  /// whatever the Student model's toMap() does.
  Future<String?> _readAvatarPath() async {
    try {
      final db = await DatabaseService.instance.database;
      final rows = await db.rawQuery('SELECT avatarPath FROM students LIMIT 1');
      if (rows.isEmpty) return null;
      return rows.first['avatarPath'] as String?;
    } catch (_) {
      return null;
    }
  }

  /// Writes the backup JSON to a temp file and returns its path.
  Future<String> exportToFile() async {

    final dir = await getTemporaryDirectory();
    final timestamp =

    final file = File('${dir.path}/study_architect_backup_$timestamp.json');
    await file.writeAsString(const JsonEncoder.withIndent('  ').convert(json));
    return file.path;

      final dir = await getApplicationDocumentsDirectory();
      final file = File('${dir.path}/study_architect_before_restore.json');
      await file

      return file.path;
    } catch (_) {
      return null;
    }

  /// Validates and parses backup JSON text. Throws FormatException
  /// with a human-readable message if it is not a Study Architect backup.
  Map<String, dynamic> parseBackup(String jsonText) {
    final text =
        jsonText.startsWith('\uFEFF') ? jsonText.substring(1) : jsonText;

    final dynamic decoded;
    try {
      decoded = jsonDecode(text);
    } on FormatException {
      throw const FormatException(
          'This file is not a valid Study Architect backup.');
    }

    if (decoded is! Map<String, dynamic>) {
      throw const FormatException(
          'This file is not a valid Study Architect backup.');
    }

    final version = decoded['version'];
    if (version == null) {
      throw const FormatException('This file is missing a version marker.');
    }
    if (version is! int) {
      throw const FormatException(
          'This file is not a valid Study Architect backup.');
    }
    if (version > _currentVersion) {
      throw const FormatException(
          'This backup was made by a newer version of the app.');
    }

    for (final key in const [
      'subjects',
      'topics',
      'sessions',
      'tasks',
      'exams',
      'examTopics',
      'xpLog',
      'achievements',
    ]) {
      final value = decoded[key];
      if (value != null && value is! List) {
        throw FormatException('The "$key" section of this backup is damaged.');
      }
    }
    return decoded;
  }

  /// How many items each section of a parsed backup holds.
  Map<String, int> describe(Map<String, dynamic> backup) {
    int count(String key) {
      final v = backup[key];
      return v is List ? v.length : 0;
    }

    return {
      'subjects': count('subjects'),
      'topics': count('topics'),
      'sessions': count('sessions'),
      'tasks': count('tasks'),
      'exams': count('exams'),
      'xpLog': count('xpLog'),
      'achievements': count('achievements'),
    };
  }

  /// Same counts, for the data currently stored on this device.
  Future<Map<String, int>> countLocal() async {
    final db = await DatabaseService.instance.database;
    final out = <String, int>{};
    for (final entry in _countTables.entries) {
      final rows =
          await db.rawQuery('SELECT COUNT(*) AS c FROM ${entry.value}');
      final c = rows.first['c'];
      out[entry.key] = c is int ? c : int.tryParse('$c') ?? 0;
    }
    return out;
  }

  /// Fully replaces local data with the contents of [backup].
  Future<void> restore(Map<String, dynamic> backup) async {
    await DatabaseService.instance.restoreFromBackup(
      student: (backup['student'] as Map?)?.cast<String, dynamic>(),
      settings: (backup['settings'] as Map?)?.cast<String, dynamic>(),
      subjects: _asMapList(backup['subjects']),
      topics: _asMapList(backup['topics']),
      sessions: _asMapList(backup['sessions']),
      tasks: _asMapList(backup['tasks']),
      exams: _asMapList(backup['exams']),
      examTopics: _asMapList(backup['examTopics']),
      xpLog: _asMapList(backup['xpLog']),
      achievements: _asMapList(backup['achievements']),
    );

    await _restoreAvatarPath(backup['avatarPath']);
    await _clearStaleTombstones();

    final prefs = backup['preferences'];
    if (prefs is Map) {
      await AppPrefs.importFromBackup(prefs.cast<String, dynamic>());
    }
  }

  Future<void> _restoreAvatarPath(dynamic value) async {
    if (value is! String || value.isEmpty) return;
    try {
      final db = await DatabaseService.instance.database;
      await db.execute('UPDATE students SET avatarPath = ?', [value]);
    } catch (_) {
      // The photo link is a nice-to-have; the restore itself succeeded.
    }
  }

  /// The restore deletes every row first, which can leave "deleted"
  /// markers for ids that were just inserted again. Those markers would
  /// make the next sync delete the restored rows in the cloud, so any
  /// marker whose row exists again is removed.
  Future<void> _clearStaleTombstones() async {
    try {
      final db = await DatabaseService.instance.database;
      const tables = [
        'students',
        'settings',
        'subjects',
        'topics',
        'study_sessions',
        'tasks',
        'exams',
        'exam_topics',
        'achievements',
        'xp_log',
      ];
      for (final t in tables) {
        await db.execute(
          'DELETE FROM sync_tombstones WHERE tableName = ? '
          'AND rowId IN (SELECT CAST(id AS TEXT) FROM $t)',
          [t],
        );
      }
    } catch (_) {
      // No tombstone table (or nothing to clean): nothing to do.
    }
  }

  List<Map<String, dynamic>> _asMapList(dynamic value) {
    if (value is! List) return [];
    return value.map((e) => (e as Map).cast<String, dynamic>()).toList();
  }
}

    _db ??= await _open();
    return _db!;

  Future<Database> _open() async {
    final dbPath = await getDatabasesPath();
    final path = join(dbPath, _dbName);
    return openDatabase(
      path,
      version: _dbVersion,
      onCreate: (db, version) async {
        await _createSchema(db, version);
        await applySyncSchema(db);
        await _addExams(db);
        await _addGamification(db);
        await _addAvatarPath(db);
      },
      onUpgrade: (db, oldVersion, newVersion) async {
        if (oldVersion < 2) await applySyncSchema(db);
        if (oldVersion < 3) await _addSessionRecurrence(db);
        if (oldVersion < 4) await _addTopicSortOrder(db);
        if (oldVersion < 5) await _addExams(db);
        if (oldVersion < 6) await _addGamification(db);
        if (oldVersion < 7) await _addAvatarPath(db);
      },
      // sqflite does not enforce foreign keys by default â€” turn them on
      // so the ON DELETE CASCADE / SET NULL rules above actually run.
      onConfigure: (db) => db.execute('PRAGMA foreign_keys = ON'),
    );
  }

      if (settings != null) {
        await txn.insert('settings', {...settings, 'id': 0});
      }
      for (final row in subjects) {
        await txn.insert('subjects', row);
      }
      for (final row in topics) {
        await txn.insert('topics', row);
      }
      for (final row in sessions) {
        await txn.insert('study_sessions', row);
      }
      for (final row in tasks) {
        await txn.insert('tasks', row);
      }
      for (final row in exams) {
        await txn.insert('exams', row);
      }
      for (final row in examTopics) {
        await txn.insert('exam_topics', row);
      }
      for (final row in xpLog) {
        await txn.insert('xp_log', row);
      }
      for (final row in achievements) {
        await txn.insert('achievements', row);
      }
    });
  }

import 'dart:async';

import 'package:flutter/foundation.dart';
import 'package:sqflite/sqflite.dart';
import 'package:supabase_flutter/supabase_flutter.dart';

import '../core/data_refresh_bus.dart';

class _SyncTable {
  final String name;

  /// settings uses an INTEGER id locally; the cloud stores ids as text.
  final bool intId;
  const _SyncTable(this.name, {this.intId = false});
}

// Parents first, so children never arrive before the row they reference.
const _tables = [
  _SyncTable('students'),
  _SyncTable('settings', intId: true),
  _SyncTable('subjects'),
  _SyncTable('topics'),
  _SyncTable('study_sessions'),
  _SyncTable('tasks'),
  _SyncTable('exams'),
  _SyncTable('exam_topics'),
  _SyncTable('achievements'),
  _SyncTable('xp_log'),
];

/// UTC ISO-8601 with millisecond precision â€” the same shape SQLite writes,
/// so plain string comparison orders timestamps correctly.
String _iso(DateTime d) => DateTime.fromMillisecondsSinceEpoch(
      d.millisecondsSinceEpoch,
      isUtc: true,
    ).toIso8601String();

class SyncService {
  SyncService._();
  static final SyncService instance = SyncService._();

  static const _pageSize = 500;

  SupabaseClient get _client => Supabase.instance.client;

  bool _running = false;
  bool _runAgain = false;
  Timer? _debounce;
  Timer? _poller;
  RealtimeChannel? _channel;
  StreamSubscription<AuthState>? _authSub;

  /// Call once after Supabase.initialize().
  Future<void> start() async {
    await _authSub?.cancel();
    _authSub = _client.auth.onAuthStateChange.listen((state) {
      if (state.session != null) {
        _subscribeRealtime();
        scheduleSync();
      } else {
        _unsubscribeRealtime();
      }
    });

    // Picks up local edits (repositories don't notify the sync layer).
    _poller ??= Timer.periodic(
      const Duration(seconds: 30),
      (_) => scheduleSync(),
    );

    if (_client.auth.currentUser != null) {
      _subscribeRealtime();
      scheduleSync();
    }
  }

  void stop() {
    _poller?.cancel();
    _poller = null;
    _debounce?.cancel();
    _authSub?.cancel();
    _unsubscribeRealtime();
  }

  /// Debounced: many quick calls collapse into one sync.
  void scheduleSync() {
    _debounce?.cancel();
    _debounce = Timer(const Duration(seconds: 2), syncNow);
  }

  Future<void> syncNow() async {
    final user = _client.auth.currentUser;
    if (user == null) return;
    if (_running) {
      _runAgain = true;
      return;
    }
    _running = true;
    try {
      final changed = await _pull(user.id); // cloud -> local
      await _push(user.id); // local -> cloud
      // Only when the pull actually changed local data, refresh every screen.
      if (changed.isNotEmpty) {
        DataRefreshBus.instance.notify(RefreshScope.all);
      }
    } catch (e) {
      // Offline or server error: nothing is lost, the next run retries.
      debugPrint('Sync failed: $e');
    } finally {
      _running = false;
      if (_runAgain) {
        _runAgain = false;
        scheduleSync();
      }
    }
  }

  // ---------------------------------------------------------------- realtime

  void _subscribeRealtime() {
    _unsubscribeRealtime();
    final channel = _client.channel('study-architect-sync');
    for (final t in _tables) {
      channel.onPostgresChanges(
        event: PostgresChangeEvent.all,
        schema: 'public',
        table: t.name,
        callback: (_) => scheduleSync(),
      );
    }
    channel.subscribe();
    _channel = channel;
  }

  void _unsubscribeRealtime() {
    final channel = _channel;
    if (channel != null) _client.removeChannel(channel);
    _channel = null;
  }

  // -------------------------------------------------------------------- meta

  Future<String?> _getMeta(Database db, String key) async {
    final rows =
        await db.query('sync_meta', where: 'key = ?', whereArgs: [key]);
    return rows.isEmpty ? null : rows.first['value'] as String;
  }

  Future<void> _setMeta(Database db, String key, String value) {
    return db.insert(
      'sync_meta',
      {'key': key, 'value': value},
      conflictAlgorithm: ConflictAlgorithm.replace,
    );
  }

  // -------------------------------------------------------------------- pull

  /// Downloads rows changed in the cloud since the last pull.
  /// Returns the names of local tables that were modified.
  Future<Set<String>> _pull(String uid) async {
    final db = await DatabaseService.instance.database;
    final touched = <String>{};

    for (final t in _tables) {
      final key = 'pull:$uid:${t.name}';
      var cursor = await _getMeta(db, key);

      while (true) {
        final base = _client.from(t.name).select();
        final rows =
            await (cursor == null ? base : base.gt('updatedAt', cursor))
                .order('updatedAt', ascending: true)
                .limit(_pageSize);

        for (final raw in rows) {
          final row = Map<String, dynamic>.from(raw);
          try {
            if (await _applyRemote(db, t, row)) touched.add(t.name);
          } catch (e) {
            debugPrint('Skipped ${t.name} row ${row['id']}: $e');
          }
          cursor = _iso(DateTime.parse(row['updatedAt'] as String));
        }

        if (cursor != null) await _setMeta(db, key, cursor);
        if (rows.length < _pageSize) break;
      }
    }
    return touched;
  }

  /// Applies one cloud row with last-write-wins. Returns true if the
  /// local table changed.
  Future<bool> _applyRemote(
    Database db,
    _SyncTable t,
    Map<String, dynamic> remote,
  ) async {
    final id = remote['id'] as String;
    final localId = t.intId ? int.parse(id) : id;
    final remoteAt = _iso(DateTime.parse(remote['updatedAt'] as String));

    final local = await db.query(
      t.name,
      columns: ['updatedAt'],
      where: 'id = ?',
      whereArgs: [localId],
    );
    final localAt = local.isEmpty ? null : local.first['updatedAt'] as String?;

    // Local copy is newer or identical: keep it.
    if (localAt != null && localAt.compareTo(remoteAt) >= 0) return false;

    // Cloud row was deleted.
    if (remote['deletedAt'] != null) {
      await db.delete(t.name, where: 'id = ?', whereArgs: [localId]);
      await db.delete(
        'sync_tombstones',
        where: 'tableName = ? AND rowId = ?',
        whereArgs: [t.name, id],
      );
      return local.isNotEmpty;
    }

    // Deleted locally after the cloud version was written: keep it deleted.
    final tomb = await db.query(
      'sync_tombstones',
      where: 'tableName = ? AND rowId = ?',
      whereArgs: [t.name, id],
    );
    if (tomb.isNotEmpty &&
        (tomb.first['deletedAt'] as String).compareTo(remoteAt) >= 0) {
      return false;
    }

    final values = Map<String, dynamic>.from(remote)
      ..remove('user_id')
      ..remove('deletedAt')
      ..['id'] = localId
      ..['updatedAt'] = remoteAt;

    if (local.isEmpty) {
      await db.insert(t.name, values);
    } else {
      // update, not "insert or replace": REPLACE would delete the row first
      // and cascade-delete its children.
      await db.update(t.name, values, where: 'id = ?', whereArgs: [localId]);
    }
    return true;
  }

  // -------------------------------------------------------------------- push

  Future<void> _push(String uid) async {
    final db = await DatabaseService.instance.database;

    // 1) New and edited rows.
    for (final t in _tables) {
      final key = 'push:$uid:${t.name}';
      final since = await _getMeta(db, key);

      final rows = await db.query(
        t.name,
        where: since == null ? null : 'updatedAt > ?',
        whereArgs: since == null ? null : [since],
        orderBy: 'updatedAt ASC',
      );
      if (rows.isEmpty) continue;

      final payload = rows
          .map((r) => <String, dynamic>{
                ...r,
                'id': r['id'].toString(),
                'user_id': uid,
                'deletedAt': null,
              })
          .toList();

      for (var i = 0; i < payload.length; i += 200) {
        final end = i + 200 > payload.length ? payload.length : i + 200;
        await _client
            .from(t.name)
            .upsert(payload.sublist(i, end), onConflict: 'user_id,id');
      }
      await _setMeta(db, key, rows.last['updatedAt'] as String);
    }

    // 2) Deletes.
    final tombstones = await db.query('sync_tombstones');
    for (final tomb in tombstones) {
      final table = tomb['tableName'] as String;
      final rowId = tomb['rowId'] as String;
      final at = tomb['deletedAt'] as String;

      await _client
          .from(table)
          .update({'deletedAt': at, 'updatedAt': at})
          .eq('user_id', uid)
          .eq('id', rowId);

      await db.delete(
        'sync_tombstones',
        where: 'tableName = ? AND rowId = ?',
        whereArgs: [table, rowId],
      );
    }
  }
}
