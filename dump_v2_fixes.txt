import 'package:flutter/material.dart';
import '../repositories/settings_repository.dart';

/// Single source of truth for the app's current ThemeMode. MaterialApp
/// listens to [mode] directly, so calling [setMode] anywhere in the
/// app (from the More/Settings screen) updates every screen
/// immediately — no restart needed.
class ThemeController {
  ThemeController._internal();
  static final ThemeController instance = ThemeController._internal();

  final ValueNotifier<ThemeMode> mode = ValueNotifier(ThemeMode.system);
  final _settingsRepo = SettingsRepository();

  Future<void> loadFromSettings() async {
    final settings = await _settingsRepo.get();
    mode.value = settings.themeMode;
  }

  Future<void> setMode(ThemeMode newMode) async {
    mode.value = newMode;
    final current = await _settingsRepo.get();
    await _settingsRepo.save(current.copyWith(themeMode: newMode));
  }
}

import 'dart:async';

import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'package:supabase_flutter/supabase_flutter.dart';

import 'auth_service.dart';

/// Debug-only screen that tests the auth system beyond "did login work?".
/// Never prints tokens. Safe to keep in the project; it is only reachable
/// from the More screen when running in debug mode.
class AuthDiagnosticsScreen extends StatefulWidget {
  const AuthDiagnosticsScreen({super.key});

  @override
  State<AuthDiagnosticsScreen> createState() => _AuthDiagnosticsScreenState();
}

class _Check {
  final String name;
  final bool ok;
  final String detail;
  const _Check(this.name, this.ok, this.detail);
}

class _AuthDiagnosticsScreenState extends State<AuthDiagnosticsScreen> {
  final List<_Check> _checks = [];
  final List<String> _events = [];
  StreamSubscription<AuthState>? _sub;
  bool _running = false;

  @override
  void initState() {
    super.initState();
    if (AuthService.isAvailable) {
      _sub = AuthService.instance.authStateChanges.listen((state) {
        if (!mounted) return;
        setState(() => _events.insert(0, '${_now()}  ${state.event.name}'));
      });
    }
    _runChecks();
  }

  @override
  void dispose() {
    _sub?.cancel();
    super.dispose();
  }

  String _now() {
    final t = DateTime.now();
    String two(int n) => n.toString().padLeft(2, '0');
    return '${two(t.hour)}:${two(t.minute)}:${two(t.second)}';
  }

  String _timeLeft(Duration d) {
    if (d.isNegative) return 'already expired';
    if (d.inMinutes < 1) return 'in under a minute';
    if (d.inHours < 1) return 'in ${d.inMinutes} min';
    return 'in ${d.inHours} h ${d.inMinutes % 60} min';
  }

  Future<void> _runChecks() async {
    setState(() {
      _running = true;
      _checks.clear();
    });

    final results = <_Check>[];

    void finish() {
      if (!mounted) return;
      setState(() {
        _checks
          ..clear()
          ..addAll(results);
        _running = false;
      });
    }

    // 1. Client initialized
    results.add(_Check(
      'Supabase initialized',
      AuthService.isAvailable,
      AuthService.isAvailable
          ? 'Client is ready'
          : 'main() could not start Supabase. Check the .env file.',
    ));
    if (!AuthService.isAvailable) {
      finish();
      return;
    }

    final client = Supabase.instance.client;
    final session = client.auth.currentSession;
    final user = client.auth.currentUser;

    // 2. Session exists
    results.add(_Check(
      'Session present',
      session != null && user != null,
      session != null
          ? 'A session is stored on this device'
          : 'No one is signed in',
    ));
    if (session == null || user == null) {
      finish();
      return;
    }

    // 3. Account info
    results.add(_Check(
      'Signed-in account',
      true,
      '${user.email ?? 'no email'}\nid: ${user.id}',
    ));

    // 4. Token expiry
    final expiry = DateTime.fromMillisecondsSinceEpoch(
      (session.expiresAt ?? 0) * 1000,
    );
    results.add(_Check(
      'Access token valid',
      !session.isExpired,
      'Expires ${_timeLeft(expiry.difference(DateTime.now()))}',
    ));

    // 5. Server accepts the token
    try {
      final res = await client.auth.getUser();
      final same = res.user?.id == user.id;
      results.add(_Check(
        'Server accepts the token',
        same,
        same
            ? 'Supabase verified this user'
            : 'Server returned a different user',
      ));
    } catch (e) {
      results.add(_Check(
        'Server accepts the token',
        false,
        AuthService.friendlyError(e),
      ));
    }

    // 6. Token refresh works
    try {
      final res = await client.auth.refreshSession();
      results.add(_Check(
        'Token refresh works',
        res.session != null,
        res.session != null
            ? 'New token issued, expires ${_timeLeft(DateTime.fromMillisecondsSinceEpoch((res.session!.expiresAt ?? 0) * 1000).difference(DateTime.now()))}'
            : 'Refresh returned no session',
      ));
    } catch (e) {
      results.add(_Check(
        'Token refresh works',
        false,
        AuthService.friendlyError(e),
      ));
    }

    // 7. Linked sign-in methods
    final providers = (user.identities ?? []).map((i) => i.provider).toSet();
    results.add(_Check(
      'Linked sign-in methods',
      providers.isNotEmpty,
      providers.isEmpty ? 'None reported' : providers.join(', '),
    ));

    // 8. Email confirmed
    results.add(_Check(
      'Email confirmed',
      user.emailConfirmedAt != null,
      user.emailConfirmedAt != null
          ? 'Confirmed'
          : 'Not confirmed (expected only if email confirmation is off)',
    ));

    finish();
  }

  Future<void> _copyReport() async {
    final buffer = StringBuffer('Auth diagnostics ${DateTime.now()}\n');
    for (final c in _checks) {
      buffer.writeln('[${c.ok ? 'OK' : 'FAIL'}] ${c.name}: ${c.detail}');
    }
    await Clipboard.setData(ClipboardData(text: buffer.toString()));
    if (!mounted) return;
    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(content: Text('Report copied to clipboard')),
    );
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Auth diagnostics'),
        actions: [
          IconButton(
            tooltip: 'Copy report',
            onPressed: _checks.isEmpty ? null : _copyReport,
            icon: const Icon(Icons.copy_rounded),
          ),
        ],
      ),
      body: ListView(
        padding: const EdgeInsets.all(16),
        children: [
          FilledButton.icon(
            onPressed: _running ? null : _runChecks,
            icon: _running
                ? const SizedBox(
                    width: 18,
                    height: 18,
                    child: CircularProgressIndicator(strokeWidth: 2),
                  )
                : const Icon(Icons.play_arrow_rounded),
            label: Text(_running ? 'Running checks...' : 'Run checks again'),
          ),
          const SizedBox(height: 16),
          for (final c in _checks)
            Card(
              child: ListTile(
                leading: Icon(
                  c.ok ? Icons.check_circle_rounded : Icons.cancel_rounded,
                  color: c.ok ? Colors.green : Colors.red,
                ),
                title: Text(c.name),
                subtitle: Text(c.detail),
              ),
            ),
          const SizedBox(height: 16),
          Text(
            'Auth events since this screen opened',
            style: Theme.of(context).textTheme.titleSmall,
          ),
          const SizedBox(height: 8),
          if (_events.isEmpty)
            const Text('No events yet.')
          else
            for (final e in _events) Text(e),
        ],
      ),
    );
  }
}

import 'dart:async';

import 'package:flutter/material.dart';
import 'package:supabase_flutter/supabase_flutter.dart';

import 'auth_service.dart';
import 'password_prompt_gate.dart';
import 'set_password_screen.dart';
import 'sign_in_screen.dart';

/// Decides what the user sees:
/// - Supabase not ready         -> your app shell directly (local-first fallback)
/// - password-reset link opened -> Set password screen
/// - session exists             -> your signed-in app shell, wrapped in the
///                                 password prompt (bottom sheet on entry)
/// - no session                 -> Sign In screen
/// It listens to Supabase, so sign-in, sign-out and expired sessions
/// switch screens automatically with no manual navigation.
class AuthGate extends StatefulWidget {
  /// Your existing V1 shell (the screen with Home • Timetable • + • Subjects • More).
  final WidgetBuilder signedInBuilder;

  const AuthGate({super.key, required this.signedInBuilder});

  @override
  State<AuthGate> createState() => _AuthGateState();
}

class _AuthGateState extends State<AuthGate> {
  bool _recovering = false;
  StreamSubscription<AuthState>? _sub;

  @override
  void initState() {
    super.initState();
    if (AuthService.isAvailable) {
      _sub = AuthService.instance.authStateChanges.listen((state) {
        if (!mounted) return;
        if (state.event == AuthChangeEvent.passwordRecovery) {
          setState(() => _recovering = true);
        } else if (state.event == AuthChangeEvent.signedOut && _recovering) {
          setState(() => _recovering = false);
        }
      });
    }
  }

  @override
  void dispose() {
    _sub?.cancel();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    // Supabase failed to start (missing .env, bad key...): keep the
    // local-first app usable instead of trapping the user on a login
    // screen that can never succeed.
    if (!AuthService.isAvailable) {
      return widget.signedInBuilder(context);
    }

    return StreamBuilder<AuthState>(
      stream: AuthService.instance.authStateChanges,
      initialData: AuthState(
        AuthChangeEvent.initialSession,
        AuthService.instance.currentSession,
      ),
      builder: (context, snapshot) {
        final session = snapshot.data?.session;
        if (session == null) {
          return const SignInScreen();
        }
        if (_recovering) {
          return SetPasswordScreen(
            recoveryMode: true,
            onDone: () {
              if (mounted) setState(() => _recovering = false);
            },
          );
        }
        return PasswordPromptGate(
          child: widget.signedInBuilder(context),
        );
      },
    );
  }
}

import 'package:flutter/foundation.dart' show kIsWeb;
import 'package:supabase_flutter/supabase_flutter.dart';
import 'package:url_launcher/url_launcher.dart' show LaunchMode;

/// Thin wrapper around Supabase Auth so no screen talks to Supabase directly.
class AuthService {
  AuthService._();
  static final AuthService instance = AuthService._();

  /// Set to true in main() only after Supabase.initialize succeeds.
  static bool isAvailable = false;

  /// True after the user taps "Not now" on the password banner.
  /// Lives in memory only, so the banner returns on the next app launch.
  static bool passwordPromptDismissed = false;

  SupabaseClient get _client => Supabase.instance.client;

  /// Deep link used to come back into the app after Google sign-in, after
  /// tapping the confirmation link in a sign-up email, or after tapping a
  /// password-reset link (Android/iOS/Windows).
  static const String mobileRedirect =
      'io.supabase.studyarchitect://login-callback/';

  User? get currentUser => _client.auth.currentUser;
  Session? get currentSession => _client.auth.currentSession;
  Stream<AuthState> get authStateChanges => _client.auth.onAuthStateChange;

  /// Whether the signed-in user can sign in with email + password.
  /// True when the account has an email identity (email sign-up) or when a
  /// password was set through the app (has_password marker in metadata).
  /// A Google-only account returns false.
  bool get hasPassword {
    final user = currentUser;
    if (user == null) return true;
    if (user.userMetadata?['has_password'] == true) return true;
    final identities = user.identities ?? [];
    return identities.any((i) => i.provider == 'email');
  }

  // ---------------------------------------------------------------------------
  // Email + password
  // ---------------------------------------------------------------------------

  /// Creates an account with email and password.
  ///
  /// - emailRedirectTo sends the confirmation link back into the app
  ///   (without it the link opens the Site URL, which is localhost).
  /// - When "Confirm email" is ON, Supabase answers a sign-up for an address
  ///   that already exists with a fake success (no error, empty identities)
  ///   so attackers cannot discover registered emails. We detect that case and
  ///   raise a clear error instead of showing "check your inbox" for nothing.
  Future<AuthResponse> signUpWithEmail({
    required String fullName,
    required String email,
    required String password,
  }) async {
    final res = await _client.auth.signUp(
      email: email.trim(),
      password: password,
      data: {'full_name': fullName.trim(), 'has_password': true},
      emailRedirectTo: kIsWeb ? null : mobileRedirect,
    );

    final identities = res.user?.identities;
    if (res.user != null && identities != null && identities.isEmpty) {
      throw AuthException('User already registered');
    }
    return res;
  }

  Future<AuthResponse> signInWithEmail({
    required String email,
    required String password,
  }) {
    return _client.auth.signInWithPassword(
      email: email.trim(),
      password: password,
    );
  }

  Future<void> sendPasswordReset(String email) {
    return _client.auth.resetPasswordForEmail(
      email.trim(),
      redirectTo: kIsWeb ? null : mobileRedirect,
    );
  }

  /// Sets a new password for the account that is signed in right now.
  /// Used both after a reset link and from More > Password.
  /// For a Google-only account this adds a password, so the same email
  /// can also be used in the email sign-in form. The has_password marker
  /// stops the "Add a password" banner from showing again.
  Future<UserResponse> updatePassword(String newPassword) {
    return _client.auth.updateUser(
      UserAttributes(
        password: newPassword,
        data: {'has_password': true},
      ),
    );
  }

  // ---------------------------------------------------------------------------
  // OAuth providers (Google now, others later)
  // ---------------------------------------------------------------------------

  /// Shared launcher so every provider behaves the same way.
  /// In-app browser (Chrome Custom Tab on Android) returns to the app
  /// reliably; a normal browser tab often stalls on a blank page.
  Future<bool> _signInWithOAuth(OAuthProvider provider) {
    return _client.auth.signInWithOAuth(
      provider,
      redirectTo: kIsWeb ? null : mobileRedirect,
      authScreenLaunchMode:
          kIsWeb ? LaunchMode.platformDefault : LaunchMode.inAppBrowserView,
    );
  }

  Future<bool> signInWithGoogle() => _signInWithOAuth(OAuthProvider.google);

  // Later, adding a provider is one line once it is enabled in Supabase:
  // Future<bool> signInWithGitHub() => _signInWithOAuth(OAuthProvider.github);

  /// Attaches another sign-in method to the account that is signed in now.
  /// Requires "Allow manual linking" to be enabled in Supabase.
  Future<bool> linkProvider(OAuthProvider provider) {
    return _client.auth.linkIdentity(
      provider,
      redirectTo: kIsWeb ? null : mobileRedirect,
      authScreenLaunchMode:
          kIsWeb ? LaunchMode.platformDefault : LaunchMode.inAppBrowserView,
    );
  }

  // ---------------------------------------------------------------------------
  // Session
  // ---------------------------------------------------------------------------

  Future<void> signOut() async {
    passwordPromptDismissed = false;
    await _client.auth.signOut();
  }

  // ---------------------------------------------------------------------------
  // Errors
  // ---------------------------------------------------------------------------

  /// Turns Supabase errors into short, human messages.
  static String friendlyError(Object error) {
    if (error is AuthException) {
      final m = error.message.toLowerCase();
      if (m.contains('invalid login credentials')) {
        return 'Incorrect email or password. If you created your account with Google, tap "Continue with Google" instead.';
      }
      if (m.contains('email not confirmed')) {
        return 'Please confirm your email first. Check your inbox.';
      }
      if (m.contains('already registered') ||
          m.contains('already been registered')) {
        return 'An account with this email already exists. Try signing in, or tap "Continue with Google" if you signed up with Google.';
      }
      if (m.contains('different from the old password') ||
          m.contains('same as the old password')) {
        return 'Choose a password that is different from your current one.';
      }
      if (m.contains('reauthentication')) {
        return 'For security, please sign out, sign in again, then change your password.';
      }
      if (m.contains('auth session missing')) {
        return 'Your session expired. Please sign in again.';
      }
      if (m.contains('password should be at least')) {
        return 'Password is too short. Use at least 6 characters.';
      }
      if (m.contains('unable to validate email') ||
          m.contains('invalid email')) {
        return 'That email address does not look valid.';
      }
      if (m.contains('signups not allowed') ||
          m.contains('signup is disabled')) {
        return 'Sign-ups are currently turned off.';
      }
      if (m.contains('provider is not enabled') ||
          m.contains('unsupported provider')) {
        return 'This sign-in method is not enabled yet.';
      }
      if (m.contains('rate limit') || m.contains('too many requests')) {
        return 'Too many attempts. Please wait a minute and try again.';
      }
      return error.message;
    }
    final text = error.toString().toLowerCase();
    if (text.contains('socket') ||
        text.contains('failed host lookup') ||
        text.contains('network')) {
      return 'No internet connection. Please check your network and retry.';
    }
    return 'Something went wrong. Please try again.';
  }
}

class SubjectRepository {
  final _db = DatabaseService.instance;

  Future<void> create(Subject subject) async {
    final db = await _db.database;
    await db.insert('subjects', subject.toMap());
  }

  Future<void> update(Subject subject) async {
    final db = await _db.database;
    await db.update(
      'subjects',
      subject.toMap(),
      where: 'id = ?',
      whereArgs: [subject.id],
    );
  }

  /// Deletes the subject. Related topics/sessions/tasks cascade via
  /// the foreign keys in DatabaseService — callers should warn the
  /// user before calling this (business rule from the V1 spec).
  Future<void> delete(String id) async {
    final db = await _db.database;
    await db.delete('subjects', where: 'id = ?', whereArgs: [id]);
  }

  Future<Subject?> getById(String id) async {
    final db = await _db.database;
    final rows = await db.query('subjects', where: 'id = ?', whereArgs: [id]);
    if (rows.isEmpty) return null;
    return Subject.fromMap(rows.first);
  }

  Future<List<Subject>> getAll() async {
    final db = await _db.database;
    final rows = await db.query('subjects', orderBy: 'createdAt ASC');
    return rows.map(Subject.fromMap).toList();
  }

  /// Counts topics + sessions + tasks tied to a subject, so the UI can
  /// show "this will also delete N related items" before confirming.
  Future<int> countRelatedRecords(String subjectId) async {
    final db = await _db.database;
    final topics = Sqflite.firstIntValue(await db.rawQuery(
      'SELECT COUNT(*) FROM topics WHERE subjectId = ?',
      [subjectId],
    ));
    final sessions = Sqflite.firstIntValue(await db.rawQuery(
      'SELECT COUNT(*) FROM study_sessions WHERE subjectId = ?',
      [subjectId],
    ));
    final tasks = Sqflite.firstIntValue(await db.rawQuery(
      'SELECT COUNT(*) FROM tasks WHERE subjectId = ?',
      [subjectId],
    ));
    return (topics ?? 0) + (sessions ?? 0) + (tasks ?? 0);
  }
}

    final db = await _db.database;
    await db.delete('tasks', where: 'id = ?', whereArgs: [id]);
  }

  Future<void> setCompleted(String id, bool completed) async {
    final db = await _db.database;
    await db.update(
      'tasks',
      {'completed': completed ? 1 : 0},
      where: 'id = ?',
      whereArgs: [id],
    );
  }

  Future<List<Task>> getForDate(DateTime date) async {
    final db = await _db.database;
    final startOfDay = DateTime(date.year, date.month, date.day);
    final endOfDay = startOfDay.add(const Duration(days: 1));
    final rows = await db.query(
      'tasks',
      where: 'dueDate >= ? AND dueDate < ?',
      whereArgs: [startOfDay.toIso8601String(), endOfDay.toIso8601String()],
      orderBy: 'dueDate ASC',
    );
    return rows.map(Task.fromMap).toList();
  }

  Future<List<Task>> getAll() async {
    final db = await _db.database;
    final rows = await db.query('tasks', orderBy: 'dueDate ASC');
    return rows.map(Task.fromMap).toList();
  }

  Future<List<Task>> getForSubject(String subjectId) async {
    final db = await _db.database;
    final rows = await db.query(
      'tasks',
      where: 'subjectId = ?',
      whereArgs: [subjectId],
      orderBy: 'dueDate ASC',

      _route = route;
      routeObserver.subscribe(this, route);

/// Everything the Home screen needs, loaded in one pass so the UI
/// never has to juggle five separate FutureBuilders.
class HomeViewData {
  final Student student;
  final int todayMinutes;
  final int dailyGoalMinutes;
  final int weeklyMinutes;
  final int weeklyGoalMinutes;
  final int subjectsCount;
  final int tasksTodayCount;
  final int streak;
  final List<StudySession> todaySessions;
  final Map<String, Subject> subjectsById;
  final List<Task> todayTasks;

  /// Days (yyyy-MM-dd) with at least one completed session. Powers the
  /// 7-day streak strip on Home (V2 step 6).
  final Set<String> completedDates;

  /// V2 Step 13: level, XP and the streak (null if the engine failed, in
  /// which case Home falls back to the old streak).
  final GamificationSnapshot? gamification;

  const HomeViewData({
    required this.student,
    required this.todayMinutes,
    required this.dailyGoalMinutes,
    required this.weeklyMinutes,
    required this.weeklyGoalMinutes,
    required this.subjectsCount,
    required this.tasksTodayCount,
    required this.streak,
    required this.todaySessions,
    required this.subjectsById,
    required this.todayTasks,
    this.completedDates = const <String>{},
    this.gamification,
  });

  double get todayProgress => dailyGoalMinutes == 0
      ? 0
      : (todayMinutes / dailyGoalMinutes).clamp(0.0, 1.0);

  int get weeklyGoalPercent => weeklyGoalMinutes == 0
      ? 0
      : ((weeklyMinutes / weeklyGoalMinutes) * 100).round();

  /// True when [day] counts for the streak (10+ minutes studied). Falls
  /// back to "at least one completed session" if the engine failed.
  bool studiedOn(DateTime day) {
    final g = gamification;
    if (g != null) return g.studyDays.contains(_dateKey(day));
    return completedDates.contains(_dateKey(day));
  }

  /// True when the weekly freeze forgave the missed [day].
  bool bridgedOn(DateTime day) =>
      gamification?.streak.bridgedDays.contains(_dateKey(day)) ?? false;

  /// True when this week's free missed day has already been used.
  bool get freezeUsed => gamification?.streak.freezeUsedThisWeek ?? false;

  static String _dateKey(DateTime date) {
    final y = date.year.toString().padLeft(4, '0');
    final m = date.month.toString().padLeft(2, '0');
    final d = date.day.toString().padLeft(2, '0');
    return '$y-$m-$d';
  }

  static Future<HomeViewData> load() async {
    final studentRepo = StudentRepository();
    final subjectRepo = SubjectRepository();
    final sessionRepo = SessionRepository();
    final taskRepo = TaskRepository();

    final now = DateTime.now();
    final weekStart = startOfWeek(now);
    final weekEnd = endOfWeek(now);

    final student = await studentRepo.getCurrent();
    if (student == null) {
      throw StateError(
          'HomeViewData.load() called before onboarding completed.');
    }

    final subjects = await subjectRepo.getAll();
    final subjectsById = {for (final s in subjects) s.id: s};

    final todaySessions = await sessionRepo.getForDate(now);
    final weekSessions = await sessionRepo.getForRange(weekStart, weekEnd);
    final todayTasks = await taskRepo.getForDate(now);
    final completedDates = await sessionRepo.getCompletedDates();

    // V2 Step 13: level, XP and the new streak. If the engine fails, Home
    // still loads.
    GamificationSnapshot? gamification;
    try {
      gamification = await GamificationEngine.instance.getSnapshot(now: now);
    } catch (_) {
      gamification = null;
    }

    return HomeViewData(
      student: student,
      todayMinutes: completedMinutesForDate(todaySessions, now),
      dailyGoalMinutes: student.dailyGoalMinutes,
      weeklyMinutes: completedMinutesTotal(weekSessions),
      weeklyGoalMinutes: student.weeklyGoalMinutes,
      subjectsCount: subjects.length,
      tasksTodayCount: todayTasks.where((t) => !t.completed).length,
      streak: gamification?.streak.length ??
          currentStreak(completedDates, today: now),
      todaySessions: todaySessions,
      subjectsById: subjectsById,
      todayTasks: todayTasks,
      completedDates: completedDates,
      gamification: gamification,
    );
  }
}

  static const _windowDays = 30;

        RefreshScope.exams,

  void initState() {
    super.initState();
    WidgetsBinding.instance.addObserver(this);
    _load();
  }

  @override
  void dispose() {
    WidgetsBinding.instance.removeObserver(this);
    super.dispose();
  }

  @override
  void didUpdateWidget(covariant NextExamCard oldWidget) {
    super.didUpdateWidget(oldWidget);
    if (oldWidget.loadToken != widget.loadToken) {
      _load();
    }
  }

  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    if (state == AppLifecycleState.resumed) {
      _load();
    }
  }

  @override
  Future<void> onRefresh() => _load();

  Future<void> _load() async {
    try {
      final repo = ExamRepository();
      final upcoming = await repo.getUpcoming(limit: 12);
      final subjects = await SubjectRepository().getAll();
      final progress = await repo.getProgressForAll();
      final subjectsById = {for (final s in subjects) s.id: s};

      final items = <ExamSummary>[];
      for (final exam in upcoming) {
        // Sorted nearest first, so the first one outside the window ends it.
        if (exam.daysLeft() > _windowDays) break;
        items.add(
          ExamSummary(
            exam: exam,
            subject: subjectsById[exam.subjectId],
            progress: progress[exam.id] ?? const ExamProgress.empty(),
          ),
        );
      }

      if (!mounted) return;

      setState(() {
        _items = items.take(_maxExtra + 1).toList();
        if (_items.length <= 1) _expanded = false;

      // The card is optional: keep showing what we had.
    }

      name: _nameController.text.trim(),
      school: _schoolController.text.trim(),
      program: _programController.text.trim(),
      level: _levelController.text.trim(),

  static const _pageCount = 4;

  final _pageController = PageController();

  final _subjectRepository = SubjectRepository();

  String _school = '';
  String _program = '';
  String _level = '';
  int _dailyGoalMinutes = 180;

  String? _studentId;

  void _goTo(int page) {
    if (_saving) return;

    _pageController.animateToPage(

      curve: Curves.easeOut,

  void _next() => _goTo(_page + 1);

  Future<void> _finish(List<SubjectDraft> drafts) async {
    if (_saving) return;

    FocusScope.of(context).unfocus();

    setState(() {
      _saving = true;
      _error = null;
    });

    try {
      _studentId ??= _uuid.v4();

      final student = Student(
        id: _studentId!,
        name: _name.trim(),
        school: _school.trim(),
        program: _program.trim(),
        level: _level.trim(),
        dailyGoalMinutes: _dailyGoalMinutes,
        // Weekly goal follows the daily goal until the user changes it
        // from More/Settings.
        weeklyGoalMinutes: _dailyGoalMinutes * 7,
      );

      if (_studentSaved) {
        await _studentRepository.update(student);
      } else {
        await _studentRepository.save(student);
        _studentSaved = true;
      }

      final now = DateTime.now();
      for (final d in drafts) {
        if (_savedSubjectIds.contains(d.id)) continue;

        await _subjectRepository.create(
          Subject(
            id: d.id,
            name: d.name.trim(),
            difficulty: d.difficulty,
            colorValue: d.colorValue,
            weeklyTargetMinutes: d.weeklyTargetHours * 60,
            createdAt: now,
          ),
        );
        _savedSubjectIds.add(d.id);
      }

      // Notification setup should not prevent onboarding from
      // completing if the platform rejects permission or scheduling.
      try {
        await NotificationService.instance.requestPermissions();

        await NotificationService.instance.scheduleDailyGoalReminder(
          const TimeOfDay(hour: 19, minute: 0),
        );
      } catch (e, stackTrace) {
        debugPrint(
          'Onboarding notification setup failed: $e\n$stackTrace',
        );
      }

      // Ask the sync layer to upload the new rows soon.
      if (AuthService.isAvailable) {
        SyncService.instance.scheduleSync();
      }

      if (!mounted) return;

      setState(() => _saving = false);

      widget.onComplete();
    } catch (e, stackTrace) {
      debugPrint('Onboarding save failed: $e\n$stackTrace');

      if (!mounted) return;

      setState(() {
        _saving = false;
        _error = 'Could not finish setup. Please try again.';
      });
    }
  }

/// When creating, the person can also list the subject's first topics.

  final List<String> _topics = [];

  late double _weeklyTargetHours;

  String? _error;

    AppColors.accentPurpleStrong,
    AppColors.accentBlueStrong,
    AppColors.accentGreenStrong,

  void initState() {
    super.initState();

      text: existing?.name ?? '',
    );

    _difficulty = existing?.difficulty ?? SubjectDifficulty.medium;

    _weeklyTargetHours =
        ((existing?.weeklyTargetMinutes ?? 360) / 60).toDouble();

        ? Color(existing.colorValue)
        : AppColors.accentPurpleStrong;
  }

    _nameController.dispose();

  void _addTopicToList() {
    final value = _topicController.text.trim();
    if (value.isEmpty) return;

    final exists = _topics.any((t) => t.toLowerCase() == value.toLowerCase());

    setState(() {
      if (!exists) _topics.add(value);
      _topicController.clear();
      _error = null;
    });
  }

  Future<void> _save() async {
    final name = _nameController.text.trim();

    if (name.isEmpty || _saving) return;

    FocusScope.of(context).unfocus();

    // A topic typed but not yet added with "+" still counts.
    final pendingTopic = _topicController.text.trim();
    if (!_isEditing && pendingTopic.isNotEmpty) {
      final exists =
          _topics.any((t) => t.toLowerCase() == pendingTopic.toLowerCase());
      if (!exists) _topics.add(pendingTopic);
      _topicController.clear();
    }

    setState(() {
      _saving = true;
      _error = null;
    });

    var topicsFailed = false;

    try {
      final repo = SubjectRepository();

      if (_isEditing) {
        final updated = widget.existing!.copyWith(
          name: name,
          difficulty: _difficulty,
          colorValue: _color.toARGB32(),
          weeklyTargetMinutes: _weeklyTargetHours.round() * 60,
        );

        await repo.update(updated);
      } else {
        final subject = Subject(
          id: const Uuid().v4(),
          name: name,
          difficulty: _difficulty,
          colorValue: _color.toARGB32(),
          weeklyTargetMinutes: _weeklyTargetHours.round() * 60,
          createdAt: DateTime.now(),
        );

        await repo.create(subject);

        // Topics are created one by one, in the order listed; the
        // repository appends each one at the end of the roadmap.
        // If this part fails the subject already exists, so we do not
        // report a failure (a retry would create a duplicate subject).
        try {
          final topicRepo = TopicRepository();
          for (final topicName in _topics) {
            await topicRepo.create(
              Topic(
                id: const Uuid().v4(),
                subjectId: subject.id,
                name: topicName,
              ),
            );
          }
        } catch (_) {
          topicsFailed = true;
        }
      }

      if (!mounted) return;

      if (topicsFailed) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(
            content: Text(
              'Subject saved, but some topics could not be added. '
              'You can add them from the subject page.',
            ),
          ),
        );
      }

      DataRefreshBus.instance.notify(RefreshScope.subjects);
      Navigator.of(context).pop(true);
    } catch (_) {
      if (!mounted) return;

      setState(() {
        _saving = false;
        _error = 'Could not save the subject. Please try again.';
      });
    }
  }

class _SubjectDetailScreenState extends State<SubjectDetailScreen>
    with WidgetsBindingObserver, RefreshableScreenMixin {
  final _subjectRepo = SubjectRepository();
  final _topicRepo = TopicRepository();
  final _sessionRepo = SessionRepository();

  late Future<_DetailData> _future;

  /// Topic ids in the order the person just dragged them to. Shown
  /// immediately so the list does not jump back while the new order is
  /// being saved; cleared once the reload from the database is done.
  List<String>? _pendingOrder;

  @override
  void initState() {
    super.initState();

    WidgetsBinding.instance.addObserver(this);

    _future = _load();
  }

  @override
  void dispose() {
    WidgetsBinding.instance.removeObserver(this);
    super.dispose();
  }

  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    if (state == AppLifecycleState.resumed && mounted) {
      _reload();
    }
  }

  Future<_DetailData> _load() async {
    final subject = await _subjectRepo.getById(widget.subjectId);

    if (subject == null) {
      throw StateError('Subject not found');
    }

    final topics = await _topicRepo.getForSubject(widget.subjectId);
    final sessions = await _sessionRepo.getForSubject(widget.subjectId);

    return _DetailData(
      subject: subject,
      topics: topics,
      sessions: sessions,
    );
  }

  void _reload() {
    if (!mounted) return;

    setState(() {
      _future = _load();
    });
  }

  Future<void> _refresh() async {
    if (!mounted) return;

    final future = _load();

    setState(() {
      _future = future;
    });

    try {
      await future;
    } catch (_) {
      // AsyncBuilder will display the error state.
    }
  }

  /// Called by RefreshableScreenMixin when sync brings in cloud changes.
  @override
  Future<void> onRefresh() => _refresh();

  @override
  List<RefreshScope> get watchedScopes => [
      RefreshScope.subjects,
      RefreshScope.sessions,
      RefreshScope.tasks,
      RefreshScope.exams,
    ];

  /// Tells the other screens (for example the Subjects list underneath
  /// this one) that subject or topic data changed.
  void _notifyChanged() {
    DataRefreshBus.instance.notify(RefreshScope.all);
  }

  /// Applies the not-yet-saved drag order on top of the loaded topics.
  List<Topic> _ordered(List<Topic> topics) {
    final pending = _pendingOrder;
    if (pending == null) return topics;

    final byId = {for (final t in topics) t.id: t};
    final result = <Topic>[];
    for (final id in pending) {
      final topic = byId.remove(id);
      if (topic != null) result.add(topic);
    }
    result.addAll(byId.values);
    return result;
  }

  /// Asks for a topic name. Returns null if cancelled.
  Future<String?> _promptTopicName({
    required String title,
    required String confirmLabel,
    String initial = '',
  }) async {
    final controller = TextEditingController(text: initial);
    controller.selection = TextSelection(
      baseOffset: 0,
      extentOffset: initial.length,
    );

    TransitionRoute<dynamic>? dialogRoute;

    final name = await showDialog<String>(
      context: context,
      builder: (dialogContext) {
        dialogRoute = ModalRoute.of(dialogContext);

        return AlertDialog(
          title: Text(title),
          content: TextField(
            controller: controller,
            autofocus: true,
            textCapitalization: TextCapitalization.sentences,
            decoration: const InputDecoration(
              hintText: 'e.g. Differential Equations',
            ),
            onSubmitted: (value) {
              Navigator.pop(dialogContext, value.trim());
            },
          ),
          actions: [
            TextButton(
              onPressed: () => Navigator.pop(dialogContext),
              child: const Text('Cancel'),
            ),
            TextButton(
              onPressed: () {
                Navigator.pop(
                  dialogContext,
                  controller.text.trim(),
                );
              },
              child: Text(confirmLabel),
            ),
          ],
        );
      },
    );

    // Wait until the dialog route has completely finished
    // its exit transition and has been removed from the overlay.
    if (dialogRoute != null) {
      await dialogRoute!.completed;
    }

    controller.dispose();

    return name;
  }

    final name = await _promptTopicName(
      title: 'New topic',
      confirmLabel: 'Add',
    );

    if (name == null || name.isEmpty) return;

    await _topicRepo.create(
      Topic(
        id: const Uuid().v4(),
        subjectId: widget.subjectId,
        name: name,
      ),
    );

    _notifyChanged();

    await _refresh();
    _notifyChanged();
  }

  Future<void> _onReorder(
    List<Topic> current,
    int oldIndex,
    int newIndex,
  ) async {
    if (newIndex > oldIndex) newIndex -= 1;
    if (newIndex == oldIndex) return;

    final ids = current.map((t) => t.id).toList();
    final moved = ids.removeAt(oldIndex);
    ids.insert(newIndex, moved);

    // 1) Instant feedback on screen.
    setState(() {
      _pendingOrder = ids;
    });

    // 2) Save to the database.
    try {
      await _topicRepo.reorder(ids);
    } catch (_) {
      if (!mounted) return;
      setState(() {
        _pendingOrder = null;
      });
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Could not save the new order.')),
      );
      return;
    }

    if (!mounted) return;

    // 3) Reload from the database, then drop the temporary order.
    await _refresh();

    if (!mounted) return;

    setState(() {
      _pendingOrder = null;
    });
    _notifyChanged();
  }

import '../../models/subject.dart';
import '../../repositories/subject_repository.dart';
import '../../repositories/topic_repository.dart';

/// A subject plus the two numbers its list-row card needs: how many
/// topics it has, and what fraction of them are completed.
class SubjectSummary {
  final Subject subject;
  final int topicCount;
  final int completionPercent;

  const SubjectSummary({
    required this.subject,
    required this.topicCount,
    required this.completionPercent,
  });

  static Future<List<SubjectSummary>> loadAll() async {
    final subjectRepo = SubjectRepository();
    final topicRepo = TopicRepository();

    final subjects = await subjectRepo.getAll();
    final summaries = <SubjectSummary>[];

    for (final subject in subjects) {
      final topics = await topicRepo.getForSubject(subject.id);
      final completed = topics.where((t) => t.completed).length;
      final percent =
          topics.isEmpty ? 0 : ((completed / topics.length) * 100).round();
      summaries.add(SubjectSummary(
        subject: subject,
        topicCount: topics.length,
        completionPercent: percent,
      ));
    }

    return summaries;
  }
}

  /// every change goes through a plain setState.
  TasksViewData? _data;
  Object? _error;

  /// Ticks the person just made, shown immediately on screen until the
  /// reload from the database confirms them.
  final Map<String, bool> _pending = {};

  void initState() {
    super.initState();
    _reload();
  }

  /// Reloads from the database. Called after every action on this
  /// screen and by pull-to-refresh.
  Future<void> _reload() async {
    try {
      final data = await TasksViewData.load();
      if (!mounted) return;
      debugPrint(
        '[Tasks] reload: overdue=${data.overdue.length} today=${data.today.length} '
        'tomorrow=${data.tomorrow.length} upcoming=${data.upcoming.length}',
      );
      setState(() {
        _data = data;
        _error = null;
        _pending.clear();
      });
    } catch (e, st) {
      debugPrint('[Tasks] reload failed: $e\n$st');
      if (!mounted) return;
      setState(() {
        if (_data == null) _error = e;
      });
    }
  }

  Future<void> _retry() async {
    setState(() => _error = null);
    await _reload();
  }

  Future<void> _toggle(Task task) async {
    final newValue = !(_pending[task.id] ?? task.completed);
    debugPrint('[Tasks] toggle ${task.id} -> $newValue');

    // 1) Instant feedback on screen.
    setState(() => _pending[task.id] = newValue);

    // 2) Save to the database.
    try {
      await _taskRepo.setCompleted(task.id, newValue);
      debugPrint('[Tasks] saved ${task.id}');

      // V2 Step 13: XP for a completed task, removed again if un-ticked.
      if (newValue) {
        await GamificationHooks.taskCompleted(task.copyWith(completed: true));
      } else {
        await GamificationHooks.taskRemoved(task.id);
      }
    } catch (e, st) {
      debugPrint('[Tasks] toggle failed: $e\n$st');
      if (!mounted) return;
      setState(() => _pending.remove(task.id));
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('Could not update the task: $e')),
      );
      return;
    }

    // V2 Step 17: tell Home, Progress and the other screens that tasks changed.
    DataRefreshBus.instance.notify(RefreshScope.tasks);

    // 3) Reload from the database.
    await _reload();

/// All tasks bucketed the way the Tasks screen displays them, plus
/// subject lookups for each task's colored tag.
class TasksViewData {
  final List<Task> overdue;
  final List<Task> today;
  final List<Task> tomorrow;
  final List<Task> upcoming;
  final Map<String, Subject> subjectsById;

  const TasksViewData({
    required this.overdue,
    required this.today,
    required this.tomorrow,
    required this.upcoming,
    required this.subjectsById,
  });

  static Future<TasksViewData> load() async {
    final taskRepo = TaskRepository();
    final subjectRepo = SubjectRepository();

    final allTasks = await taskRepo.getAll();
    final subjects = await subjectRepo.getAll();
    final subjectsById = {for (final s in subjects) s.id: s};

    final now = DateTime.now();
    final today = startOfDay(now);
    final tomorrow = today.add(const Duration(days: 1));
    final dayAfterTomorrow = today.add(const Duration(days: 2));

    // Completed tasks stay visible in their original bucket (crossed
    // out) rather than disappearing, so finishing something today
    // doesn't make it look like it vanished.
    final pending = allTasks;

    final overdueList = pending
        .where((t) => !t.completed && t.dueDate.isBefore(today))
        .toList();
    final todayList =
        pending.where((t) => isSameDay(t.dueDate, today)).toList();
    final tomorrowList =
        pending.where((t) => isSameDay(t.dueDate, tomorrow)).toList();
    final upcomingList = pending
        .where((t) =>
            t.dueDate.isAfter(dayAfterTomorrow) ||
            isSameDay(t.dueDate, dayAfterTomorrow))
        .toList();

    for (final list in [overdueList, todayList, tomorrowList, upcomingList]) {
      list.sort((a, b) => a.dueDate.compareTo(b.dueDate));
    }

    return TasksViewData(
      overdue: overdueList,
      today: todayList,
      tomorrow: tomorrowList,
      upcoming: upcomingList,
      subjectsById: subjectsById,
    );
  }
}

import 'dart:io';

import 'package:flutter/foundation.dart';

/// An error whose message is safe to show to the user as-is.
class AvatarException implements Exception {
  final String message;
  const AvatarException(this.message);

  @override
  String toString() => message;
}

/// V2 Step 15: profile photo. Picks an image, uploads it to the private
/// Supabase Storage bucket 'avatars' (folder = the user's id), saves the
/// storage path in students.avatarPath (which SyncService then syncs to
/// the other devices) and keeps a local cached copy so the photo shows
/// instantly and offline.
class AvatarService {
  AvatarService._();

  static const bucket = 'avatars';
  static const maxBytes = 2 * 1024 * 1024; // matches the bucket limit (2 MB)

  static final Map<String, File> _memo = {};
  static final Map<String, Future<File?>> _inFlight = {};
  static final Map<String, DateTime> _failedAt = {};

  static SupabaseClient get _client => Supabase.instance.client;

  // ------------------------------------------------------------------ public

  /// Opens the picker, uploads the photo and saves it on the student.
  /// Returns the new storage path, or null if the person cancelled.
  /// Throws AvatarException with a friendly message on failure; in that
  /// case nothing has changed.
  static Future<String?> pickAndUpload(ImageSource source) async {
    final uid = _requireUserId();
    final oldPath = await _readStudentAvatarPath();

    final picked = await ImagePicker().pickImage(
      source: source,
      maxWidth: 512,
      maxHeight: 512,
      imageQuality: 85,
    );
    if (picked == null) return null;

    var ext = p.extension(picked.name).toLowerCase().replaceFirst('.', '');
    if (ext == 'jpeg') ext = 'jpg';
    final mime = _mimeFor(ext);
    if (mime == null) {
      throw const AvatarException('Please choose a JPG, PNG or WebP image.');
    }

    final bytes = await picked.readAsBytes();
    if (bytes.length > maxBytes) {
      throw const AvatarException(
        'That photo is too large (max 2 MB). Please choose a smaller one.',
      );
    }

    // A new file name every time: other devices see the path change and
    // download the new photo, and no cached copy can ever be stale.
    final newPath = '$uid/avatar_${DateTime.now().millisecondsSinceEpoch}.$ext';

    try {
      await _client.storage
          .from(bucket)
          .uploadBinary(
            newPath,
            bytes,
            fileOptions: FileOptions(contentType: mime),
          )
          .timeout(const Duration(seconds: 30));
    } catch (e) {
      debugPrint('Avatar upload failed: $e');
      throw const AvatarException(
        'Could not upload the photo. Check your connection and try again.',
      );
    }

    await _writeCache(newPath, bytes);
    await _setStudentAvatarPath(newPath);

    if (oldPath != null && oldPath != newPath) {
      await _discard(oldPath);
    }

    _announceChange();
    return newPath;
  }

  /// Removes the photo (the initial is shown again).
  static Future<void> removePhoto() async {
    final oldPath = await _readStudentAvatarPath();
    if (oldPath == null) return;

    await _setStudentAvatarPath(null);
    await _discard(oldPath);
    _announceChange();
  }

  /// The photo file if it is already known in memory (instant, no await).
  static File? cachedFile(String storagePath) => _memo[storagePath];

  /// The local file for [storagePath]: the cached copy, or a one-time
  /// download from Supabase. Returns null when it is not available (for
  /// example offline on a new device); the initial is shown meanwhile.
  static Future<File?> localFileFor(String storagePath) {
    final memo = _memo[storagePath];
    if (memo != null) return Future.value(memo);

    // Do not hammer the network: after a failed download wait 30 s.
    final failed = _failedAt[storagePath];
    if (failed != null &&
        DateTime.now().difference(failed) < const Duration(seconds: 30)) {
      return Future.value(null);
    }

    return _inFlight[storagePath] ??=
        _resolve(storagePath).whenComplete(() => _inFlight.remove(storagePath));
  }

  // ----------------------------------------------------------------- private

  static String _requireUserId() {
    final user = AuthService.isAvailable ? _client.auth.currentUser : null;
    if (user == null) {
      throw const AvatarException('Sign in to add a profile photo.');
    }
    return user.id;
  }

  static String? _mimeFor(String ext) {
    switch (ext) {
      case 'jpg':
        return 'image/jpeg';
      case 'png':
        return 'image/png';
      case 'webp':
        return 'image/webp';
    }
    return null;
  }

  static Future<String?> _readStudentAvatarPath() async {
    final db = await DatabaseService.instance.database;
    final rows = await db.query('students', columns: ['avatarPath'], limit: 1);
    if (rows.isEmpty) {
      throw const AvatarException('Finish setting up your profile first.');
    }
    return rows.first['avatarPath'] as String?;
  }

  /// Updates only this column. The students update trigger stamps
  /// updatedAt, so SyncService uploads it on the next sync.
  static Future<void> _setStudentAvatarPath(String? path) async {
    final db = await DatabaseService.instance.database;
    await db.update('students', {'avatarPath': path});
  }

  static void _announceChange() {
    DataRefreshBus.instance.notify(RefreshScope.all);
    if (AuthService.isAvailable) {
      SyncService.instance.scheduleSync();
    }
  }

  static Future<File> _cacheFile(String storagePath) async {
    final base = await getApplicationSupportDirectory();
    final folder = Directory(p.join(base.path, 'avatars'));
    if (!await folder.exists()) {
      await folder.create(recursive: true);
    }
    return File(p.join(folder.path, storagePath.replaceAll('/', '_')));
  }

  static Future<void> _writeCache(String storagePath, Uint8List bytes) async {
    try {
      final file = await _cacheFile(storagePath);
      final tmp = File('${file.path}.tmp');
      await tmp.writeAsBytes(bytes, flush: true);
      await tmp.rename(file.path);
      _memo[storagePath] = file;
    } catch (e) {
      debugPrint('Avatar cache write failed: $e');
    }
  }

  static Future<File?> _resolve(String storagePath) async {
    try {
      final file = await _cacheFile(storagePath);
      if (await file.exists() && await file.length() > 0) {
        _memo[storagePath] = file;
        return file;
      }

      if (!AuthService.isAvailable || _client.auth.currentUser == null) {
        return null;
      }

      final bytes = await _client.storage.from(bucket).download(storagePath);
      await _writeCache(storagePath, bytes);
      if (await file.exists()) {
        _memo[storagePath] = file;
        _failedAt.remove(storagePath);
        return file;
      }
    } catch (e) {
      debugPrint('Avatar load failed: $e');
    }
    _failedAt[storagePath] = DateTime.now();
    return null;
  }

  /// Best effort: deletes the cached copy and the file in Storage.
  static Future<void> _discard(String storagePath) async {
    _memo.remove(storagePath);
    _failedAt.remove(storagePath);

    try {
      final file = await _cacheFile(storagePath);
      if (await file.exists()) await file.delete();
    } catch (_) {}

    if (!AuthService.isAvailable || _client.auth.currentUser == null) return;
    try {
      await _client.storage.from(bucket).remove([storagePath]);
    } catch (e) {
      debugPrint('Avatar remote delete failed: $e');
    }
  }
}

/// Shorthand for the isDark-branch pattern already used in
/// StatCard/ProgressCard since Step 2 — lets every other screen pick
/// up the same "dedicated dark scheme, not just inverted" behavior
/// without repeating `Theme.of(context).brightness == Brightness.dark`
/// everywhere.
///
/// Deliberately NOT used inside form input fields (TextField /
/// DropdownButtonFormField fill colors and their typed-text color) —
/// those stay a fixed light surface with fixed dark text in both
/// themes, since that was a specific, already-debugged fix and
/// switching it per-theme risks reintroducing invisible typed text.
extension ThemeAwareColors on BuildContext {
  bool get isDarkMode => Theme.of(this).brightness == Brightness.dark;

  /// Plain info/stat card background — e.g. a settings row, a summary
  /// box. Not for pastel accent cards, which stay the same hue in
  /// both themes by design.
  Color get cardSurface =>
      isDarkMode ? AppColors.surfaceDark : AppColors.surfaceLight;

  Color get cardSurfaceAlt =>
      isDarkMode ? AppColors.surfaceDarkAlt : AppColors.surfaceLightAlt;

  /// Body text sitting on a [cardSurface] (not on a pastel card, and
  /// not typed text inside a form field).
  Color get primaryText =>
      isDarkMode ? AppColors.textPrimaryDark : AppColors.textPrimary;

  Color get secondaryText =>
      isDarkMode ? AppColors.textSecondaryDark : AppColors.textSecondary;
}

import 'dart:io';

import 'package:flutter/material.dart';

import '../services/avatar_service.dart';
import '../theme/app_theme.dart';

/// V2 Step 15: the one avatar used everywhere (Home, More, Edit profile).
/// Shows the real profile photo when there is one, otherwise the initial
/// on the green gradient circle (also while the photo is loading, or if it
/// cannot be loaded).
class UserAvatar extends StatefulWidget {
  final String name;
  final String? avatarPath;
  final double size;
  final double? fontSize;

  const UserAvatar({
    super.key,
    required this.name,
    this.avatarPath,
    this.size = 48,
    this.fontSize,
  });

  @override
  State<UserAvatar> createState() => _UserAvatarState();
}

class _UserAvatarState extends State<UserAvatar> {
  Future<File?>? _future;

  @override
  void initState() {
    super.initState();
    _future = _resolve();
  }

  @override
  void didUpdateWidget(UserAvatar oldWidget) {
    super.didUpdateWidget(oldWidget);
    final path = widget.avatarPath;
    final changed = oldWidget.avatarPath != path;
    // Also retry when a photo is expected but not on this device yet
    // (for example the download failed while offline).
    final missing = path != null &&
        path.isNotEmpty &&
        AvatarService.cachedFile(path) == null;
    if (changed || missing) {
      _future = _resolve();
    }
  }

  Future<File?>? _resolve() {
    final path = widget.avatarPath;
    if (path == null || path.isEmpty) return null;
    return AvatarService.localFileFor(path);
  }

  String _initial() {
    final trimmed = widget.name.trim();
    if (trimmed.isEmpty) return '?';
    return String.fromCharCode(trimmed.runes.first).toUpperCase();
  }

  Widget _initialCircle() {
    return Container(
      width: widget.size,
      height: widget.size,
      decoration: const BoxDecoration(
        shape: BoxShape.circle,
        gradient: LinearGradient(
          begin: Alignment.topLeft,
          end: Alignment.bottomRight,
          colors: [AppColors.primary, AppColors.teal],
        ),
      ),
      alignment: Alignment.center,
      child: Text(
        _initial(),
        style: TextStyle(
          color: Colors.white,
          fontSize: widget.fontSize ?? widget.size * 0.42,
          fontWeight: FontWeight.w700,
        ),
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    final path = widget.avatarPath;
    final future = _future;

    if (path == null || path.isEmpty || future == null) {
      return _initialCircle();
    }

    return FutureBuilder<File?>(
      future: future,
      initialData: AvatarService.cachedFile(path),
      builder: (context, snapshot) {
        final file = snapshot.data;
        if (file == null) return _initialCircle();

        return ClipOval(
          child: Image.file(
            file,
            width: widget.size,
            height: widget.size,
            fit: BoxFit.cover,
            cacheWidth: (widget.size * 3).round(),
            gaplessPlayback: true,
            errorBuilder: (context, error, stack) => _initialCircle(),
          ),
        );
      },
    );
  }
}

import 'dart:io';
import 'package:flutter/foundation.dart';
import 'package:flutter/material.dart';
import 'package:flutter_dotenv/flutter_dotenv.dart';
import 'package:sqflite_common_ffi/sqflite_ffi.dart';
import 'package:supabase_flutter/supabase_flutter.dart';
import 'app/app.dart';
import 'features/auth/auth_service.dart';
import 'services/notification_service.dart';
import 'gamification/gamification_hooks.dart';
import 'services/sync_service.dart';

void main() async {
  WidgetsFlutterBinding.ensureInitialized();

  // Plain sqflite only ships a native database engine for Android/iOS.
  // On Windows and Linux desktop there is no such engine bundled, so
  // sqflite_common_ffi's databaseFactory must be installed explicitly
  // before DatabaseService (or anything else) opens a database —
  // otherwise every DB call throws "databaseFactory not initialized."
  if (!kIsWeb && (Platform.isWindows || Platform.isLinux)) {
    sqfliteFfiInit();
    databaseFactory = databaseFactoryFfi;
  }

  // V2: load Supabase credentials from .env and start the client.
  // Wrapped in try/catch so a missing .env or a Supabase problem never
  // stops the local-first app from launching.
  try {
    await dotenv.load(fileName: '.env');
    await Supabase.initialize(
      url: dotenv.env['SUPABASE_URL']!,
      publishableKey: dotenv.env['SUPABASE_ANON_KEY']!,
    );
    // Only flips to true when init fully succeeded. AuthGate reads this
    // so it never touches Supabase.instance when the client isn't ready.
    AuthService.isAvailable = true;
    debugPrint('Supabase initialized OK');

    // V2 Step 4: start local <-> cloud sync (does nothing until signed in).
    SyncService.instance.start();
  } catch (e) {
    AuthService.isAvailable = false;
    debugPrint('Supabase init failed: $e');
  }

  // Safe no-op on Web — see notification_service.dart for why.
  await NotificationService.instance.init();

  // V2 Step 16: re-apply reminders from the saved settings (daily goal,
  // exams). Never throws, runs in the background.
  NotificationService.instance.applyAllSafe();

  // V2 Step 13: gives XP to existing sessions and tasks the first time,
  // then keeps the ledger in line with the data. Runs in the background.
  GamificationHooks.startupRebuild();
  runApp(const StudyArchitectApp());
}
