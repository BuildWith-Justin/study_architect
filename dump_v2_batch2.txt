enum RefreshScope {
  subjects,
  sessions,
  tasks,
  exams,
  achievements,
  all,
}

class DataRefreshBus {
  DataRefreshBus._internal();
  static final DataRefreshBus instance = DataRefreshBus._internal();

  final StreamController<RefreshScope> _controller =
      StreamController<RefreshScope>.broadcast();

  Stream<RefreshScope> get stream => _controller.stream;

  void notify(RefreshScope scope) {
    _controller.add(scope);
  }

  void dispose() {
    _controller.close();
  }
}

import 'dart:async';
import 'package:flutter/material.dart';
import 'data_refresh_bus.dart';

/// Mixin any StatefulWidget's State with this to get:
/// - auto-refresh when relevant data changes elsewhere in the app
/// - pull-to-refresh wired to the same reload function
mixin RefreshableScreenMixin<T extends StatefulWidget> on State<T> {
  StreamSubscription<RefreshScope>? _refreshSub;

  /// Override: which scopes should trigger a refresh on this screen
  List<RefreshScope> get watchedScopes => [RefreshScope.all];

  /// Override: how this screen reloads its data
  Future<void> onRefresh();

  @override
  void initState() {
    super.initState();
    _refreshSub = DataRefreshBus.instance.stream.listen((scope) {
      if (watchedScopes.contains(scope) || scope == RefreshScope.all) {
        onRefresh();
      }
    });
  }

  @override
  void dispose() {
    _refreshSub?.cancel();
    super.dispose();
  }

  Widget buildRefreshable({required Widget child}) {
    return RefreshIndicator(
      onRefresh: onRefresh,
      child: child,
    );
  }
}

  switch (hue) {
    case PastelHue.blue:
      return AppColors.accentBlueStrong;
    case PastelHue.purple:
      return AppColors.accentPurpleStrong;
    case PastelHue.green:
      return AppColors.accentGreenStrong;
  }

/// Generic tinted container: the base every task/class/event card builds
/// on. Light mode: soft pastel. Dark mode: the strong hue at low opacity
/// over the navy background, so light text stays readable.
class PastelCard extends StatelessWidget {
  final PastelHue hue;
  final Widget child;
  final EdgeInsetsGeometry padding;
  final VoidCallback? onTap;

  const PastelCard({
    super.key,
    required this.hue,
    required this.child,
    this.padding = const EdgeInsets.all(AppSpacing.md),
    this.onTap,
  });

  @override
  Widget build(BuildContext context) {
    final isDark = AppColors.isDark(context);
    final bg = isDark
        ? pastelStrong(hue).withValues(alpha: 0.18)
        : pastelBackground(hue);

    final card = Container(
      width: double.infinity,
      padding: padding,
      decoration: BoxDecoration(
        color: bg,
        borderRadius: BorderRadius.circular(AppRadii.cardSmall),
      ),
      child: child,
    );

    if (onTap == null) return card;
    return Material(
      color: Colors.transparent,
      child: InkWell(
        borderRadius: BorderRadius.circular(AppRadii.cardSmall),
        onTap: onTap,
        child: card,
      ),
    );
  }
}

import 'dart:async';

import 'package:flutter/foundation.dart';
import 'package:sqflite/sqflite.dart';
import 'package:supabase_flutter/supabase_flutter.dart';

import '../core/data_refresh_bus.dart';
import 'database_service.dart';

class _SyncTable {
  final String name;

  /// settings uses an INTEGER id locally; the cloud stores ids as text.
  final bool intId;
  const _SyncTable(this.name, {this.intId = false});
}

// Parents first, so children never arrive before the row they reference.
const _tables = [
  _SyncTable('students'),
  _SyncTable('settings', intId: true),
  _SyncTable('subjects'),
  _SyncTable('topics'),
  _SyncTable('study_sessions'),
  _SyncTable('tasks'),
  _SyncTable('exams'),
  _SyncTable('exam_topics'),
  _SyncTable('achievements'),
  _SyncTable('xp_log'),
];

/// UTC ISO-8601 with millisecond precision — the same shape SQLite writes,
/// so plain string comparison orders timestamps correctly.
String _iso(DateTime d) => DateTime.fromMillisecondsSinceEpoch(
      d.millisecondsSinceEpoch,
      isUtc: true,
    ).toIso8601String();

class SyncService {
  SyncService._();
  static final SyncService instance = SyncService._();

  static const _pageSize = 500;

  SupabaseClient get _client => Supabase.instance.client;

  bool _running = false;
  bool _runAgain = false;
  Timer? _debounce;
  Timer? _poller;
  RealtimeChannel? _channel;
  StreamSubscription<AuthState>? _authSub;

  /// Call once after Supabase.initialize().
  Future<void> start() async {
    await _authSub?.cancel();
    _authSub = _client.auth.onAuthStateChange.listen((state) {
      if (state.session != null) {
        _subscribeRealtime();
        scheduleSync();
      } else {
        _unsubscribeRealtime();
      }
    });

    // Picks up local edits (repositories don't notify the sync layer).
    _poller ??= Timer.periodic(
      const Duration(seconds: 30),
      (_) => scheduleSync(),
    );

    if (_client.auth.currentUser != null) {
      _subscribeRealtime();
      scheduleSync();
    }
  }

  void stop() {
    _poller?.cancel();
    _poller = null;
    _debounce?.cancel();
    _authSub?.cancel();
    _unsubscribeRealtime();
  }

  /// Debounced: many quick calls collapse into one sync.
  void scheduleSync() {
    _debounce?.cancel();
    _debounce = Timer(const Duration(seconds: 2), syncNow);
  }

  Future<void> syncNow() async {
    final user = _client.auth.currentUser;
    if (user == null) return;
    if (_running) {
      _runAgain = true;
      return;
    }
    _running = true;
    try {
      final changed = await _pull(user.id); // cloud -> local
      await _push(user.id); // local -> cloud
      // Only when the pull actually changed local data, refresh every screen.
      if (changed.isNotEmpty) {
        DataRefreshBus.instance.notify(RefreshScope.all);
      }
    } catch (e) {
      // Offline or server error: nothing is lost, the next run retries.
      debugPrint('Sync failed: $e');
    } finally {
      _running = false;
      if (_runAgain) {
        _runAgain = false;
        scheduleSync();
      }
    }
  }

  // ---------------------------------------------------------------- realtime

  void _subscribeRealtime() {
    _unsubscribeRealtime();
    final channel = _client.channel('study-architect-sync');
    for (final t in _tables) {
      channel.onPostgresChanges(
        event: PostgresChangeEvent.all,
        schema: 'public',
        table: t.name,
        callback: (_) => scheduleSync(),
      );
    }
    channel.subscribe();
    _channel = channel;
  }

  void _unsubscribeRealtime() {
    final channel = _channel;
    if (channel != null) _client.removeChannel(channel);
    _channel = null;
  }

  // -------------------------------------------------------------------- meta

  Future<String?> _getMeta(Database db, String key) async {
    final rows =
        await db.query('sync_meta', where: 'key = ?', whereArgs: [key]);
    return rows.isEmpty ? null : rows.first['value'] as String;
  }

  Future<void> _setMeta(Database db, String key, String value) {
    return db.insert(
      'sync_meta',
      {'key': key, 'value': value},
      conflictAlgorithm: ConflictAlgorithm.replace,
    );
  }

  // -------------------------------------------------------------------- pull

  /// Downloads rows changed in the cloud since the last pull.
  /// Returns the names of local tables that were modified.
  Future<Set<String>> _pull(String uid) async {
    final db = await DatabaseService.instance.database;
    final touched = <String>{};

    for (final t in _tables) {
      final key = 'pull:$uid:${t.name}';
      var cursor = await _getMeta(db, key);

      while (true) {
        final base = _client.from(t.name).select();
        final rows =
            await (cursor == null ? base : base.gt('updatedAt', cursor))
                .order('updatedAt', ascending: true)
                .limit(_pageSize);

        for (final raw in rows) {
          final row = Map<String, dynamic>.from(raw);
          try {
            if (await _applyRemote(db, t, row)) touched.add(t.name);
          } catch (e) {
            debugPrint('Skipped ${t.name} row ${row['id']}: $e');
          }
          cursor = _iso(DateTime.parse(row['updatedAt'] as String));
        }

        if (cursor != null) await _setMeta(db, key, cursor);
        if (rows.length < _pageSize) break;
      }
    }
    return touched;
  }

  /// Applies one cloud row with last-write-wins. Returns true if the
  /// local table changed.
  Future<bool> _applyRemote(
    Database db,
    _SyncTable t,
    Map<String, dynamic> remote,
  ) async {
    final id = remote['id'] as String;
    final localId = t.intId ? int.parse(id) : id;
    final remoteAt = _iso(DateTime.parse(remote['updatedAt'] as String));

    final local = await db.query(
      t.name,
      columns: ['updatedAt'],
      where: 'id = ?',
      whereArgs: [localId],
    );
    final localAt = local.isEmpty ? null : local.first['updatedAt'] as String?;

    // Local copy is newer or identical: keep it.
    if (localAt != null && localAt.compareTo(remoteAt) >= 0) return false;

    // Cloud row was deleted.
    if (remote['deletedAt'] != null) {
      await db.delete(t.name, where: 'id = ?', whereArgs: [localId]);
      await db.delete(
        'sync_tombstones',
        where: 'tableName = ? AND rowId = ?',
        whereArgs: [t.name, id],
      );
      return local.isNotEmpty;
    }

    // Deleted locally after the cloud version was written: keep it deleted.
    final tomb = await db.query(
      'sync_tombstones',
      where: 'tableName = ? AND rowId = ?',
      whereArgs: [t.name, id],
    );
    if (tomb.isNotEmpty &&
        (tomb.first['deletedAt'] as String).compareTo(remoteAt) >= 0) {
      return false;
    }

    final values = Map<String, dynamic>.from(remote)
      ..remove('user_id')
      ..remove('deletedAt')
      ..['id'] = localId
      ..['updatedAt'] = remoteAt;

    if (local.isEmpty) {
      await db.insert(t.name, values);
    } else {
      // update, not "insert or replace": REPLACE would delete the row first
      // and cascade-delete its children.
      await db.update(t.name, values, where: 'id = ?', whereArgs: [localId]);
    }
    return true;
  }

  // -------------------------------------------------------------------- push

  Future<void> _push(String uid) async {
    final db = await DatabaseService.instance.database;

    // 1) New and edited rows.
    for (final t in _tables) {
      final key = 'push:$uid:${t.name}';
      final since = await _getMeta(db, key);

      final rows = await db.query(
        t.name,
        where: since == null ? null : 'updatedAt > ?',
        whereArgs: since == null ? null : [since],
        orderBy: 'updatedAt ASC',
      );
      if (rows.isEmpty) continue;

      final payload = rows
          .map((r) => <String, dynamic>{
                ...r,
                'id': r['id'].toString(),
                'user_id': uid,
                'deletedAt': null,
              })
          .toList();

      for (var i = 0; i < payload.length; i += 200) {
        final end = i + 200 > payload.length ? payload.length : i + 200;
        await _client
            .from(t.name)
            .upsert(payload.sublist(i, end), onConflict: 'user_id,id');
      }
      await _setMeta(db, key, rows.last['updatedAt'] as String);
    }

    // 2) Deletes.
    final tombstones = await db.query('sync_tombstones');
    for (final tomb in tombstones) {
      final table = tomb['tableName'] as String;
      final rowId = tomb['rowId'] as String;
      final at = tomb['deletedAt'] as String;

      await _client
          .from(table)
          .update({'deletedAt': at, 'updatedAt': at})
          .eq('user_id', uid)
          .eq('id', rowId);

      await db.delete(
        'sync_tombstones',
        where: 'tableName = ? AND rowId = ?',
        whereArgs: [table, rowId],
      );
    }
  }
}

import '../models/student.dart';

/// V1 supports a single local student profile — created once during
/// onboarding (Step 5) and updated from More/Settings (Step 13).
class StudentRepository {
  final _db = DatabaseService.instance;

  Future<void> save(Student student) async {
    final db = await _db.database;
    await db.insert('students', student.toMap());
  }

  Future<void> update(Student student) async {
    final db = await _db.database;
    await db.update(
      'students',
      student.toMap(),
      where: 'id = ?',
      whereArgs: [student.id],
    );
  }

  /// Returns null before onboarding has been completed.
  Future<Student?> getCurrent() async {
    final db = await _db.database;
    final rows = await db.query('students', limit: 1);
    if (rows.isEmpty) return null;
    return Student.fromMap(rows.first);
  }
}

class Student {
  final String id;
  final String name;
  final String school;
  final String program;
  final String level;
  final int dailyGoalMinutes;
  final int weeklyGoalMinutes;

  /// V2 Step 15: path of the profile photo in the Supabase 'avatars'
  /// bucket (null = no photo, the initial is shown instead).
  /// Read-only here: only AvatarService writes it, so it is deliberately
  /// NOT part of toMap(). Saving a Student can never erase the photo.
  final String? avatarPath;

  const Student({
    required this.id,
    required this.name,
    required this.school,
    required this.program,
    required this.level,
    required this.dailyGoalMinutes,
    required this.weeklyGoalMinutes,
    this.avatarPath,
  });

  Student copyWith({
    String? id,
    String? name,
    String? school,
    String? program,
    String? level,
    int? dailyGoalMinutes,
    int? weeklyGoalMinutes,
    String? avatarPath,
  }) {
    return Student(
      id: id ?? this.id,
      name: name ?? this.name,
      school: school ?? this.school,
      program: program ?? this.program,
      level: level ?? this.level,
      dailyGoalMinutes: dailyGoalMinutes ?? this.dailyGoalMinutes,
      weeklyGoalMinutes: weeklyGoalMinutes ?? this.weeklyGoalMinutes,

import 'package:flutter/material.dart';

import 'auth_service.dart';
import 'auth_widgets.dart';
import 'sign_up_screen.dart';

class SignInScreen extends StatefulWidget {
  const SignInScreen({super.key});

  @override
  State<SignInScreen> createState() => _SignInScreenState();
}

class _SignInScreenState extends State<SignInScreen> {
  final _formKey = GlobalKey<FormState>();
  final _emailCtrl = TextEditingController();
  final _passwordCtrl = TextEditingController();

  bool _loading = false;
  bool _obscure = true;
  String? _error;
  String? _info;

  @override
  void dispose() {
    _emailCtrl.dispose();
    _passwordCtrl.dispose();
    super.dispose();
  }

  String? _validateEmail(String? v) {
    final value = v?.trim() ?? '';
    if (value.isEmpty) return 'Enter your email';
    final ok = RegExp(r'^[^@\s]+@[^@\s]+\.[^@\s]+$').hasMatch(value);
    return ok ? null : 'Enter a valid email';
  }

  Future<void> _signIn() async {
    if (!_formKey.currentState!.validate()) return;
    setState(() {
      _loading = true;
      _error = null;
      _info = null;
    });
    try {
      await AuthService.instance.signInWithEmail(
        email: _emailCtrl.text,
        password: _passwordCtrl.text,
      );
      // AuthGate reacts to the new session automatically.
      if (mounted) Navigator.of(context).popUntil((r) => r.isFirst);
    } catch (e) {
      if (mounted) setState(() => _error = AuthService.friendlyError(e));
    } finally {
      if (mounted) setState(() => _loading = false);
    }
  }

  Future<void> _google() async {
    setState(() {
      _loading = true;
      _error = null;
      _info = null;
    });
    try {
      final launched = await AuthService.instance.signInWithGoogle();
      debugPrint('Google sign-in launched: $launched');
      if (!launched && mounted) {
        setState(() => _error =
            'Could not open the browser for Google sign-in. Please try again.');
      }
      // On success the browser opens; the session arrives through the deep link.
    } catch (e) {
      debugPrint('Google sign-in error: $e');
      if (mounted) setState(() => _error = AuthService.friendlyError(e));
    } finally {
      if (mounted) setState(() => _loading = false);
    }
  }

  Future<void> _forgotPassword() async {
    final emailError = _validateEmail(_emailCtrl.text);
    if (emailError != null) {
      setState(() {
        _info = null;
        _error = 'Type your email above first, then tap "Forgot password?".';
      });
      return;
    }
    setState(() {
      _loading = true;
      _error = null;
      _info = null;
    });
    try {
      await AuthService.instance.sendPasswordReset(_emailCtrl.text);
      if (mounted) {
        setState(() => _info = 'Password reset link sent. Check your inbox.');
      }
    } catch (e) {
      if (mounted) setState(() => _error = AuthService.friendlyError(e));
    } finally {
      if (mounted) setState(() => _loading = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    return AuthBackground(
      child: AuthCard(
        child: Form(
          key: _formKey,
          child: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              const AuthHeader(
                title: 'Welcome back',
                subtitle: 'Sign in to continue your study journey',
              ),
              if (_error != null) AuthErrorBox(message: _error!),
              if (_info != null) AuthInfoBox(message: _info!),
              AuthTextField(
                controller: _emailCtrl,
                label: 'Email',
                icon: Icons.mail_outline,
                keyboardType: TextInputType.emailAddress,
                textInputAction: TextInputAction.next,
                validator: _validateEmail,
                autofillHints: const [AutofillHints.email],
              ),
              const SizedBox(height: 14),
              AuthTextField(
                controller: _passwordCtrl,
                label: 'Password',
                icon: Icons.lock_outline,
                obscure: _obscure,
                textInputAction: TextInputAction.done,
                onSubmitted: (_) => _signIn(),
                autofillHints: const [AutofillHints.password],
                validator: (v) =>
                    (v == null || v.isEmpty) ? 'Enter your password' : null,
                suffix: IconButton(
                  icon: Icon(
                    _obscure
                        ? Icons.visibility_off_outlined
                        : Icons.visibility_outlined,
                    color: AuthColors.muted,
                  ),
                  onPressed: () => setState(() => _obscure = !_obscure),
                ),
              ),
              Align(
                alignment: Alignment.centerRight,
                child: TextButton(
                  onPressed: _loading ? null : _forgotPassword,
                  child: const Text(
                    'Forgot password?',
                    style: TextStyle(color: AuthColors.primaryDark),
                  ),
                ),
              ),
              const SizedBox(height: 4),
              AuthPrimaryButton(
                label: 'Sign In',
                loading: _loading,
                onPressed: _signIn,
              ),
              const AuthDivider(),
              AuthGoogleButton(loading: _loading, onPressed: _google),
              const SizedBox(height: 18),
              Row(
                mainAxisAlignment: MainAxisAlignment.center,
                children: [
                  const Text(
                    "Don't have an account?",
                    style: TextStyle(color: AuthColors.muted),
                  ),
                  TextButton(
                    onPressed: _loading
                        ? null
                        : () => Navigator.of(context).push(
                              MaterialPageRoute(
                                  builder: (_) => const SignUpScreen()),

import 'dart:async';

import 'package:flutter/material.dart';

import '../../theme/app_theme.dart';
import '../../theme/theme_helpers.dart';
import 'auth_service.dart';
import 'set_password_screen.dart';

/// Wraps the signed-in app. A few seconds after entry it slides up a bottom
/// sheet inviting the user to add a password, when the account has none
/// (for example a Google-only account).
///
/// Rules:
/// - Shown at most once per app launch for each user.
/// - Skipped when the account already has a password.
/// - Skipped for accounts created in the last few minutes, so it does not
///   land on top of onboarding. Those users still get the banner on More and
///   see this sheet on their next launch.
/// - "Not now" only closes the sheet. The More banner stays as a reminder.
class PasswordPromptGate extends StatefulWidget {
  final Widget child;

  const PasswordPromptGate({super.key, required this.child});

  @override
  State<PasswordPromptGate> createState() => _PasswordPromptGateState();
}

class _PasswordPromptGateState extends State<PasswordPromptGate> {
  /// Id of the user who has already seen the sheet in this launch.
  static String? _promptedUserId;

  /// How long to wait after entering the app before showing the sheet.
  static const Duration _delay = Duration(milliseconds: 2500);

  /// Accounts younger than this are treated as brand new (likely onboarding).
  /// Set to Duration.zero while testing with a fresh account.
  static const Duration _newAccountGrace = Duration(minutes: 10);

  Timer? _timer;

  @override
  void initState() {
    super.initState();
    if (_eligible()) {
      _timer = Timer(_delay, _maybeShow);
    }
  }

  @override
  void dispose() {
    _timer?.cancel();
    super.dispose();
  }

  bool _eligible() {
    if (!AuthService.isAvailable) return false;
    final user = AuthService.instance.currentUser;
    if (user == null) return false;
    if (AuthService.instance.hasPassword) return false;
    if (_promptedUserId == user.id) return false;

    final created = DateTime.tryParse(user.createdAt);
    if (created != null &&
        DateTime.now().toUtc().difference(created.toUtc()) < _newAccountGrace) {
      return false;
    }
    return true;
  }

  Future<void> _maybeShow() async {
    if (!mounted || !_eligible()) return;

    // Do not cover another screen (a pushed page or a dialog).
    final route = ModalRoute.of(context);
    if (route != null && !route.isCurrent) return;

    _promptedUserId = AuthService.instance.currentUser?.id;

    final wantsToSet = await showModalBottomSheet<bool>(
      context: context,
      backgroundColor: context.cardSurface,
      isScrollControlled: true,
      showDragHandle: true,
      shape: const RoundedRectangleBorder(
        borderRadius: BorderRadius.vertical(top: Radius.circular(24)),
      ),
      builder: (sheetContext) => const _PasswordSheet(),
    );

    if (wantsToSet == true && mounted) {
      await Navigator.of(context).push(
        MaterialPageRoute(builder: (_) => const SetPasswordScreen()),
      );
    }
  }

  @override
  Widget build(BuildContext context) => widget.child;
}

class _PasswordSheet extends StatelessWidget {
  const _PasswordSheet();

  @override
  Widget build(BuildContext context) {
    return SafeArea(
      child: Padding(
        padding: const EdgeInsets.fromLTRB(
          AppSpacing.lg,
          0,
          AppSpacing.lg,
          AppSpacing.lg,
        ),
        child: Column(
          mainAxisSize: MainAxisSize.min,
          children: [
            Container(
              width: 56,
              height: 56,
              decoration: BoxDecoration(
                color: AppColors.accentPurpleStrong.withOpacity(0.18),
                shape: BoxShape.circle,
              ),
              child: const Icon(
                Icons.lock_outline_rounded,
                color: AppColors.accentPurpleStrong,
                size: 28,
              ),
            ),
            const SizedBox(height: AppSpacing.md),
            Text(
              'Add a password',
              textAlign: TextAlign.center,
              style: TextStyle(
                color: context.primaryText,
                fontSize: 20,
                fontWeight: FontWeight.w700,
              ),
            ),
            const SizedBox(height: AppSpacing.sm),
            const Text(
              'You signed in with Google. Add a password so you can also '
              'sign in with your email if Google is ever unavailable.',
              textAlign: TextAlign.center,
              style: TextStyle(
                color: AppColors.textSecondary,
                fontSize: 14,
                height: 1.4,
              ),
            ),
            const SizedBox(height: AppSpacing.lg),
            SizedBox(
              width: double.infinity,
              child: FilledButton(
                onPressed: () => Navigator.of(context).pop(true),
                style: FilledButton.styleFrom(
                  backgroundColor: AppColors.accentPurpleStrong,
                  foregroundColor: Colors.white,
                  padding: const EdgeInsets.symmetric(vertical: 14),
                ),
                child: const Text('Set password'),
              ),
            ),
            const SizedBox(height: AppSpacing.xs),
            TextButton(
              onPressed: () => Navigator.of(context).pop(false),
              child: const Text(
                'Not now',
                style: TextStyle(color: AppColors.textSecondary),
              ),
            ),
          ],
        ),
      ),
    );
  }
}
