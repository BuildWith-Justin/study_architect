import 'package:flutter/material.dart';

class AppSettings {
  final bool notificationsEnabled;
  final ThemeMode themeMode;

  const AppSettings({
    this.notificationsEnabled = true,
    this.themeMode = ThemeMode.system,
  });

  AppSettings copyWith({
    bool? notificationsEnabled,
    ThemeMode? themeMode,
  }) {
    return AppSettings(
      notificationsEnabled: notificationsEnabled ?? this.notificationsEnabled,
      themeMode: themeMode ?? this.themeMode,
    );
  }

  Map<String, dynamic> toMap() => {
        'notificationsEnabled': notificationsEnabled ? 1 : 0,
        'themeMode': themeMode.name,
      };

  factory AppSettings.fromMap(Map<String, dynamic> map) => AppSettings(
        notificationsEnabled: (map['notificationsEnabled'] as int) == 1,
        themeMode: ThemeMode.values.firstWhere(
          (m) => m.name == map['themeMode'],
          orElse: () => ThemeMode.system,
        ),
      );
}

class Student {
  final String id;
  final String name;
  final String school;
  final String program;
  final String level;
  final int dailyGoalMinutes;
  final int weeklyGoalMinutes;

  const Student({
    required this.id,
    required this.name,
    required this.school,
    required this.program,
    required this.level,
    required this.dailyGoalMinutes,
    required this.weeklyGoalMinutes,
  });

  Student copyWith({
    String? id,
    String? name,
    String? school,
    String? program,
    String? level,
    int? dailyGoalMinutes,
    int? weeklyGoalMinutes,
  }) {
    return Student(
      id: id ?? this.id,
      name: name ?? this.name,
      school: school ?? this.school,
      program: program ?? this.program,
      level: level ?? this.level,
      dailyGoalMinutes: dailyGoalMinutes ?? this.dailyGoalMinutes,
      weeklyGoalMinutes: weeklyGoalMinutes ?? this.weeklyGoalMinutes,
    );
  }

  Map<String, dynamic> toMap() => {
        'id': id,
        'name': name,
        'school': school,
        'program': program,
        'level': level,
        'dailyGoalMinutes': dailyGoalMinutes,
        'weeklyGoalMinutes': weeklyGoalMinutes,
      };

        id: map['id'] as String,
        name: map['name'] as String,

import 'package:sqflite/sqflite.dart';
import '../models/settings.dart';
import '../services/database_service.dart';

/// Single-row settings table (id is always 0).
class SettingsRepository {
  final _db = DatabaseService.instance;

  Future<AppSettings> get() async {
    final db = await _db.database;
    final rows = await db.query('settings', where: 'id = 0');
    if (rows.isEmpty) return const AppSettings();
    return AppSettings.fromMap(rows.first);
  }

  Future<void> save(AppSettings settings) async {
    final db = await _db.database;
    final map = settings.toMap()..['id'] = 0;
    await db.insert(
      'settings',
      map,
      conflictAlgorithm: ConflictAlgorithm.replace,
    );
  }
}

import '../models/student.dart';
import '../services/database_service.dart';

/// V1 supports a single local student profile — created once during
/// onboarding (Step 5) and updated from More/Settings (Step 13).
class StudentRepository {
  final _db = DatabaseService.instance;

  Future<void> save(Student student) async {
    final db = await _db.database;
    await db.insert('students', student.toMap());
  }

  Future<void> update(Student student) async {
    final db = await _db.database;
    await db.update(
      'students',
      student.toMap(),
      where: 'id = ?',
      whereArgs: [student.id],
    );
  }

  /// Returns null before onboarding has been completed.
  Future<Student?> getCurrent() async {
    final db = await _db.database;
    final rows = await db.query('students', limit: 1);
    if (rows.isEmpty) return null;
    return Student.fromMap(rows.first);
  }
}

    with
        WidgetsBindingObserver,
        RouteAware,
        RefreshableScreenMixin<HomeScreen> {
  late Future<HomeViewData> _future;

  ModalRoute<dynamic>? _route;

  /// Data scopes that should make Home reload itself automatically.
  @override
  List<RefreshScope> get watchedScopes => [
        RefreshScope.sessions,
        RefreshScope.tasks,
        RefreshScope.subjects,
        RefreshScope.achievements,
      ];

  @override
  void initState() {
    super.initState(); // RefreshableScreenMixin subscribes to the bus here.
    WidgetsBinding.instance.addObserver(this);
    _future = HomeViewData.load();
  }

  @override
  void didChangeDependencies() {
    super.didChangeDependencies();

    final route = ModalRoute.of(context);

    if (route != null && route != _route) {
      if (_route != null) {
        routeObserver.unsubscribe(this);
      }

      _route = route;
      routeObserver.subscribe(this, route);
    }
  }

  @override
  void dispose() {
    routeObserver.unsubscribe(this);
    WidgetsBinding.instance.removeObserver(this);
    super.dispose(); // RefreshableScreenMixin cancels its subscription here.
  }

  @override
  void didPopNext() {
    onRefresh();
  }

  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    if (state == AppLifecycleState.resumed) {
      onRefresh();
    }
  }

  /// Used by pull-to-refresh, the DataRefreshBus, returning from another
  /// screen and app resume.
  @override
  Future<void> onRefresh() async {
    final next = HomeViewData.load();

    if (!mounted) {
      next.ignore();
      return;
    }

    setState(() {
      _future = next;
    });

    try {
      await next;
    } catch (_) {
      // FutureBuilder handles the error state.
    }
  }

    final hour = DateTime.now().hour;

    if (hour < 12) return 'Good morning';
    if (hour < 17) return 'Good afternoon';
    return 'Good evening';
  }

    await Navigator.of(context).push(
      MaterialPageRoute(builder: (_) => screen),
    );

    if (mounted) {
      await onRefresh();
    }
  }

                    final subject = data.subjectsById[session.subjectId];

                    return Padding(
                      padding: const EdgeInsets.only(bottom: AppSpacing.sm),
                      child: ScheduleRow(
                        title: subject?.name ?? 'Unknown subject',
                        timeRange:
                            '${formatTimeOfDay(session.startTime)} - ${formatTimeOfDay(session.endTime)}',
                        accent: subject != null
                            ? Color(subject.colorValue)
                            : AppColors.primary,
                      ),
                    );
                  }),
                ] else
                  Text(
                    'No sessions scheduled today — tap + to plan one.',
                    style: Theme.of(context).textTheme.bodyMedium?.copyWith(
                          color: context.secondaryText,
                        ),
                  ),
              ],
            ),
          ),
        );
      },
    );
  }

  /// The session currently running, or the next upcoming one today.
  _SessionAndSubject? _currentOrNextSession(HomeViewData data) {
    final now = DateTime.now();

    final sorted = [...data.todaySessions]
      ..sort((a, b) => a.startTime.compareTo(b.startTime));

    for (final session in sorted) {
      if (session.endTime.isAfter(now)) {
        return _SessionAndSubject(
          session,
          data.subjectsById[session.subjectId],
        );
      }
    }

    return null;
  }

  Widget _buildClassCard(HomeViewData data) {
    final match = _currentOrNextSession(data)!;
    final session = match.session;
    final subject = match.subject;
    final now = DateTime.now();

    final isLive =
        session.startTime.isBefore(now) && session.endTime.isAfter(now);

    return ClassCard(
      timeRange:
          '${formatTimeOfDay(session.startTime)} - ${formatTimeOfDay(session.endTime)}',
      title: subject?.name ?? 'Unknown subject',
      hue: _hueForSubject(subject),
      actionLabel: isLive ? 'Start session' : 'View details',
      onAction: isLive
          ? () => _openScreen(
                FocusTimerScreen(
                  subjectId: session.subjectId,
                  subjectName: subject?.name ?? 'Unknown subject',
                  subjectColorValue:
                      subject?.colorValue ?? AppColors.accentPurpleStrong.value,
                  topicId: session.topicId,
                  existingSessionId: session.id,
                  goal: session.endTime.difference(session.startTime),
                ),
              )
          : () {},
    );
  }

  PastelHue _hueForSubject(Subject? subject) {
    if (subject == null) {
      return PastelHue.blue;
    }

    final color = subject.colorValue;

    if (color == AppColors.accentGreenStrong.value) {
      return PastelHue.green;
    }

    if (color == AppColors.accentBlueStrong.value) {
      return PastelHue.blue;
    }

    return PastelHue.purple;
  }
}

  @override
  Widget build(BuildContext context) {

    final title = streak == 0
        ? 'Start your streak today'

                    fontWeight: FontWeight.w600,
                  ),

              if (bonusActive)
                Container(

                  ),
                  decoration: BoxDecoration(
                    color: AppColors.primary.withValues(alpha: 0.15),
                    borderRadius: BorderRadius.circular(999),
                  ),
                  child: Text(
                    '+${XpRules.streakBonusPercent}% XP',
                    style: theme.textTheme.bodySmall?.copyWith(
                      color: AppColors.primary,
                      fontWeight: FontWeight.w600,
                    ),
                  ),
                ),
            ],
          ),
          const SizedBox(height: AppSpacing.md),
          Row(
            mainAxisAlignment: MainAxisAlignment.spaceBetween,
            children: List.generate(7, (i) => _dayCell(context, i)),
          ),
          if (streak > 0) ...[
            const SizedBox(height: AppSpacing.sm),
            Text(
              freezeUsed
                  ? 'Missed-day freeze used this week'
                  : 'One missed day per week is forgiven',

                color: context.secondaryText,
              ),
            ),
          ],
        ],
      ),
    );
  }

  Widget _dayCell(BuildContext context, int index) {
    final day = weekDays[index];
    final studied = studiedOn(day);
    final bridged = !studied && bridgedOn(day);
    final isToday = isSameDay(day, today);

    final neutralFill = context.isDarkMode
        ? Colors.white.withValues(alpha: 0.08)
        : AppColors.primary.withValues(alpha: 0.08);

    return Column(
      children: [
        Text(

                color: context.secondaryText,
                fontWeight: isToday ? FontWeight.w700 : FontWeight.w400,

        Container(
          width: 32,
          height: 32,
          decoration: BoxDecoration(
            shape: BoxShape.circle,
            color: studied ? AppColors.primary : neutralFill,
            border: isToday && !studied
                ? Border.all(color: AppColors.primary, width: 2)
                : null,
          ),
          child: studied
              ? const Icon(Icons.check_rounded, size: 18, color: Colors.white)
              : bridged
                  ? const Icon(
                      Icons.ac_unit_rounded,
                      size: 16,

      ],

import 'dart:async';

import 'package:flutter/foundation.dart';
import 'package:sqflite/sqflite.dart';
import 'package:supabase_flutter/supabase_flutter.dart';

import '../core/data_refresh_bus.dart';
import 'database_service.dart';

class _SyncTable {
  final String name;

  /// settings uses an INTEGER id locally; the cloud stores ids as text.
  final bool intId;
  const _SyncTable(this.name, {this.intId = false});
}

// Parents first, so children never arrive before the row they reference.
const _tables = [
  _SyncTable('students'),
  _SyncTable('settings', intId: true),
  _SyncTable('subjects'),
  _SyncTable('topics'),
  _SyncTable('study_sessions'),
  _SyncTable('tasks'),
  _SyncTable('exams'),
  _SyncTable('exam_topics'),
  _SyncTable('achievements'),
  _SyncTable('xp_log'),
];

/// UTC ISO-8601 with millisecond precision — the same shape SQLite writes,
/// so plain string comparison orders timestamps correctly.
String _iso(DateTime d) => DateTime.fromMillisecondsSinceEpoch(
      d.millisecondsSinceEpoch,
      isUtc: true,
    ).toIso8601String();

class SyncService {
  SyncService._();
  static final SyncService instance = SyncService._();

  static const _pageSize = 500;

  SupabaseClient get _client => Supabase.instance.client;

  bool _running = false;
  bool _runAgain = false;
  Timer? _debounce;
  Timer? _poller;
  RealtimeChannel? _channel;
  StreamSubscription<AuthState>? _authSub;

  /// Call once after Supabase.initialize().
  Future<void> start() async {
    await _authSub?.cancel();
    _authSub = _client.auth.onAuthStateChange.listen((state) {
      if (state.session != null) {
        _subscribeRealtime();
        scheduleSync();
      } else {
        _unsubscribeRealtime();
      }
    });

    // Picks up local edits (repositories don't notify the sync layer).
    _poller ??= Timer.periodic(
      const Duration(seconds: 30),
      (_) => scheduleSync(),
    );

    if (_client.auth.currentUser != null) {
      _subscribeRealtime();
      scheduleSync();
    }
  }

  void stop() {
    _poller?.cancel();
    _poller = null;
    _debounce?.cancel();
    _authSub?.cancel();
    _unsubscribeRealtime();
  }

  /// Debounced: many quick calls collapse into one sync.
  void scheduleSync() {
    _debounce?.cancel();
    _debounce = Timer(const Duration(seconds: 2), syncNow);
  }

  Future<void> syncNow() async {
    final user = _client.auth.currentUser;
    if (user == null) return;
    if (_running) {
      _runAgain = true;
      return;
    }
    _running = true;
    try {
      final changed = await _pull(user.id); // cloud -> local
      await _push(user.id); // local -> cloud
      // Only when the pull actually changed local data, refresh every screen.
      if (changed.isNotEmpty) {
        DataRefreshBus.instance.notify(RefreshScope.all);
      }
    } catch (e) {
      // Offline or server error: nothing is lost, the next run retries.
      debugPrint('Sync failed: $e');
    } finally {
      _running = false;
      if (_runAgain) {
        _runAgain = false;
        scheduleSync();
      }
    }
  }

  // ---------------------------------------------------------------- realtime

  void _subscribeRealtime() {
    _unsubscribeRealtime();
    final channel = _client.channel('study-architect-sync');
    for (final t in _tables) {
      channel.onPostgresChanges(
        event: PostgresChangeEvent.all,
        schema: 'public',
        table: t.name,
        callback: (_) => scheduleSync(),
      );
    }
    channel.subscribe();
    _channel = channel;
  }

  void _unsubscribeRealtime() {
    final channel = _channel;
    if (channel != null) _client.removeChannel(channel);
    _channel = null;
  }

  // -------------------------------------------------------------------- meta

  Future<String?> _getMeta(Database db, String key) async {
    final rows =
        await db.query('sync_meta', where: 'key = ?', whereArgs: [key]);
    return rows.isEmpty ? null : rows.first['value'] as String;
  }

  Future<void> _setMeta(Database db, String key, String value) {
    return db.insert(
      'sync_meta',
      {'key': key, 'value': value},
      conflictAlgorithm: ConflictAlgorithm.replace,
    );
  }

  // -------------------------------------------------------------------- pull

  /// Downloads rows changed in the cloud since the last pull.
  /// Returns the names of local tables that were modified.
  Future<Set<String>> _pull(String uid) async {
    final db = await DatabaseService.instance.database;
    final touched = <String>{};

    for (final t in _tables) {
      final key = 'pull:$uid:${t.name}';
      var cursor = await _getMeta(db, key);

      while (true) {
        final base = _client.from(t.name).select();
        final rows =
            await (cursor == null ? base : base.gt('updatedAt', cursor))
                .order('updatedAt', ascending: true)
                .limit(_pageSize);

        for (final raw in rows) {
          final row = Map<String, dynamic>.from(raw);
          try {
            if (await _applyRemote(db, t, row)) touched.add(t.name);
          } catch (e) {
            debugPrint('Skipped ${t.name} row ${row['id']}: $e');
          }
          cursor = _iso(DateTime.parse(row['updatedAt'] as String));
        }

        if (cursor != null) await _setMeta(db, key, cursor);
        if (rows.length < _pageSize) break;
      }
    }
    return touched;
  }

  /// Applies one cloud row with last-write-wins. Returns true if the
  /// local table changed.
  Future<bool> _applyRemote(
    Database db,
    _SyncTable t,
    Map<String, dynamic> remote,
  ) async {
    final id = remote['id'] as String;
    final localId = t.intId ? int.parse(id) : id;
    final remoteAt = _iso(DateTime.parse(remote['updatedAt'] as String));

    final local = await db.query(
      t.name,
      columns: ['updatedAt'],
      where: 'id = ?',
      whereArgs: [localId],
    );
    final localAt = local.isEmpty ? null : local.first['updatedAt'] as String?;

    // Local copy is newer or identical: keep it.
    if (localAt != null && localAt.compareTo(remoteAt) >= 0) return false;

    // Cloud row was deleted.
    if (remote['deletedAt'] != null) {
      await db.delete(t.name, where: 'id = ?', whereArgs: [localId]);
      await db.delete(
        'sync_tombstones',
        where: 'tableName = ? AND rowId = ?',
        whereArgs: [t.name, id],
      );
      return local.isNotEmpty;
    }

    // Deleted locally after the cloud version was written: keep it deleted.
    final tomb = await db.query(
      'sync_tombstones',
      where: 'tableName = ? AND rowId = ?',
      whereArgs: [t.name, id],
    );
    if (tomb.isNotEmpty &&
        (tomb.first['deletedAt'] as String).compareTo(remoteAt) >= 0) {
      return false;
    }

    final values = Map<String, dynamic>.from(remote)
      ..remove('user_id')
      ..remove('deletedAt')
      ..['id'] = localId
      ..['updatedAt'] = remoteAt;

    if (local.isEmpty) {
      await db.insert(t.name, values);
    } else {
      // update, not "insert or replace": REPLACE would delete the row first
      // and cascade-delete its children.
      await db.update(t.name, values, where: 'id = ?', whereArgs: [localId]);
    }
    return true;
  }

  // -------------------------------------------------------------------- push

  Future<void> _push(String uid) async {
    final db = await DatabaseService.instance.database;

    // 1) New and edited rows.
    for (final t in _tables) {
      final key = 'push:$uid:${t.name}';
      final since = await _getMeta(db, key);

      final rows = await db.query(
        t.name,
        where: since == null ? null : 'updatedAt > ?',
        whereArgs: since == null ? null : [since],
        orderBy: 'updatedAt ASC',
      );
      if (rows.isEmpty) continue;

      final payload = rows
          .map((r) => <String, dynamic>{
                ...r,
                'id': r['id'].toString(),
                'user_id': uid,
                'deletedAt': null,
              })
          .toList();

      for (var i = 0; i < payload.length; i += 200) {
        final end = i + 200 > payload.length ? payload.length : i + 200;
        await _client
            .from(t.name)
            .upsert(payload.sublist(i, end), onConflict: 'user_id,id');
      }
      await _setMeta(db, key, rows.last['updatedAt'] as String);
    }

    // 2) Deletes.
    final tombstones = await db.query('sync_tombstones');
    for (final tomb in tombstones) {
      final table = tomb['tableName'] as String;
      final rowId = tomb['rowId'] as String;
      final at = tomb['deletedAt'] as String;

      await _client
          .from(table)
          .update({'deletedAt': at, 'updatedAt': at})
          .eq('user_id', uid)
          .eq('id', rowId);

      await db.delete(
        'sync_tombstones',
        where: 'tableName = ? AND rowId = ?',
        whereArgs: [table, rowId],
      );
    }
  }
}
