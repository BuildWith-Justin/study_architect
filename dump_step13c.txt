class _StudyArchitectAppState extends State<StudyArchitectApp> {
  @override
  void initState() {
    super.initState();

    // Loads the saved theme preference (if onboarding has already
    // happened) so the app opens in the right mode instead of
    // flashing system default first.
    ThemeController.instance.loadFromSettings();
  }

  /// V2: wraps the existing root route in AuthGate.
  /// - Signed in (or Supabase unavailable) -> the original root screen,
  ///   exactly what V1 showed.
  /// - Signed out -> Sign In screen.
  /// Every other route in AppRoutes.table is left untouched.
  Map<String, WidgetBuilder> _buildRoutes() {
    final originalRoot = AppRoutes.table[AppRoutes.root];

    if (originalRoot == null) {
      return AppRoutes.table;
    }

    return {
      ...AppRoutes.table,
      AppRoutes.root: (context) => AuthGate(
            signedInBuilder: (gateContext) => originalRoot(gateContext),
          ),
    };
  }

  @override
  Widget build(BuildContext context) {
    return ValueListenableBuilder<ThemeMode>(
      valueListenable: ThemeController.instance.mode,
      builder: (context, themeMode, _) {
        return MaterialApp(
          title: 'Study Architect',
          debugShowCheckedModeBanner: false,
          theme: buildLightTheme(),
          darkTheme: buildDarkTheme(),
          themeMode: themeMode,
          initialRoute: AppRoutes.root,
          routes: _buildRoutes(),
          navigatorObservers: [routeObserver],
        );
      },
    );
  }
}

/// The app's actual root widget.
///
/// Checks whether a Student profile already exists in local storage.
/// If no profile exists, onboarding is shown.
/// If a profile exists, the main tab shell is shown.
///
/// V2: if the local database is empty but the user is signed in, the
/// profile may already be in the cloud (new phone, reinstall), so one
/// sync is attempted before deciding.
class StartupDecider extends StatefulWidget {
  const StartupDecider({super.key});

  @override
  State<StartupDecider> createState() => _StartupDeciderState();
}

class _StartupDeciderState extends State<StartupDecider> {
  final _studentRepository = StudentRepository();

  bool? _onboardingComplete;
  Object? _error;

  @override
  void initState() {
    super.initState();
    _checkOnboardingStatus();
  }

  bool get _signedIn =>
      AuthService.isAvailable &&
      Supabase.instance.client.auth.currentUser != null;

  Future<void> _checkOnboardingStatus() async {
    if (!mounted) return;

    setState(() {
      _onboardingComplete = null;
      _error = null;
    });

    try {
      var student = await _studentRepository.getCurrent();

      if (student == null && _signedIn) {
        try {
          await SyncService.instance
              .syncNow()
              .timeout(const Duration(seconds: 8));
        } catch (e) {
          debugPrint('StartupDecider pre-onboarding sync skipped: $e');
        }
        student = await _studentRepository.getCurrent();
      }

      if (!mounted) return;

      setState(() {
        _onboardingComplete = student != null;
      });
    } catch (e, stackTrace) {
      debugPrint(
        'StartupDecider failed to load student: $e\n$stackTrace',
      );

      if (!mounted) return;

      setState(() {
        _error = e;
      });
    }
  }

  Future<void> _handleOnboardingComplete() async {
    await _checkOnboardingStatus();
  }

  @override
  Widget build(BuildContext context) {
    if (_error != null) {
      return GradientBackground(
        child: SafeArea(
          child: Padding(
            padding: const EdgeInsets.all(AppSpacing.lg),
            child: Center(
              child: SingleChildScrollView(
                child: Column(
                  mainAxisSize: MainAxisSize.min,
                  crossAxisAlignment: CrossAxisAlignment.center,
                  children: [
                    const Icon(
                      Icons.error_outline_rounded,
                      size: 48,
                    ),
                    const SizedBox(height: AppSpacing.md),
                    Text(
                      'Could not start Study Architect.',
                      style: Theme.of(context).textTheme.titleLarge,
                      textAlign: TextAlign.center,
                    ),
                    const SizedBox(height: AppSpacing.sm),
                    const Text(
                      'We could not load your local profile. '
                      'Please try again.',
                      textAlign: TextAlign.center,
                      style: TextStyle(
                        color: AppColors.textSecondary,
                      ),
                    ),
                    const SizedBox(height: AppSpacing.lg),
                    FilledButton.icon(
                      onPressed: _checkOnboardingStatus,
                      icon: const Icon(Icons.refresh_rounded),
                      label: const Text('Try Again'),
                    ),
                  ],
                ),
              ),
            ),
          ),
        ),
      );
    }

    if (_onboardingComplete == null) {
      return const SplashView();
    }

    if (_onboardingComplete == false) {
      return OnboardingFlow(
        onComplete: _handleOnboardingComplete,
      );
    }

    return const MainScaffold();
  }
}

import 'package:flutter/material.dart';

/// Single source of truth for the app's current ThemeMode. MaterialApp
/// listens to [mode] directly, so calling [setMode] anywhere in the
/// app (from the More/Settings screen) updates every screen
/// immediately — no restart needed.
class ThemeController {
  ThemeController._internal();
  static final ThemeController instance = ThemeController._internal();

  final ValueNotifier<ThemeMode> mode = ValueNotifier(ThemeMode.system);
  final _settingsRepo = SettingsRepository();

  Future<void> loadFromSettings() async {
    final settings = await _settingsRepo.get();
    mode.value = settings.themeMode;
  }

  Future<void> setMode(ThemeMode newMode) async {
    mode.value = newMode;
    final current = await _settingsRepo.get();
    await _settingsRepo.save(current.copyWith(themeMode: newMode));
  }
}

/// Creates a new exam (no [existing]) or edits one. Pops with `true`
/// when something was saved.
///
/// When creating, every topic of the chosen subject starts ticked, so the
/// checklist is ready in one tap. The subject cannot be changed after the
/// exam exists, because the checklist is made of that subject's topics.
class AddEditExamScreen extends StatefulWidget {
  final Exam? existing;
  final String? initialSubjectId;

  const AddEditExamScreen({super.key, this.existing, this.initialSubjectId});

  @override
  State<AddEditExamScreen> createState() => _AddEditExamScreenState();

class _AddEditExamScreenState extends State<AddEditExamScreen> {
  final TextEditingController _titleController = TextEditingController();
  final TextEditingController _locationController = TextEditingController();
  final TextEditingController _notesController = TextEditingController();

  List<Subject> _subjects = [];
  List<Topic> _topics = [];
  final Set<String> _selected = {};
  String? _subjectId;

  late DateTime _date;
  late TimeOfDay _time;

  bool _loading = true;
  bool _saving = false;
  String? _error;

  bool get _isEditing => widget.existing != null;

  bool get _canSave =>
      _titleController.text.trim().isNotEmpty && _subjectId != null;

  @override
  void initState() {
    super.initState();

    final existing = widget.existing;
    if (existing != null) {
      _titleController.text = existing.title;
      _locationController.text = existing.location;
      _notesController.text = existing.notes;
      _date = DateTime(
        existing.examDate.year,
        existing.examDate.month,
        existing.examDate.day,
      );
      _time = TimeOfDay(
        hour: existing.examDate.hour,
        minute: existing.examDate.minute,
      );
    } else {
      final inTwoWeeks = DateTime.now().add(const Duration(days: 14));
      _date = DateTime(inTwoWeeks.year, inTwoWeeks.month, inTwoWeeks.day);
      _time = const TimeOfDay(hour: 9, minute: 0);
    }

    _init();
  }

  @override
  void dispose() {
    _titleController.dispose();
    _locationController.dispose();
    _notesController.dispose();
    super.dispose();
  }

  Future<void> _init() async {
    try {
      final subjects = await SubjectRepository().getAll();

      String? subjectId = widget.existing?.subjectId ?? widget.initialSubjectId;
      if (!_isEditing &&
          (subjectId == null || !subjects.any((s) => s.id == subjectId))) {
        subjectId = subjects.isEmpty ? null : subjects.first.id;
      }

      var topics = <Topic>[];
      var selected = <String>{};
      if (subjectId != null) {
        topics = await TopicRepository().getForSubject(subjectId);
        if (_isEditing) {
          final links =
              await ExamRepository().getTopicLinks(widget.existing!.id);
          selected = links.map((l) => l.topicId).toSet();
        } else {
          selected = topics.map((t) => t.id).toSet();
        }
      }

      if (!mounted) return;

      setState(() {
        _subjects = subjects;
        _subjectId = subjectId;
        _topics = topics;
        _selected
          ..clear()
          ..addAll(selected);
        _loading = false;
      });
    } catch (_) {
      if (!mounted) return;
      setState(() {
        _loading = false;
        _error = 'Could not load your subjects. Please go back and try again.';
      });
    }
  }

  Future<void> _pickSubject(String id) async {
    if (_isEditing || id == _subjectId) return;

    try {
      final topics = await TopicRepository().getForSubject(id);
      if (!mounted) return;
      setState(() {
        _subjectId = id;
        _topics = topics;
        _selected
          ..clear()
          ..addAll(topics.map((t) => t.id));
        _error = null;
      });
    } catch (_) {
      if (!mounted) return;
      setState(() => _error = 'Could not load that subject\'s topics.');
    }
  }

  Future<void> _pickDate() async {
    final now = DateTime.now();
    final base = DateTime(now.year - 1);
    final end = DateTime(now.year + 3, 12, 31);

    final picked = await showDatePicker(
      context: context,
      initialDate: _date,
      firstDate: _date.isBefore(base) ? DateTime(_date.year) : base,
      lastDate: _date.isAfter(end) ? DateTime(_date.year + 1, 12, 31) : end,
    );

    if (picked != null && mounted) {
      setState(() => _date = picked);
    }
  }

  Future<void> _pickTime() async {
    final picked = await showTimePicker(context: context, initialTime: _time);

    if (picked != null && mounted) {
      setState(() => _time = picked);
    }
  }

  Future<void> _save() async {
    if (!_canSave || _saving) return;

    FocusScope.of(context).unfocus();

    setState(() {
      _saving = true;
      _error = null;
    });

    try {
      final repo = ExamRepository();
      final examDate = DateTime(
        _date.year,
        _date.month,
        _date.day,
        _time.hour,
        _time.minute,
      );
      // Keep roadmap order.
      final topicIds = _topics
          .where((t) => _selected.contains(t.id))
          .map((t) => t.id)
          .toList();

      final existing = widget.existing;
      if (existing != null) {
        await repo.update(
          existing.copyWith(
            title: _titleController.text.trim(),
            examDate: examDate,
            location: _locationController.text.trim(),
            notes: _notesController.text.trim(),
          ),
          topicIds: topicIds,
        );
      } else {
        await repo.create(
          Exam(
            id: const Uuid().v4(),
            subjectId: _subjectId!,
            title: _titleController.text.trim(),
            examDate: examDate,
            location: _locationController.text.trim(),
            notes: _notesController.text.trim(),
            createdAt: DateTime.now(),
          ),
          topicIds: topicIds,
        );
      }

      DataRefreshBus.instance.notify(RefreshScope.exams);

      if (!mounted) return;
      Navigator.of(context).pop(true);
    } catch (_) {
      if (!mounted) return;
      setState(() {
        _saving = false;
        _error = 'Could not save the exam. Please try again.';
      });
    }
  }

        RefreshScope.exams,
        RefreshScope.subjects,

    super.initState();

  Future<_ExamDetailData?> _load() async {
    final examRepo = ExamRepository();
    final exam = await examRepo.getById(widget.examId);
    if (exam == null) return null;

    final subject = await SubjectRepository().getById(exam.subjectId);
    final links = await examRepo.getTopicLinks(widget.examId);
    final topics = await TopicRepository().getForSubject(exam.subjectId);
    final topicsById = {for (final t in topics) t.id: t};

    final rows = [
      for (final l in links) _TopicRow(l, topicsById[l.topicId]),
    ];
    final prepared = links.where((l) => l.prepared).length;

    return _ExamDetailData(
      exam: exam,
      subject: subject,
      rows: rows,
      progress: ExamProgress(prepared: prepared, total: links.length),
    );
  }

  Future<void> _reload() async {
    final next = _load();

    if (!mounted) {
      next.ignore();
      return;
    }

    setState(() {
      _future = next;
    });

    try {
      await next;
    } catch (_) {
      // The FutureBuilder shows the error state.
    }
  }

    } catch (_) {
      if (!mounted) return;

        const SnackBar(content: Text('Could not update that topic.')),

    }

import '../../models/exam.dart';
import '../../models/subject.dart';
import '../../repositories/exam_repository.dart';
import '../../repositories/subject_repository.dart';

/// An exam plus the two things every list row needs: its subject and
/// how far along its preparation checklist is.
class ExamSummary {
  final Exam exam;
  final Subject? subject;
  final ExamProgress progress;

  const ExamSummary({
    required this.exam,
    required this.subject,
    required this.progress,
  });
}

/// Everything the exams list needs, loaded in one pass.
class ExamsViewData {
  /// From today onwards, nearest first.
  final List<ExamSummary> upcoming;

  /// Exam day already gone by, most recent first.
  final List<ExamSummary> past;

  const ExamsViewData({required this.upcoming, required this.past});

  bool get isEmpty => upcoming.isEmpty && past.isEmpty;
  int get totalCount => upcoming.length + past.length;

  static Future<ExamsViewData> load() async {
    final examRepo = ExamRepository();
    final exams = await examRepo.getAll();
    final subjects = await SubjectRepository().getAll();
    final progress = await examRepo.getProgressForAll();
    final subjectsById = {for (final s in subjects) s.id: s};

    final upcoming = <ExamSummary>[];
    final past = <ExamSummary>[];

    for (final exam in exams) {
      final summary = ExamSummary(
        exam: exam,
        subject: subjectsById[exam.subjectId],
        progress: progress[exam.id] ?? const ExamProgress.empty(),
      );
      if (exam.isPast()) {
        past.add(summary);
      } else {
        upcoming.add(summary);
      }
    }

    past.sort((a, b) => b.exam.examDate.compareTo(a.exam.examDate));
    return ExamsViewData(upcoming: upcoming, past: past);
  }
}

class _CreateSessionScreenState extends State<CreateSessionScreen> {
  static const _lengthOptions = [25, 45, 60, 90, 120];

  final _subjectRepo = SubjectRepository();
  final _topicRepo = TopicRepository();

  List<Subject> _subjects = [];
  List<Topic> _topics = [];

  String? _subjectId;
  String? _topicId;
  int _minutes = 60;

  bool _loading = true;
  String? _error;

  @override
  void initState() {
    super.initState();
    _loadSubjects();
  }

  Subject? get _selectedSubject {
    for (final s in _subjects) {
      if (s.id == _subjectId) return s;

  }

  Future<void> _loadSubjects() async {
    try {
      final subjects = await _subjectRepo.getAll();

      if (!mounted) return;

      setState(() {
        _subjects = subjects;
        _subjectId = subjects.isNotEmpty ? subjects.first.id : null;
        _loading = false;
        _error = null;
      });

      if (_subjectId != null) {
        await _loadTopics(_subjectId!);
      }
    } catch (_) {
      if (!mounted) return;

      setState(() {
        _loading = false;
        _error = 'Could not load your subjects.';
      });
    }
  }

  Future<void> _loadTopics(String subjectId) async {
    try {
      final topics = await _topicRepo.getForSubject(subjectId);

      if (!mounted) return;

      setState(() {
        _topics = topics;
        if (!_topics.any((t) => t.id == _topicId)) {
          _topicId = null;
        }

/// Runs a live study session, either against an existing planned
/// StudySession (pass [existingSessionId]) or as a fresh ad-hoc one
/// (leave it null — a new session is created immediately, in progress).
///
/// Elapsed time is always computed as DateTime.now().difference(startedAt)
/// minus accumulated pause time, never by counting timer ticks. Ticks
/// don't fire while the app is backgrounded, so a tick-counted timer
/// would silently lose that time; a timestamp diff recomputes correctly
/// the instant the UI updates again, however long the app was away.
///
/// V2 Step 8b: the display is a countdown ring (time remaining until
/// [goal]). Past the goal the ring turns green and the centre shows
/// overtime. The timing logic itself is unchanged from V1.
///
/// V2 Step 8c: ending the session saves it as completed straight away,
/// then opens the Session Complete screen (mood + notes + celebration).
/// The close (X) button asks first and discards a session that was
/// just started from scratch.

  final String? topicId;
  final String? topicName;

    super.key,

    required this.subjectName,
    required this.subjectColorValue,
    this.topicId,

  bool get _isPaused => _pauseStartedAt != null;

  bool get _overGoal => _elapsed >= widget.goal;

  Duration get _remaining {
    final r = widget.goal - _elapsed;
    return r.isNegative ? Duration.zero : r;
  }

  Duration get _overtime => _overGoal ? _elapsed - widget.goal : Duration.zero;

  /// How much of the ring is still filled (1.0 at the start, 0.0 at the goal).
  double get _remainingFraction {
    final goalMs = widget.goal.inMilliseconds;
    if (goalMs <= 0) return 0;
    return (1 - _elapsed.inMilliseconds / goalMs).clamp(0.0, 1.0).toDouble();
  }

  @override

    super.initState();

    WidgetsBinding.instance.addObserver(this);

  }

  Future<void> _init() async {
    try {
      if (widget.existingSessionId != null) {
        final existing = await _sessionRepo.getById(
          widget.existingSessionId!,
        );

        if (!mounted) return;

        if (existing != null) {
          // Resuming a session already marked in progress keeps its
          // original start timestamp. This keeps elapsed time correct
          // even if the screen was closed and reopened.
          final startedAt = existing.status == SessionStatus.inProgress
              ? existing.startTime
              : DateTime.now();

          final updated = existing.copyWith(
            status: SessionStatus.inProgress,
            startTime: startedAt,
          );

          await _sessionRepo.update(updated);

          if (!mounted) return;

          _session = updated;
          _startedAt = startedAt;
        }
      }

      if (_session == null) {
        final now = DateTime.now();

        final session = StudySession(
          id: const Uuid().v4(),
          subjectId: widget.subjectId,
          topicId: widget.topicId,
          scheduledDate: DateTime(
            now.year,
            now.month,
            now.day,
          ),
          startTime: now,
          // Placeholder end time to satisfy the repository's
          // end-after-start rule. Replaced when the session ends.
          endTime: now.add(widget.goal),
          status: SessionStatus.inProgress,
        );

        await _sessionRepo.create(session);

        if (!mounted) return;

        _session = session;
        _startedAt = now;
      }

      if (!mounted) return;

      setState(() {
        _loading = false;
        _error = null;
      });

      _startTicker();
    } catch (_) {

        _loading = false;
        _error = 'Could not start the study session. Please try again.';

class _ConfettiPiece {
  final double x; // 0..1 across the screen
  final double startY; // starts slightly above the screen
  final double speed;
  final double sway;
  final double phase;
  final double spin;
  final double width;
  final double height;
  final Color color;

  const _ConfettiPiece({
    required this.x,
    required this.startY,
    required this.speed,
    required this.sway,
    required this.phase,
    required this.spin,
    required this.width,
    required this.height,
    required this.color,
  });
}

class _ConfettiPainter extends CustomPainter {
  final double progress; // 0..1
  final List<_ConfettiPiece> pieces;

  const _ConfettiPainter({
    required this.progress,
    required this.pieces,
  });

  @override
  void paint(Canvas canvas, Size size) {
    if (progress >= 1) return;

    // Fade everything out over the last 30% of the animation.
    final fade = progress < 0.7 ? 1.0 : (1 - (progress - 0.7) / 0.3);

    for (final p in pieces) {
      final y = (p.startY + p.speed * progress * 1.25) * size.height;
      if (y > size.height || y < -20) continue;

      final x = (p.x + math.sin(p.phase + progress * 6) * p.sway) * size.width;

      final paint = Paint()
        ..color = p.color.withValues(alpha: fade.clamp(0.0, 1.0));

      canvas.save();
      canvas.translate(x, y);
      canvas.rotate(p.phase + progress * p.spin);
      canvas.drawRRect(
        RRect.fromRectAndRadius(
          Rect.fromCenter(
            center: Offset.zero,
            width: p.width,
            height: p.height,
          ),
          const Radius.circular(2),
        ),
        paint,
      );
      canvas.restore();
    }
  }

  @override
  bool shouldRepaint(covariant _ConfettiPainter old) {
    return old.progress != progress;
  }
}

    final value = _topicController.text.trim();
    if (value.isEmpty) return;

    final exists = _topics.any((t) => t.toLowerCase() == value.toLowerCase());

    setState(() {

  Future<void> _save() async {
    final name = _nameController.text.trim();

    if (name.isEmpty || _saving) return;

    FocusScope.of(context).unfocus();

    // A topic typed but not yet added with "+" still counts.
    final pendingTopic = _topicController.text.trim();
    if (!_isEditing && pendingTopic.isNotEmpty) {
      final exists =
          _topics.any((t) => t.toLowerCase() == pendingTopic.toLowerCase());
      if (!exists) _topics.add(pendingTopic);
      _topicController.clear();
    }

    setState(() {
      _saving = true;
      _error = null;
    });

    var topicsFailed = false;

    try {
      final repo = SubjectRepository();

      if (_isEditing) {
        final updated = widget.existing!.copyWith(
          name: name,
          difficulty: _difficulty,
          colorValue: _color.toARGB32(),
          weeklyTargetMinutes: _weeklyTargetHours.round() * 60,
        );

        await repo.update(updated);
      } else {
        final subject = Subject(
          id: const Uuid().v4(),
          name: name,
          difficulty: _difficulty,
          colorValue: _color.toARGB32(),
          weeklyTargetMinutes: _weeklyTargetHours.round() * 60,
          createdAt: DateTime.now(),
        );

        await repo.create(subject);

        // Topics are created one by one, in the order listed; the
        // repository appends each one at the end of the roadmap.
        // If this part fails the subject already exists, so we do not
        // report a failure (a retry would create a duplicate subject).
        try {
          final topicRepo = TopicRepository();
          for (final topicName in _topics) {
            await topicRepo.create(
              Topic(
                id: const Uuid().v4(),
                subjectId: subject.id,
                name: topicName,
              ),
            );
          }
        } catch (_) {
          topicsFailed = true;
        }
      }

      if (!mounted) return;

      if (topicsFailed) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(
            content: Text(
              'Subject saved, but some topics could not be added. '
              'You can add them from the subject page.',
            ),
          ),
        );
      }

      Navigator.of(context).pop(true);
    } catch (_) {
      if (!mounted) return;

      setState(() {
        _saving = false;
        _error = 'Could not save the subject. Please try again.';
      });

class SubjectDetailScreen extends StatefulWidget {
  final String subjectId;

  const SubjectDetailScreen({
    super.key,
    required this.subjectId,
  });

  @override
  State<SubjectDetailScreen> createState() => _SubjectDetailScreenState();
}

class _SubjectDetailScreenState extends State<SubjectDetailScreen>
    with WidgetsBindingObserver, RefreshableScreenMixin {
  final _subjectRepo = SubjectRepository();
  final _topicRepo = TopicRepository();
  final _sessionRepo = SessionRepository();

  late Future<_DetailData> _future;

  /// Topic ids in the order the person just dragged them to. Shown
  /// immediately so the list does not jump back while the new order is
  /// being saved; cleared once the reload from the database is done.
  List<String>? _pendingOrder;

  @override
  void initState() {
    super.initState();

    WidgetsBinding.instance.addObserver(this);

    _future = _load();
  }

  @override
  void dispose() {
    WidgetsBinding.instance.removeObserver(this);
    super.dispose();
  }

  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    if (state == AppLifecycleState.resumed && mounted) {
      _reload();
    }
  }

  Future<_DetailData> _load() async {
    final subject = await _subjectRepo.getById(widget.subjectId);

    if (subject == null) {
      throw StateError('Subject not found');
    }

    final topics = await _topicRepo.getForSubject(widget.subjectId);
    final sessions = await _sessionRepo.getForSubject(widget.subjectId);

    return _DetailData(
      subject: subject,
      topics: topics,
      sessions: sessions,
    );
  }

  void _reload() {
    if (!mounted) return;

    setState(() {
      _future = _load();
    });
  }

  Future<void> _refresh() async {
    if (!mounted) return;

    final future = _load();

    setState(() {
      _future = future;
    });

    try {
      await future;
    } catch (_) {
      // AsyncBuilder will display the error state.
    }
  }

  /// Called by RefreshableScreenMixin when sync brings in cloud changes.
  @override
  Future<void> onRefresh() => _refresh();

  /// Tells the other screens (for example the Subjects list underneath
  /// this one) that subject or topic data changed.
  void _notifyChanged() {
    DataRefreshBus.instance.notify(RefreshScope.all);
  }

  /// Applies the not-yet-saved drag order on top of the loaded topics.
  List<Topic> _ordered(List<Topic> topics) {
    final pending = _pendingOrder;
    if (pending == null) return topics;

    final byId = {for (final t in topics) t.id: t};
    final result = <Topic>[];
    for (final id in pending) {
      final topic = byId.remove(id);
      if (topic != null) result.add(topic);
    }
    result.addAll(byId.values);
    return result;
  }

  // --------------------------------------------------------------- topics

  /// Asks for a topic name. Returns null if cancelled.
  Future<String?> _promptTopicName({
    required String title,
    required String confirmLabel,
    String initial = '',
  }) async {
    final controller = TextEditingController(text: initial);
    controller.selection = TextSelection(
      baseOffset: 0,
      extentOffset: initial.length,
    );

    TransitionRoute<dynamic>? dialogRoute;

    final name = await showDialog<String>(
      context: context,
      builder: (dialogContext) {
        dialogRoute = ModalRoute.of(dialogContext);

        return AlertDialog(
          title: Text(title),
          content: TextField(
            controller: controller,
            autofocus: true,
            textCapitalization: TextCapitalization.sentences,
            decoration: const InputDecoration(
              hintText: 'e.g. Differential Equations',
            ),
            onSubmitted: (value) {
              Navigator.pop(dialogContext, value.trim());
            },
          ),
          actions: [
            TextButton(
              onPressed: () => Navigator.pop(dialogContext),
              child: const Text('Cancel'),
            ),
            TextButton(
              onPressed: () {
                Navigator.pop(
                  dialogContext,
                  controller.text.trim(),
                );
              },
              child: Text(confirmLabel),
            ),
          ],
        );
      },
    );

    // Wait until the dialog route has completely finished
    // its exit transition and has been removed from the overlay.
    if (dialogRoute != null) {
      await dialogRoute!.completed;
    }

    controller.dispose();

    return name;
  }

  Future<void> _addTopic() async {
    final name = await _promptTopicName(
      title: 'New topic',
      confirmLabel: 'Add',
    );

    if (name == null || name.isEmpty) return;

    await _topicRepo.create(
      Topic(
        id: const Uuid().v4(),
        subjectId: widget.subjectId,
        name: name,
      ),
    );

    if (!mounted) return;

    await _refresh();
    _notifyChanged();
  }

  Future<void> _renameTopic(Topic topic) async {
    final name = await _promptTopicName(
      title: 'Rename topic',
      confirmLabel: 'Save',
      initial: topic.name,
    );

    if (name == null || name.isEmpty || name == topic.name) return;

    await _topicRepo.rename(topic.id, name);

    if (!mounted) return;

    await _refresh();
    _notifyChanged();
  }

  Future<void> _toggleTopic(Topic topic) async {
    await _topicRepo.setCompleted(
      topic.id,
      !topic.completed,
    );

    if (!mounted) return;

    await _refresh();
    _notifyChanged();
  }

  Future<void> _deleteTopic(Topic topic) async {
    final confirmed = await showDialog<bool>(
      context: context,
      builder: (dialogContext) => AlertDialog(
        title: const Text('Delete topic?'),
        content: Text(
          '"${topic.name}" will be removed from the roadmap. '
          'Sessions and tasks linked to it are kept but unlinked.',
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(dialogContext, false),
            child: const Text('Cancel'),
          ),
          TextButton(
            onPressed: () => Navigator.pop(dialogContext, true),
            child: const Text(
              'Delete',
              style: TextStyle(color: AppColors.danger),
            ),
          ),
        ],
      ),
    );

    if (confirmed != true) return;

    await _topicRepo.delete(topic.id);

    if (!mounted) return;

    await _refresh();
    _notifyChanged();
  }

  Future<void> _onReorder(
    List<Topic> current,
    int oldIndex,
    int newIndex,
  ) async {
    if (newIndex > oldIndex) newIndex -= 1;
    if (newIndex == oldIndex) return;

    final ids = current.map((t) => t.id).toList();
    final moved = ids.removeAt(oldIndex);
    ids.insert(newIndex, moved);

    // 1) Instant feedback on screen.
    setState(() {
      _pendingOrder = ids;
    });

    // 2) Save to the database.
    try {
      await _topicRepo.reorder(ids);
    } catch (_) {
      if (!mounted) return;
      setState(() {
        _pendingOrder = null;
      });
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Could not save the new order.')),
      );
      return;
    }

    if (!mounted) return;

    // 3) Reload from the database, then drop the temporary order.
    await _refresh();

    if (!mounted) return;

    setState(() {
      _pendingOrder = null;
    });
    _notifyChanged();
  }

  // -------------------------------------------------------------- subject

  Future<void> _confirmDelete(
    BuildContext context,
    Subject subject,
  ) async {
    final relatedCount = await _subjectRepo.countRelatedRecords(subject.id);

    if (!context.mounted) return;

    final confirmed = await showDialog<bool>(
      context: context,
      builder: (context) => AlertDialog(
        title: const Text('Delete subject?'),
        content: Text(
          relatedCount > 0
              ? 'This will also delete $relatedCount related topic(s), session(s), and task(s). This cannot be undone.'
              : 'This cannot be undone.',
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(context, false),
            child: const Text('Cancel'),
          ),
          TextButton(
            onPressed: () => Navigator.pop(context, true),
            child: const Text(
              'Delete',
              style: TextStyle(
                color: Colors.redAccent,
              ),
            ),
          ),
        ],
      ),
    );

    if (confirmed != true) return;

    await _subjectRepo.delete(subject.id);

    if (!context.mounted) return;

    Navigator.of(context).pop(true);
  }

  // ------------------------------------------------------------ UI pieces

  Widget _summaryCard(
    BuildContext context, {
    required Subject subject,
    required Color accent,
    required int topicCount,
    required int completedTopics,
    required int percent,
    required int thisWeekMinutes,
  }) {
    final theme = Theme.of(context);
    final targetMinutes = subject.weeklyTargetMinutes;
    final weekProgress = targetMinutes <= 0
        ? 0.0
        : (thisWeekMinutes / targetMinutes).clamp(0.0, 1.0).toDouble();

    return Container(
      padding: const EdgeInsets.all(AppSpacing.lg),
      decoration: BoxDecoration(
        color: context.cardSurfaceAlt,
        borderRadius: BorderRadius.circular(AppRadii.card),
      ),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Row(
            mainAxisAlignment: MainAxisAlignment.spaceBetween,
            children: [
              Text(
                'Roadmap progress',
                style: theme.textTheme.bodySmall?.copyWith(
                  color: context.secondaryText,
                  fontWeight: FontWeight.w600,
                ),
              ),
              Text(
                '$percent%',
                style: TextStyle(
                  fontSize: 22,
                  fontWeight: FontWeight.w700,
                  color: accent,
                ),
              ),
            ],
          ),
          const SizedBox(height: AppSpacing.sm),
          ClipRRect(
            borderRadius: BorderRadius.circular(4),
            child: LinearProgressIndicator(
              value: topicCount == 0 ? 0 : percent / 100,
              minHeight: 8,
              backgroundColor: accent.withValues(alpha: 0.15),
              valueColor: AlwaysStoppedAnimation<Color>(accent),
            ),
          ),
          const SizedBox(height: AppSpacing.xs),
          Text(
            '$completedTopics of $topicCount topics done',
            style: theme.textTheme.bodySmall?.copyWith(
              color: context.secondaryText,
            ),
          ),
          const SizedBox(height: AppSpacing.lg),
          Row(
            mainAxisAlignment: MainAxisAlignment.spaceBetween,
            children: [
              Text(
                'This week',
                style: theme.textTheme.bodySmall?.copyWith(
                  color: context.secondaryText,
                  fontWeight: FontWeight.w600,
                ),
              ),
              Text(
                '${formatDuration(thisWeekMinutes)} of ${targetMinutes ~/ 60}h',
                style: theme.textTheme.bodyMedium?.copyWith(
                  fontWeight: FontWeight.w500,
                  color: context.primaryText,
                ),
              ),
            ],
          ),
          const SizedBox(height: AppSpacing.sm),
          ClipRRect(
            borderRadius: BorderRadius.circular(4),
            child: LinearProgressIndicator(
              value: weekProgress,
              minHeight: 8,
              backgroundColor: accent.withValues(alpha: 0.15),
              valueColor: AlwaysStoppedAnimation<Color>(accent),
            ),
          ),
        ],
      ),
    );
  }

  Widget _roadmapSection(
    BuildContext context, {
    required List<Topic> topics,
    required Color accent,
  }) {
    final theme = Theme.of(context);
    final nextIndex = topics.indexWhere((t) => !t.completed);

    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Row(
          mainAxisAlignment: MainAxisAlignment.spaceBetween,
          children: [
            Expanded(
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  Text(
                    'Study roadmap',
                    style: theme.textTheme.titleMedium?.copyWith(
                      fontWeight: FontWeight.w600,
                    ),
                  ),
                  if (topics.length > 1)
                    Text(
                      'Drag the handle to reorder  ·  tap a topic to rename',
                      style: theme.textTheme.bodySmall?.copyWith(
                        color: context.secondaryText,
                      ),
                    ),
                ],
              ),
            ),
            IconButton(
              icon: const Icon(
                Icons.add_circle_rounded,
                color: AppColors.primary,
              ),
              tooltip: 'Add topic',
              onPressed: _addTopic,
            ),
          ],
        ),
        const SizedBox(height: AppSpacing.sm),
        if (topics.isEmpty)
          Container(
            width: double.infinity,
            padding: const EdgeInsets.all(AppSpacing.lg),
            decoration: BoxDecoration(
              color: context.cardSurfaceAlt,
              borderRadius: BorderRadius.circular(AppRadii.cardSmall),
            ),
            child: Text(
              'No topics yet — tap + to add the first step of your roadmap.',
              textAlign: TextAlign.center,
              style: theme.textTheme.bodySmall?.copyWith(
                color: context.secondaryText,
              ),
            ),
          )
        else
          ReorderableListView.builder(
            shrinkWrap: true,
            physics: const NeverScrollableScrollPhysics(),
            buildDefaultDragHandles: false,
            itemCount: topics.length,
            onReorder: (oldIndex, newIndex) =>
                _onReorder(topics, oldIndex, newIndex),
            itemBuilder: (context, index) {
              final topic = topics[index];

              return RoadmapTopicTile(
                key: ValueKey(topic.id),
                index: index,
                name: topic.name,
                completed: topic.completed,
                isNext: index == nextIndex,
                accent: accent,
                onChanged: (_) => _toggleTopic(topic),
                onTap: () => _renameTopic(topic),
                onDelete: () => _deleteTopic(topic),
              );
            },
          ),
      ],
    );
  }

  // ---------------------------------------------------------------- build

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: GradientBackground(
        child: SafeArea(
          child: AsyncBuilder<_DetailData>(
            future: _future,
            errorMessage: 'Could not load this subject.',
            onRetry: _reload,
            builder: (context, data) {
              final subject = data.subject;
              final accent = Color(subject.colorValue);
              final topics = _ordered(data.topics);

              final completedTopics = topics.where((t) => t.completed).length;

              final percent = topics.isEmpty
                  ? 0
                  : ((completedTopics / topics.length) * 100).round();

              final weekStart = startOfWeek(DateTime.now());
              final weekEnd = endOfWeek(DateTime.now());

              final thisWeekMinutes = data.sessions
                  .where(
                    (s) =>
                        s.complete &&
                        s.scheduledDate.isAfter(
                          weekStart.subtract(
                            const Duration(seconds: 1),
                          ),
                        ) &&
                        s.scheduledDate.isBefore(weekEnd),
                  )
                  .fold<int>(
                    0,
                    (sum, s) => sum + (s.actualDuration ?? 0),
                  );

              final completedSessions =
                  data.sessions.where((s) => s.complete).toList();

              return Column(
                children: [
                  AppBar(
                    title: Text(subject.name),
                    leading: const BackButton(),
                    actions: [
                      IconButton(
                        icon: const Icon(Icons.edit_outlined),
                        tooltip: 'Edit subject',
                        onPressed: () async {
                          await Navigator.of(context).push<bool>(
                            MaterialPageRoute(
                              builder: (_) => AddEditSubjectScreen(
                                existing: subject,
                              ),
                            ),
                          );

                          if (!mounted) return;

                          // Refresh whether or not something was saved,
                          // in case data changed through another route.
                          await _refresh();
                          _notifyChanged();
                        },
                      ),
                      IconButton(
                        icon: const Icon(Icons.delete_outline),
                        tooltip: 'Delete subject',
                        onPressed: () => _confirmDelete(context, subject),
                      ),
                    ],
                  ),
                  Expanded(
                    child: RefreshIndicator(
                      onRefresh: _refresh,
                      child: ListView(
                        physics: const AlwaysScrollableScrollPhysics(),
                        padding: const EdgeInsets.all(AppSpacing.lg),
                        children: [
                          _summaryCard(
                            context,
                            subject: subject,
                            accent: accent,
                            topicCount: topics.length,
                            completedTopics: completedTopics,
                            percent: percent,
                            thisWeekMinutes: thisWeekMinutes,
                          ),
                          const SizedBox(height: AppSpacing.lg),
                          _roadmapSection(
                            context,
                            topics: topics,
                            accent: accent,
                          ),
                          const SizedBox(height: AppSpacing.lg),
                          Text(
                            'Recent Sessions',
                            style: Theme.of(context)
                                .textTheme
                                .titleMedium
                                ?.copyWith(fontWeight: FontWeight.w600),
                          ),
                          const SizedBox(
                            height: AppSpacing.sm,
                          ),
                          ...completedSessions.take(5).map(
                                (session) => _RecentSessionRow(
                                  session: session,
                                ),
                              ),
                          if (completedSessions.isEmpty)
                            Text(
                              'No sessions recorded yet.',
                              style: Theme.of(context)
                                  .textTheme
                                  .bodySmall
                                  ?.copyWith(
                                    color: context.secondaryText,
                                  ),
                            ),
                          const SizedBox(
                            height: AppSpacing.lg,
                          ),
                          PrimaryButton(
                            label: 'Start Session',
                            onPressed: () async {
                              await Navigator.of(context).push(
                                MaterialPageRoute(
                                  builder: (_) => FocusTimerScreen(
                                    subjectId: subject.id,
                                    subjectName: subject.name,
                                    subjectColorValue: subject.colorValue,
                                  ),
                                ),
                              );

                              if (!mounted) return;

                              _reload();
                            },
                          ),
                          const SizedBox(
                            height: AppSpacing.lg,
                          ),
                        ],
                      ),
                    ),
                  ),
                ],
              );
            },
          ),
        ),
      ),
    );
  }
}

class _DetailData {
  final Subject subject;
  final List<Topic> topics;
  final List<StudySession> sessions;

  const _DetailData({
    required this.subject,
    required this.topics,
    required this.sessions,
  });
}

class _RecentSessionRow extends StatelessWidget {
  final StudySession session;

  const _RecentSessionRow({
    required this.session,
  });

  @override
  Widget build(BuildContext context) {
    final weekday = _weekdayName(session.scheduledDate.weekday);

    return Padding(
      padding: const EdgeInsets.symmetric(vertical: 6),
      child: Row(
        mainAxisAlignment: MainAxisAlignment.spaceBetween,
        children: [
          Text(
            weekday,
            style: Theme.of(context).textTheme.bodyMedium,
          ),
          Text(
            formatDuration(
              session.actualDuration ?? 0,
            ),
            style: Theme.of(context).textTheme.bodyMedium?.copyWith(
                  color: context.secondaryText,
                ),
          ),
        ],
      ),
    );
  }

  String _weekdayName(int weekday) {
    const names = [
      'Monday',
      'Tuesday',
      'Wednesday',
      'Thursday',
      'Friday',
      'Saturday',
      'Sunday',
    ];

    return names[weekday - 1];
  }
}

class _SubjectsScreenState extends State<SubjectsScreen>
    with WidgetsBindingObserver, RefreshableScreenMixin {

  bool _showExams = false;

  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addObserver(this);
    _future = SubjectSummary.loadAll();
  }

  @override
  void dispose() {
    WidgetsBinding.instance.removeObserver(this);
    super.dispose();
  }

  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    if (state == AppLifecycleState.resumed) {
      _reload();

  Future<void> _reload() async {
    final future = SubjectSummary.loadAll();

    if (!mounted) return;

    setState(() {
      _future = future;
    });

    try {
      await future;
    } catch (_) {
      // The FutureBuilder displays the error state.
    }
  }

  /// Called by RefreshableScreenMixin when sync brings in cloud changes.
  @override
  Future<void> onRefresh() => _reload();

  Future<void> _openAdd() async {
    await Navigator.of(context).push<bool>(
      MaterialPageRoute(
        builder: (_) => const AddEditSubjectScreen(),
      ),
    );

  }

                mainAxisSize: MainAxisSize.min,

import '../../models/subject.dart';
import '../../repositories/subject_repository.dart';
import '../../repositories/topic_repository.dart';

/// A subject plus the two numbers its list-row card needs: how many
/// topics it has, and what fraction of them are completed.
class SubjectSummary {
  final Subject subject;
  final int topicCount;
  final int completionPercent;

  const SubjectSummary({
    required this.subject,
    required this.topicCount,
    required this.completionPercent,
  });

  static Future<List<SubjectSummary>> loadAll() async {
    final subjectRepo = SubjectRepository();
    final topicRepo = TopicRepository();

    final subjects = await subjectRepo.getAll();
    final summaries = <SubjectSummary>[];

    for (final subject in subjects) {
      final topics = await topicRepo.getForSubject(subject.id);
      final completed = topics.where((t) => t.completed).length;
      final percent =
          topics.isEmpty ? 0 : ((completed / topics.length) * 100).round();
      summaries.add(SubjectSummary(
        subject: subject,
        topicCount: topics.length,
        completionPercent: percent,
      ));
    }

    return summaries;
  }
}

      debugPrint(
        '[Tasks] reload: overdue=${data.overdue.length} today=${data.today.length} '
        'tomorrow=${data.tomorrow.length} upcoming=${data.upcoming.length}',
      );
      setState(() {
        _data = data;
        _error = null;
        _pending.clear();
      });

      debugPrint('[Tasks] reload failed: $e\n$st');
      if (!mounted) return;
      setState(() {
        if (_data == null) _error = e;
      });
    }

    debugPrint('[Tasks] toggle ${task.id} -> $newValue');

    setState(() => _pending[task.id] = newValue);

    // 2) Save to the database.
    try {
      await _taskRepo.setCompleted(task.id, newValue);
      debugPrint('[Tasks] saved ${task.id}');
    } catch (e, st) {
      debugPrint('[Tasks] toggle failed: $e\n$st');
      if (!mounted) return;
      setState(() => _pending.remove(task.id));
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('Could not update the task: $e')),

/// All tasks bucketed the way the Tasks screen displays them, plus
/// subject lookups for each task's colored tag.
class TasksViewData {
  final List<Task> overdue;
  final List<Task> today;
  final List<Task> tomorrow;
  final List<Task> upcoming;
  final Map<String, Subject> subjectsById;

  const TasksViewData({
    required this.overdue,
    required this.today,
    required this.tomorrow,
    required this.upcoming,
    required this.subjectsById,
  });

  static Future<TasksViewData> load() async {
    final taskRepo = TaskRepository();
    final subjectRepo = SubjectRepository();

    final allTasks = await taskRepo.getAll();
    final subjects = await subjectRepo.getAll();
    final subjectsById = {for (final s in subjects) s.id: s};

    final now = DateTime.now();
    final today = startOfDay(now);
    final tomorrow = today.add(const Duration(days: 1));
    final dayAfterTomorrow = today.add(const Duration(days: 2));

    // Completed tasks stay visible in their original bucket (crossed
    // out) rather than disappearing, so finishing something today
    // doesn't make it look like it vanished.
    final pending = allTasks;

    final overdueList = pending
        .where((t) => !t.completed && t.dueDate.isBefore(today))
        .toList();
    final todayList =
        pending.where((t) => isSameDay(t.dueDate, today)).toList();
    final tomorrowList =
        pending.where((t) => isSameDay(t.dueDate, tomorrow)).toList();
    final upcomingList = pending
        .where((t) =>
            t.dueDate.isAfter(dayAfterTomorrow) ||
            isSameDay(t.dueDate, dayAfterTomorrow))
        .toList();

    for (final list in [overdueList, todayList, tomorrowList, upcomingList]) {
      list.sort((a, b) => a.dueDate.compareTo(b.dueDate));
    }

    return TasksViewData(
      overdue: overdueList,
      today: todayList,
      tomorrow: tomorrowList,
      upcoming: upcomingList,
      subjectsById: subjectsById,
    );
  }
}

      recurrenceUntil: existing?.recurrenceUntil,
      seriesId: existing?.seriesId,

      });
      return;

        // notes...) on the first occurrence; the others start fresh.

/// A week's worth of sessions plus lookup maps, so the Timetable
/// screen can switch days without re-hitting the database each time.
class TimetableWeekData {
  final List<StudySession> sessions;
  final Map<String, Subject> subjectsById;
  final Map<String, Topic> topicsById;

  const TimetableWeekData({
    required this.sessions,
    required this.subjectsById,
    required this.topicsById,
  });

  List<StudySession> forDay(DateTime day) {
    final list =
        sessions.where((s) => isSameDay(s.scheduledDate, day)).toList();
    list.sort((a, b) => a.startTime.compareTo(b.startTime));
    return list;
  }

  static Future<TimetableWeekData> load(DateTime anyDayInWeek) async {
    final sessionRepo = SessionRepository();
    final subjectRepo = SubjectRepository();
    final topicRepo = TopicRepository();

    final start = startOfWeek(anyDayInWeek);
    final end = endOfWeek(anyDayInWeek);

    final sessions = await sessionRepo.getForRange(start, end);
    final subjects = await subjectRepo.getAll();
    final subjectsById = {for (final s in subjects) s.id: s};

    final topicsById = <String, Topic>{};
    for (final subject in subjects) {
      final topics = await topicRepo.getForSubject(subject.id);
      for (final topic in topics) {
        topicsById[topic.id] = topic;
      }
    }

    return TimetableWeekData(
      sessions: sessions,
      subjectsById: subjectsById,
      topicsById: topicsById,
    );
  }
}

/// A FutureBuilder wrapper that adds the three things almost every
/// screen in this app was missing before Step 16: a real error state
/// (instead of an endless spinner if something throws — the exact bug
/// StartupDecider had), an optional retry action, and a soft fade
/// between loading/error/content so screen transitions don't pop.
///
/// This does not change behavior for the happy path — it's a drop-in
/// replacement for `FutureBuilder(future: ..., builder: (context,
/// snapshot) { if (!snapshot.hasData) return CircularProgressIndicator();
/// ... })`.
class AsyncBuilder<T> extends StatelessWidget {
  final Future<T> future;
  final Widget Function(BuildContext context, T data) builder;
  final String errorMessage;
  final VoidCallback? onRetry;

  const AsyncBuilder({
    super.key,
    required this.future,
    required this.builder,
    this.errorMessage = "Something went wrong loading this screen.",
    this.onRetry,
  });

  @override
  Widget build(BuildContext context) {
    return FutureBuilder<T>(
      future: future,
      builder: (context, snapshot) {
        late final Widget child;
        late final String stateKey;

        if (snapshot.hasError) {
          child = _ErrorView(
              message: errorMessage, onRetry: onRetry, error: snapshot.error);
          stateKey = 'error';
        } else if (!snapshot.hasData) {
          child = const Center(child: CircularProgressIndicator());
          stateKey = 'loading';
        } else {
          child = builder(context, snapshot.data as T);
          // Was: 'data-${identityHashCode(snapshot.data)}'
          // That changed on every reload (new object every _load() call),
          // forcing AnimatedSwitcher to tear down and rebuild the whole
          // subtree — including IconButton tooltips/overlays — on every
          // single reload. That teardown-mid-rebuild is what triggered
          // the `_dependents.isEmpty` crash. A stable key here means we
          // only fade between loading/error/data states, not on every
          // refresh within the "data" state.
          stateKey = 'data';
        }

        return AnimatedSwitcher(
          duration: const Duration(milliseconds: 220),
          child: KeyedSubtree(key: ValueKey(stateKey), child: child),
        );
      },
    );
  }
}

class _ErrorView extends StatelessWidget {
  final String message;
  final Object? error;
  final VoidCallback? onRetry;

  const _ErrorView({required this.message, this.error, this.onRetry});

  @override
  Widget build(BuildContext context) {
    return Center(
      child: Padding(
        padding: const EdgeInsets.all(AppSpacing.xl),
        child: Column(
          mainAxisSize: MainAxisSize.min,
          children: [
            Icon(Icons.error_outline_rounded,
                size: 32, color: context.secondaryText),
            const SizedBox(height: AppSpacing.sm),
            Text(
              message,
              textAlign: TextAlign.center,
              style: TextStyle(color: context.secondaryText),
            ),
            if (onRetry != null) ...[
              const SizedBox(height: AppSpacing.md),

            ],
          ],
        ),
      ),
    );
  }

import 'dart:math' as math;
import 'package:flutter/material.dart';

class DailyGoalDial extends StatelessWidget {
  final int minutes;
  final ValueChanged<int> onChanged;
  final double size;

  static const int minMinutes = 15;
  static const int maxMinutes = 480;
  static const int step = 15;

  const DailyGoalDial({
    super.key,
    required this.minutes,
    required this.onChanged,
    this.size = 260,
  });

  void _update(Offset local) {
    final c = Offset(size / 2, size / 2);
    final d = local - c;
    var angle = math.atan2(d.dx, -d.dy); // 0 at top, clockwise
    if (angle < 0) angle += 2 * math.pi;
    final frac = angle / (2 * math.pi);
    final raw = minMinutes + frac * (maxMinutes - minMinutes);
    var snapped = ((raw / step).round() * step).clamp(minMinutes, maxMinutes);

    // Stop the dial jumping across the top: stick to the end you came from.
    if (minutes > maxMinutes * 0.75 && snapped < maxMinutes * 0.25) {
      snapped = maxMinutes;
    } else if (minutes < maxMinutes * 0.25 && snapped > maxMinutes * 0.75) {
      snapped = minMinutes;
    }
    onChanged(snapped);
  }

  static String format(int m) {
    final h = m ~/ 60;
    final r = m % 60;
    if (h == 0) return '${r}m';
    if (r == 0) return '${h}h';
    return '${h}h ${r}m';
  }

  @override
  Widget build(BuildContext context) {
    final isDark = Theme.of(context).brightness == Brightness.dark;
    final textTheme = Theme.of(context).textTheme;
    final secondary =
        isDark ? AppColors.textSecondaryDark : AppColors.textSecondary;
    final track =
        isDark ? Colors.white.withValues(alpha: 0.10) : AppColors.primarySoft;
    final frac = (minutes - minMinutes) / (maxMinutes - minMinutes);

    return GestureDetector(
      onPanStart: (d) => _update(d.localPosition),
      onPanUpdate: (d) => _update(d.localPosition),

class _DialPainter extends CustomPainter {
  final double frac;
  final Color track;
  _DialPainter(this.frac, this.track);

  @override
  void paint(Canvas canvas, Size size) {
    const stroke = 22.0;
    final center = Offset(size.width / 2, size.height / 2);
    final radius = (size.width - stroke) / 2;
    final rect = Rect.fromCircle(center: center, radius: radius);

    canvas.drawCircle(
      center,
      radius,
      Paint()
        ..style = PaintingStyle.stroke
        ..strokeWidth = stroke
        ..color = track,
    );

    final sweep = 2 * math.pi * frac;
    final arc = Paint()
      ..style = PaintingStyle.stroke
      ..strokeCap = StrokeCap.round
      ..strokeWidth = stroke
      ..shader = const SweepGradient(
        startAngle: 0,
        endAngle: 2 * math.pi,
        colors: [
          AppColors.teal,
          AppColors.primary,
          AppColors.teal,
        ],
        transform: GradientRotation(-math.pi / 2),
      ).createShader(rect);
    canvas.drawArc(rect, -math.pi / 2, sweep, false, arc);

    final a = -math.pi / 2 + sweep;
    final thumb = Offset(
      center.dx + radius * math.cos(a),
      center.dy + radius * math.sin(a),
    );
    canvas.drawCircle(
      thumb,
      stroke * 0.85,
      Paint()..color = AppColors.primary.withValues(alpha: 0.25),
    );
    canvas.drawCircle(thumb, stroke * 0.6, Paint()..color = Colors.white);
    canvas.drawCircle(
      thumb,
      stroke * 0.32,
      Paint()..color = AppColors.primary,
    );
  }

  @override
  bool shouldRepaint(covariant _DialPainter old) =>
      old.frac != frac || old.track != track;
}

/// V2 Step 8b: circular progress ring drawn with a CustomPainter.
///
/// [fraction] is how much of the ring is filled, from 0.0 (empty) to
/// 1.0 (full circle). The arc starts at the top and runs clockwise.
/// [child] is shown in the middle (the countdown text).
class FocusRing extends StatelessWidget {
  final double size;
  final double fraction;
  final Color color;
  final Color trackColor;
  final double strokeWidth;
  final Widget? child;

  const FocusRing({
    super.key,
    required this.size,
    required this.fraction,
    required this.color,
    required this.trackColor,
    this.strokeWidth = 14,
    this.child,
  });

  @override
  Widget build(BuildContext context) {
    return SizedBox(
      width: size,
      height: size,
      child: CustomPaint(
        painter: _FocusRingPainter(
          fraction: fraction,
          color: color,
          trackColor: trackColor,
          strokeWidth: strokeWidth,
        ),
        child: Center(child: child),
      ),
    );
  }
}

class _FocusRingPainter extends CustomPainter {
  final double fraction;
  final Color color;
  final Color trackColor;
  final double strokeWidth;

  const _FocusRingPainter({
    required this.fraction,
    required this.color,
    required this.trackColor,
    required this.strokeWidth,
  });

  @override
  void paint(Canvas canvas, Size size) {
    final center = size.center(Offset.zero);
    final radius = (math.min(size.width, size.height) - strokeWidth) / 2;

    // Background track (the full circle).
    final trackPaint = Paint()
      ..style = PaintingStyle.stroke
      ..strokeWidth = strokeWidth
      ..color = trackColor;
    canvas.drawCircle(center, radius, trackPaint);

    final safeFraction = fraction.clamp(0.0, 1.0);
    if (safeFraction <= 0) return;

    // Progress arc, starting at 12 o'clock.
    final arcPaint = Paint()
      ..style = PaintingStyle.stroke
      ..strokeWidth = strokeWidth
      ..strokeCap = StrokeCap.round
      ..color = color;

    canvas.drawArc(
      Rect.fromCircle(center: center, radius: radius),
      -math.pi / 2,
      2 * math.pi * safeFraction,
      false,
      arcPaint,
    );
  }

  @override
  bool shouldRepaint(covariant _FocusRingPainter old) {
    return old.fraction != fraction ||
        old.color != color ||
        old.trackColor != trackColor ||
        old.strokeWidth != strokeWidth;
  }
}

import 'package:fl_chart/fl_chart.dart';
import 'package:flutter/material.dart';

import '../theme/app_theme.dart';
import '../theme/theme_helpers.dart';
import '../utils/date_utils.dart';
import 'simple_bar_chart.dart';

/// V2 Step 10a: fl_chart bar chart for study time (values are minutes).
///
/// Takes the same [BarChartPoint] list as the old SimpleBarChart, so the
/// existing ProgressViewData does not need to change. Shows grid lines,
/// time labels on the left, rounded bars and a tap tooltip.
/// [highlightIndex] draws one bar darker (for example today).
class StudyBarChart extends StatelessWidget {
  final List<BarChartPoint> points;
  final double height;
  final int? highlightIndex;

  const StudyBarChart({
    super.key,
    required this.points,
    this.height = 200,
    this.highlightIndex,
  });

  /// Grid step candidates in minutes. The smallest one that keeps the
  /// chart to about 5 grid lines is used.
  static const _steps = [30, 60, 120, 180, 300, 600, 1200, 3000, 6000];

  static int _stepFor(int maxMinutes) {
    for (final step in _steps) {
      if (maxMinutes <= step * 5) return step;
    }
    return _steps.last;
  }

  @override
  Widget build(BuildContext context) {
    final maxMinutes = points.fold<int>(
      0,
      (m, p) => p.value > m ? p.value : m,
    );
    final step = _stepFor(maxMinutes);
    // Always leave some headroom above the tallest bar.
    final maxY = (((maxMinutes ~/ step) + 1) * step).toDouble();

    final gridColor = context.secondaryText.withValues(alpha: 0.18);
    final labelStyle = Theme.of(context).textTheme.labelSmall?.copyWith(
          color: context.secondaryText,
        );
    final barWidth = points.length > 8 ? 10.0 : 18.0;

    final groups = List.generate(points.length, (i) {
      final isHighlight = highlightIndex == i;
      return BarChartGroupData(
        x: i,
        barRods: [
          BarChartRodData(
            toY: points[i].value.toDouble(),
            width: barWidth,
            color: isHighlight ? AppColors.primaryDark : AppColors.primary,
            borderRadius: const BorderRadius.vertical(
              top: Radius.circular(5),
            ),
          ),
        ],
      );
    });

    return Semantics(
      label: 'Study time bar chart',
      child: SizedBox(
        height: height,
        child: BarChart(
          BarChartData(
            maxY: maxY,
            minY: 0,
            alignment: BarChartAlignment.spaceAround,
            barGroups: groups,
            borderData: FlBorderData(show: false),
            gridData: FlGridData(
              show: true,
              drawVerticalLine: false,
              horizontalInterval: step.toDouble(),
              getDrawingHorizontalLine: (value) =>
                  FlLine(color: gridColor, strokeWidth: 1),
            ),
            titlesData: FlTitlesData(
              topTitles: const AxisTitles(
                sideTitles: SideTitles(showTitles: false),
              ),
              rightTitles: const AxisTitles(
                sideTitles: SideTitles(showTitles: false),
              ),
              bottomTitles: AxisTitles(
                sideTitles: SideTitles(
                  showTitles: true,
                  reservedSize: 28,
                  getTitlesWidget: (value, meta) {
                    final i = value.toInt();
                    if (i < 0 || i >= points.length) {
                      return const SizedBox.shrink();
                    }
                    return Padding(
                      padding: const EdgeInsets.only(top: 8),
                      child: Text(points[i].label, style: labelStyle),
                    );
                  },
                ),
              ),
              leftTitles: AxisTitles(
                sideTitles: SideTitles(
                  showTitles: true,
                  reservedSize: 40,
                  interval: step.toDouble(),
                  getTitlesWidget: (value, meta) {
                    final text =
                        value == 0 ? '0' : formatDuration(value.round());
                    return Padding(
                      padding: const EdgeInsets.only(right: 6),
                      child: Text(
                        text,
                        textAlign: TextAlign.right,
                        style: labelStyle,
                      ),
                    );
                  },
                ),
              ),
            ),
            barTouchData: BarTouchData(
              enabled: true,
              touchTooltipData: BarTouchTooltipData(
                getTooltipColor: (_) => AppColors.primaryDark,
                getTooltipItem: (group, groupIndex, rod, rodIndex) {
                  return BarTooltipItem(
                    formatDuration(rod.toY.round()),
                    const TextStyle(
                      color: Colors.white,
                      fontWeight: FontWeight.w600,
                      fontSize: 12,
                    ),
                  );
                },
              ),
            ),
          ),
        ),
      ),
    );
  }
}
