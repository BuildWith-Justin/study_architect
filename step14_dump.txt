void main() async {
  WidgetsFlutterBinding.ensureInitialized();

  // Plain sqflite only ships a native database engine for Android/iOS.
  // On Windows and Linux desktop there is no such engine bundled, so
  // sqflite_common_ffi's databaseFactory must be installed explicitly
  // before DatabaseService (or anything else) opens a database â€”
  // otherwise every DB call throws "databaseFactory not initialized."
  if (!kIsWeb && (Platform.isWindows || Platform.isLinux)) {
    sqfliteFfiInit();
    databaseFactory = databaseFactoryFfi;
  }

  // V2: load Supabase credentials from .env and start the client.
  // Wrapped in try/catch so a missing .env or a Supabase problem never
  // stops the local-first app from launching.
  try {
    await dotenv.load(fileName: '.env');
    await Supabase.initialize(
      url: dotenv.env['SUPABASE_URL']!,
      publishableKey: dotenv.env['SUPABASE_ANON_KEY']!,
    );
    // Only flips to true when init fully succeeded. AuthGate reads this
    // so it never touches Supabase.instance when the client isn't ready.
    AuthService.isAvailable = true;
    debugPrint('Supabase initialized OK');

    // V2 Step 4: start local <-> cloud sync (does nothing until signed in).
    SyncService.instance.start();
  } catch (e) {
    AuthService.isAvailable = false;
    debugPrint('Supabase init failed: $e');
  }

  // Safe no-op on Web â€” see notification_service.dart for why.
  await NotificationService.instance.init();

  // V2 Step 13: gives XP to existing sessions and tasks the first time,
  // then keeps the ledger in line with the data. Runs in the background.
  GamificationHooks.startupRebuild();
  runApp(const StudyArchitectApp());
}

/// An exam the student is preparing for (V2 Step 11).
class Exam {
  final String id;
  final String subjectId;
  final String title;

  /// Date and time of the exam, in the device's local time.
  final DateTime examDate;
  final String location;
  final String notes;
  final DateTime createdAt;

  const Exam({
    required this.id,
    required this.subjectId,
    required this.title,
    required this.examDate,
    this.location = '',
    this.notes = '',
    required this.createdAt,
  });

  Exam copyWith({
    String? id,
    String? subjectId,
    String? title,
    DateTime? examDate,
    String? location,
    String? notes,
    DateTime? createdAt,
  }) {
    return Exam(
      id: id ?? this.id,
      subjectId: subjectId ?? this.subjectId,
      title: title ?? this.title,
      examDate: examDate ?? this.examDate,
      location: location ?? this.location,
      notes: notes ?? this.notes,
      createdAt: createdAt ?? this.createdAt,
    );
  }

  /// Whole calendar days from today to the exam day (0 = today,
  /// negative = already passed). The time of day is ignored.
  int daysLeft([DateTime? now]) {
    final today = now ?? DateTime.now();
    final todayDay = DateTime.utc(today.year, today.month, today.day);
    final examDay = DateTime.utc(examDate.year, examDate.month, examDate.day);
    return examDay.difference(todayDay).inDays;
  }

  /// True once the exam day is behind us.
  bool isPast([DateTime? now]) => daysLeft(now) < 0;

  Map<String, dynamic> toMap() => {
        'id': id,
        'subjectId': subjectId,
        'title': title,
        'examDate': examDate.toLocal().toIso8601String(),
        'location': location,
        'notes': notes,
        'createdAt': createdAt.toIso8601String(),
      };

  factory Exam.fromMap(Map<String, dynamic> map) => Exam(
        id: map['id'] as String,
        subjectId: map['subjectId'] as String,
        title: map['title'] as String,
        examDate: DateTime.parse(map['examDate'] as String),
        location: (map['location'] as String?) ?? '',
        notes: (map['notes'] as String?) ?? '',
        createdAt: DateTime.parse(map['createdAt'] as String),
      );
}

/// How far along the preparation checklist of one exam is.
class ExamProgress {
  final int prepared;
  final int total;

  const ExamProgress({required this.prepared, required this.total});

  const ExamProgress.empty()
      : prepared = 0,
        total = 0;

  /// 0.0 to 1.0 (0 when the exam has no topics yet).
  double get fraction => total == 0 ? 0 : prepared / total;

  int get percent => (fraction * 100).round();
}

/// What produced an XP award. The enum name is what gets stored.
enum XpSource { session, task, examTopic, bonus }

/// One line of the XP ledger. Total XP, level and "XP today" are always
/// computed from these rows, never stored separately.
class XpEntry {
  final String id;
  final XpSource sourceType;
  final String sourceId;
  final int amount;
  final String reason;
  final DateTime earnedAt;

  const XpEntry({
    required this.id,
    required this.sourceType,
    required this.sourceId,
    required this.amount,
    this.reason = '',
    required this.earnedAt,
  });

  /// Same source -> same id. This is what stops a session or a task from
  /// paying twice, even after a cloud sync or a backup restore.
  static String idFor(XpSource type, String sourceId) =>
      '${type.name}:$sourceId';

  factory XpEntry.forSource({
    required XpSource type,
    required String sourceId,
    required int amount,
    String reason = '',
    required DateTime earnedAt,
  }) {
    return XpEntry(
      id: idFor(type, sourceId),
      sourceType: type,
      sourceId: sourceId,
      amount: amount,
      reason: reason,
      earnedAt: earnedAt,
    );
  }

  Map<String, dynamic> toMap() => {
        'id': id,
        'sourceType': sourceType.name,
        'sourceId': sourceId,
        'amount': amount,
        'reason': reason,
        'earnedAt': earnedAt.toIso8601String(),
      };

  factory XpEntry.fromMap(Map<String, dynamic> map) => XpEntry(
        id: map['id'] as String,
        sourceType: XpSource.values.firstWhere(
          (s) => s.name == map['sourceType'],

import '../models/achievement.dart';
import '../services/database_service.dart';

/// Reads and writes unlocked badges. A badge is never removed once
/// unlocked (only "Clear All Data" wipes the table).
class AchievementRepository {
  final _db = DatabaseService.instance;

  /// Unlocks a badge. Returns true only the first time, so the caller
  /// knows when to show a celebration.
  Future<bool> unlock(String id, {DateTime? at}) async {
    final db = await _db.database;
    return db.transaction((txn) async {
      final existing = await txn.query(
        'achievements',
        columns: ['id'],
        where: 'id = ?',
        whereArgs: [id],
        limit: 1,
      );
      if (existing.isNotEmpty) return false;
      final badge = Achievement(id: id, unlockedAt: at ?? DateTime.now());
      await txn.insert('achievements', badge.toMap());
      return true;
    });
  }

  Future<bool> isUnlocked(String id) async {
    final db = await _db.database;
    final rows = await db.query(
      'achievements',
      columns: ['id'],
      where: 'id = ?',
      whereArgs: [id],
      limit: 1,
    );
    return rows.isNotEmpty;
  }

  Future<Set<String>> getUnlockedIds() async {
    final db = await _db.database;
    final rows = await db.query('achievements', columns: ['id']);
    return rows.map((r) => r['id'] as String).toSet();
  }

  /// Every unlocked badge, newest first.
  Future<List<Achievement>> getAll() async {
    final db = await _db.database;
    final rows = await db.query('achievements', orderBy: 'unlockedAt DESC');
    return rows.map(Achievement.fromMap).toList();
  }

  Future<List<Achievement>> getRecent({int limit = 3}) async {
    final db = await _db.database;
    final rows = await db.query(
      'achievements',
      orderBy: 'unlockedAt DESC',
      limit: limit,
    );
    return rows.map(Achievement.fromMap).toList();
  }
}

import 'package:sqflite/sqflite.dart';
import 'package:uuid/uuid.dart';

import '../models/exam.dart';
import '../models/exam_topic.dart';
import '../services/database_service.dart';

/// All reads and writes for exams and their topic checklist.
/// Like the other repositories, it does not notify DataRefreshBus â€”
/// the screens that change data call
/// DataRefreshBus.instance.notify(RefreshScope.exams) afterwards.
class ExamRepository {
  final _db = DatabaseService.instance;
  static const _uuid = Uuid();

  /// Creates the exam and, in the same transaction, its topic checklist.
  Future<void> create(Exam exam, {List<String> topicIds = const []}) async {
    final db = await _db.database;
    await db.transaction((txn) async {
      await txn.insert('exams', exam.toMap());
      for (final topicId in topicIds.toSet()) {
        await txn.insert(
          'exam_topics',
          ExamTopic(id: _uuid.v4(), examId: exam.id, topicId: topicId).toMap(),
        );
      }
    });
  }

  /// Updates the exam. When [topicIds] is given, the checklist is made to
  /// match it: removed topics are deleted, new ones are added, and topics
  /// that stay keep their "prepared" state.
  Future<void> update(Exam exam, {List<String>? topicIds}) async {
    final db = await _db.database;
    await db.transaction((txn) async {
      await txn.update(
        'exams',
        exam.toMap(),
        where: 'id = ?',
        whereArgs: [exam.id],
      );
      if (topicIds != null) {
        await _syncTopicLinks(txn, exam.id, topicIds);
      }
    });
  }

  Future<void> _syncTopicLinks(
    DatabaseExecutor txn,
    String examId,
    List<String> topicIds,
  ) async {
    final wanted = topicIds.toSet();
    final rows = await txn.query(
      'exam_topics',
      where: 'examId = ?',
      whereArgs: [examId],
    );
    final existing = <String, String>{
      for (final r in rows) r['topicId'] as String: r['id'] as String,
    };

      if (!wanted.contains(entry.key)) {
        await txn.delete(
          'exam_topics',
          where: 'id = ?',
          whereArgs: [entry.value],
        );
      }
    }
    for (final topicId in wanted) {
      if (!existing.containsKey(topicId)) {
        await txn.insert(
          'exam_topics',
          ExamTopic(id: _uuid.v4(), examId: examId, topicId: topicId).toMap(),
        );
      }
    }
  }

import 'package:sqflite/sqflite.dart';

/// Reads and writes the XP ledger. Like the other repositories it does
/// not notify the UI: the gamification engine does that after it awards.
class XpRepository {
  final _db = DatabaseService.instance;

  void _validate(XpEntry entry) {
    if (entry.amount <= 0) {
      throw ArgumentError('XP amount must be positive.');
    }
  }

  /// Adds an award. If this source already has one, nothing happens.
  /// Returns true only when a new row was written.
  Future<bool> award(XpEntry entry) async {
    _validate(entry);
    final db = await _db.database;
    return db.transaction((txn) async {
      final existing = await txn.query(
        'xp_log',
        columns: ['id'],
        where: 'id = ?',
        whereArgs: [entry.id],
        limit: 1,
      );
      if (existing.isNotEmpty) return false;
      await txn.insert('xp_log', entry.toMap());
      return true;
    });
  }

  /// Adds the award, or refreshes it when the same source is already
  /// there with a different amount or reason (for example a session
  /// edited after completion). Returns true if anything changed.
  Future<bool> upsert(XpEntry entry) async {
    _validate(entry);
    final db = await _db.database;
    return db.transaction((txn) async {
      final rows = await txn.query(
        'xp_log',
        where: 'id = ?',
        whereArgs: [entry.id],
        limit: 1,
      );
      if (rows.isEmpty) {
        await txn.insert('xp_log', entry.toMap());
        return true;
      }
      final current = XpEntry.fromMap(rows.first);
      final sameDay = current.earnedAt.year == entry.earnedAt.year &&
          current.earnedAt.month == entry.earnedAt.month &&
          current.earnedAt.day == entry.earnedAt.day;
      if (current.amount == entry.amount &&
          current.reason == entry.reason &&
          sameDay) {
        return false;
      }
      await txn.update(
        'xp_log',
        entry.toMap(),
        where: 'id = ?',
        whereArgs: [entry.id],
      );
      return true;
    });
  }

  /// Removes the award of one source (used when a session or task is
  /// deleted). Returns how many rows were removed.
  Future<int> removeBySource(XpSource type, String sourceId) async {
    final db = await _db.database;
    return db.delete(
      'xp_log',
      where: 'sourceType = ? AND sourceId = ?',
      whereArgs: [type.name, sourceId],
    );
  }

  Future<bool> hasEntry(String id) async {
    final db = await _db.database;
    final rows = await db.query(
      'xp_log',
      columns: ['id'],
      where: 'id = ?',
      whereArgs: [id],
      limit: 1,
    );
    return rows.isNotEmpty;
  }

  /// Source ids that already have an award of this type - lets the
  /// engine find which old sessions or tasks still need XP.
  Future<Set<String>> getSourceIds(XpSource type) async {
    final db = await _db.database;
    final rows = await db.query(
      'xp_log',
      columns: ['sourceId'],
      where: 'sourceType = ?',
      whereArgs: [type.name],
    );
    return rows.map((r) => r['sourceId'] as String).toSet();
  }

  Future<int> getTotalXp() async {
    final db = await _db.database;
    final result = await db.rawQuery(
      'SELECT COALESCE(SUM(amount), 0) FROM xp_log',
    );
    return Sqflite.firstIntValue(result) ?? 0;
  }

  /// XP earned on one calendar day (device local time).
  Future<int> getXpForDay(DateTime day) async {
    final db = await _db.database;
    final start = DateTime(day.year, day.month, day.day);
    final end = start.add(const Duration(days: 1));
    final result = await db.rawQuery(
      'SELECT COALESCE(SUM(amount), 0) FROM xp_log '
      'WHERE earnedAt >= ? AND earnedAt < ?',
      [start.toIso8601String(), end.toIso8601String()],
    );
    return Sqflite.firstIntValue(result) ?? 0;
  }

  /// XP earned on one calendar day by one kind of source. Pass
  /// [excludeSourceId] to leave one source out (used to apply the daily
  /// cap to a session without counting its own previous award).
  Future<int> getXpForDayBySource(
    DateTime day,
    XpSource type, {
    String? excludeSourceId,
  }) async {
    final db = await _db.database;
    final start = DateTime(day.year, day.month, day.day);
    final end = start.add(const Duration(days: 1));
    final args = <Object?>[
      type.name,
      start.toIso8601String(),
      end.toIso8601String(),
    ];
    var sql = 'SELECT COALESCE(SUM(amount), 0) FROM xp_log '
        'WHERE sourceType = ? AND earnedAt >= ? AND earnedAt < ?';
    if (excludeSourceId != null) {
      sql += ' AND sourceId != ?';
      args.add(excludeSourceId);
    }
    final result = await db.rawQuery(sql, args);
    return Sqflite.firstIntValue(result) ?? 0;
  }

  /// Every award, newest first.
  Future<List<XpEntry>> getAll() async {
    final db = await _db.database;
    final rows = await db.query('xp_log', orderBy: 'earnedAt DESC');
    return rows.map(XpEntry.fromMap).toList();
  }

        RefreshScope.achievements,
      ];

  @override
  void initState() {
    super.initState(); // RefreshableScreenMixin subscribes to the bus here.
    WidgetsBinding.instance.addObserver(this);
    _future = HomeViewData.load();
  }

  @override
  void didChangeDependencies() {
    super.didChangeDependencies();

    final route = ModalRoute.of(context);

    if (route != null && route != _route) {
      if (_route != null) {
        routeObserver.unsubscribe(this);
      }

      _route = route;
      routeObserver.subscribe(this, route);
    }
  }

  @override
  void dispose() {
    routeObserver.unsubscribe(this);
    WidgetsBinding.instance.removeObserver(this);
    super.dispose(); // RefreshableScreenMixin cancels its subscription here.
  }

  @override
  void didPopNext() {
    onRefresh();
  }

  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    if (state == AppLifecycleState.resumed) {

    }
  }

  /// Used by pull-to-refresh, the DataRefreshBus, returning from another
  /// screen and app resume.
  @override
  Future<void> onRefresh() async {
    final next = HomeViewData.load();

    if (!mounted) {
      next.ignore();
      return;
    }

    setState(() {
      _future = next;
    });

    try {
      await next;
    } catch (_) {
      // FutureBuilder handles the error state.
    }
  }

  String _greeting() {
    final hour = DateTime.now().hour;

    if (hour < 12) return 'Good morning';
    if (hour < 17) return 'Good afternoon';
    return 'Good evening';
  }

    await Navigator.of(context).push(
      MaterialPageRoute(builder: (_) => screen),
    );

    if (mounted) {
      await onRefresh();
    }
  }

  /// The session currently running, or the next upcoming one today.
  _SessionAndSubject? _currentOrNextSession(HomeViewData data) {
    final now = DateTime.now();

    final sorted = [...data.todaySessions]
      ..sort((a, b) => a.startTime.compareTo(b.startTime));

    for (final session in sorted) {
      if (session.endTime.isAfter(now)) {
        return _SessionAndSubject(
          session,
          data.subjectsById[session.subjectId],
        );
      }
    }

    return null;
  }

  Widget _buildClassCard(HomeViewData data) {
    final match = _currentOrNextSession(data)!;
    final session = match.session;
    final subject = match.subject;
    final now = DateTime.now();

    final isLive =
        session.startTime.isBefore(now) && session.endTime.isAfter(now);

    return ClassCard(
      timeRange:
          '${formatTimeOfDay(session.startTime)} - ${formatTimeOfDay(session.endTime)}',
      title: subject?.name ?? 'Unknown subject',
      hue: _hueForSubject(subject),
      actionLabel: isLive ? 'Start session' : 'View details',
      onAction: isLive
          ? () => _openScreen(
                FocusTimerScreen(
                  subjectId: session.subjectId,
                  subjectName: subject?.name ?? 'Unknown subject',
                  subjectColorValue:
                      subject?.colorValue ?? AppColors.accentPurpleStrong.value,
                  topicId: session.topicId,
                  existingSessionId: session.id,
                  goal: session.endTime.difference(session.startTime),
                ),
              )
          : () {},
    );
  }

  PastelHue _hueForSubject(Subject? subject) {
    if (subject == null) {
      return PastelHue.blue;
    }

    final color = subject.colorValue;

    if (color == AppColors.accentGreenStrong.value) {
      return PastelHue.green;
    }

    if (color == AppColors.accentBlueStrong.value) {
      return PastelHue.blue;
    }

    return PastelHue.purple;
  }
}

/// Current streak plus the 7 days of this week (Monday to Sunday).
/// A filled day has 10+ minutes studied; a snowflake is a missed day that
/// the weekly freeze forgave.
class _StreakStrip extends StatelessWidget {
  final int streak;
  final List<DateTime> weekDays;
  final DateTime today;
  final bool Function(DateTime day) studiedOn;
  final bool Function(DateTime day) bridgedOn;
  final bool bonusActive;
  final bool freezeUsed;

  const _StreakStrip({
    required this.streak,
    required this.weekDays,
    required this.today,
    required this.studiedOn,
    required this.bridgedOn,
    required this.bonusActive,
    required this.freezeUsed,
  });

  static const _labels = ['M', 'T', 'W', 'T', 'F', 'S', 'S'];

  @override
  Widget build(BuildContext context) {
    final theme = Theme.of(context);

    final title = streak == 0
        ? 'Start your streak today'
        : streak == 1
            ? '1 day streak'
            : '$streak day streak';

    return Container(
      padding: const EdgeInsets.all(AppSpacing.md),
      decoration: BoxDecoration(
        color: context.cardSurfaceAlt,
        borderRadius: BorderRadius.circular(AppRadii.card),
      ),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Row(
            children: [
              const Icon(
                Icons.local_fire_department_rounded,
                color: AppColors.primary,
                size: 22,
              ),
              const SizedBox(width: AppSpacing.sm),
              Expanded(
                child: Text(
                  title,
                  style: theme.textTheme.titleSmall?.copyWith(
                    fontWeight: FontWeight.w600,
                  ),
                ),
              ),
              if (bonusActive)
                Container(
                  padding: const EdgeInsets.symmetric(
                    horizontal: 8,
                    vertical: 2,
                  ),
                  decoration: BoxDecoration(
                    color: AppColors.primary.withValues(alpha: 0.15),
                    borderRadius: BorderRadius.circular(999),
                  ),
                  child: Text(
                    '+${XpRules.streakBonusPercent}% XP',
                    style: theme.textTheme.bodySmall?.copyWith(
                      color: AppColors.primary,
                      fontWeight: FontWeight.w600,
                    ),
                  ),
                ),
            ],
          ),
          const SizedBox(height: AppSpacing.md),
          Row(
            mainAxisAlignment: MainAxisAlignment.spaceBetween,
            children: List.generate(7, (i) => _dayCell(context, i)),
          ),
          if (streak > 0) ...[
            const SizedBox(height: AppSpacing.sm),
            Text(
              freezeUsed
                  ? 'Missed-day freeze used this week'
                  : 'One missed day per week is forgiven',
              style: theme.textTheme.bodySmall?.copyWith(
                color: context.secondaryText,
              ),
            ),
          ],
        ],
      ),
    );
  }

  Widget _dayCell(BuildContext context, int index) {
    final day = weekDays[index];
    final studied = studiedOn(day);
    final bridged = !studied && bridgedOn(day);
    final isToday = isSameDay(day, today);

    final neutralFill = context.isDarkMode
        ? Colors.white.withValues(alpha: 0.08)
        : AppColors.primary.withValues(alpha: 0.08);

    return Column(
      children: [
        Text(
          _labels[index],
          style: Theme.of(context).textTheme.bodySmall?.copyWith(
                color: context.secondaryText,
                fontWeight: isToday ? FontWeight.w700 : FontWeight.w400,
              ),
        ),
        const SizedBox(height: 6),
        Container(
          width: 32,
          height: 32,
          decoration: BoxDecoration(
            shape: BoxShape.circle,
            color: studied ? AppColors.primary : neutralFill,
            border: isToday && !studied
                ? Border.all(color: AppColors.primary, width: 2)
                : null,
          ),
          child: studied
              ? const Icon(Icons.check_rounded, size: 18, color: Colors.white)
              : bridged
                  ? const Icon(
                      Icons.ac_unit_rounded,
                      size: 16,
                      color: AppColors.primary,
                    )
                  : null,
        ),
      ],
    );
  }
}

/// Small date helpers shared by Home, Timetable, and Progress â€”
/// keeps "start of day" / "start of week" logic in one place.
DateTime startOfDay(DateTime date) => DateTime(date.year, date.month, date.day);

/// Monday-start week, matching the spec's "Week â€¢ September 8â€“14" style.
DateTime startOfWeek(DateTime date) {
  final day = startOfDay(date);
  return day.subtract(Duration(days: day.weekday - DateTime.monday));
}

DateTime endOfWeek(DateTime date) =>
    startOfWeek(date).add(const Duration(days: 7));

bool isSameDay(DateTime a, DateTime b) =>
    a.year == b.year && a.month == b.month && a.day == b.day;

String formatDuration(int minutes) {
  final h = minutes ~/ 60;
  final m = minutes % 60;
  if (h == 0) return '${m}m';
  if (m == 0) return '${h}h';
  return '${h}h ${m}m';
}

String formatTimeOfDay(DateTime time) {
  final hour = time.hour % 12 == 0 ? 12 : time.hour % 12;
  final minute = time.minute.toString().padLeft(2, '0');
  final period = time.hour >= 12 ? 'PM' : 'AM';
  return '$hour:$minute $period';
}

class StudyInsight {
  final IconData icon;
  final String text;

  const StudyInsight({required this.icon, required this.text});
}

/// Deterministic, rule-based insights (no ML) â€” the V1 spec calls for
/// "rule-based study insights" specifically. Each rule is independent
/// and only fires when its condition is clearly true, so the list can
/// safely be empty on a quiet day.
List<StudyInsight> generateInsights({
  required int todayMinutes,
  required int weeklyMinutes,
  required int weeklyGoalMinutes,
  required int streak,
  required int longestStreak,
  required Map<String, int> subjectPerformancePercent,
}) {
  final insights = <StudyInsight>[];
  final now = DateTime.now();

  // Hasn't studied yet today, and the day is more than half over.
  if (todayMinutes == 0 && now.hour >= 15) {
    insights.add(const StudyInsight(
      icon: Icons.schedule_rounded,
      text:
          "You haven't studied yet today â€” even 20 minutes keeps your streak alive.",
    ));
  }

  // Behind pace on the weekly goal once the week is more than half done.
  if (weeklyGoalMinutes > 0 && now.weekday >= DateTime.thursday) {
    final progress = weeklyMinutes / weeklyGoalMinutes;
    if (progress < 0.5) {
      insights.add(StudyInsight(
        icon: Icons.trending_down_rounded,
        text:
            "You're at ${(progress * 100).round()}% of this week's goal with the week more than half over.",
      ));
    }
  }

  // Subjects clearly falling behind their own weekly target â€” collapsed
  // into a single line instead of one card per subject. Previously
  // this looped and added one insight *per* behind-target subject,
  // which meant a normal early-week state (nothing logged yet, so
  // every subject reads 0%) produced a card for every single subject
  // â€” pure noise, not an insight.
  final behindSubjects = subjectPerformancePercent.entries
      .where((e) => e.value < 40)
      .toList()
    ..sort((a, b) => a.value.compareTo(b.value));

  if (behindSubjects.isNotEmpty) {
    if (weeklyMinutes == 0 &&
        behindSubjects.length == subjectPerformancePercent.length) {
      // Nothing logged at all yet this period â€” a single general
      // nudge is more useful than naming every subject.
      insights.add(const StudyInsight(
        icon: Icons.flag_outlined,
        text: "You haven't logged any study time yet this period.",
      ));
    } else {
      final shown = behindSubjects.take(2).map((e) => e.key).join(', ');
      final remaining = behindSubjects.length - 2;
      final verb = behindSubjects.length == 1 ? 'is' : 'are';
      final suffix = remaining > 0 ? ', and $remaining more,' : '';
      insights.add(StudyInsight(
        icon: Icons.flag_outlined,
        text: '$shown$suffix $verb behind on weekly targets.',
      ));
    }
  }

  // Reward consistency.
  if (streak >= 7) {
    insights.add(StudyInsight(
      icon: Icons.local_fire_department_rounded,
      text:
          "You're on a $streak-day streak â€” your longest is $longestStreak days.",
    ));
  } else if (streak == 0 && longestStreak > 0) {
    insights.add(StudyInsight(
      icon: Icons.replay_rounded,
      text:
          'Your streak reset. Your best was $longestStreak days â€” start a new one today.',
    ));
  }

  return insights;
}

/// Sum of actualDuration (minutes) across completed sessions whose
/// scheduledDate falls on [date].
int completedMinutesForDate(List<StudySession> sessions, DateTime date) {
  return sessions
      .where((s) => s.complete && isSameDay(s.scheduledDate, date))
      .fold(0, (sum, s) => sum + (s.actualDuration ?? 0));
}

/// Sum of actualDuration across completed sessions within [sessions],
/// with no date filtering â€” pass in a week's worth of sessions to get
/// a weekly total.
int completedMinutesTotal(List<StudySession> sessions) {
  return sessions
      .where((s) => s.complete)
      .fold(0, (sum, s) => sum + (s.actualDuration ?? 0));
}

/// Current study streak, per the V1 business rule: a day counts only
/// if it has at least one completed session.
///
/// [completedDates] are yyyy-MM-dd strings from
/// SessionRepository.getCompletedDates(). Today not yet having a
/// completed session does not break the streak â€” it simply isn't
/// counted yet, since the day isn't over.
int currentStreak(Set<String> completedDates, {DateTime? today}) {
  final now = today ?? DateTime.now();
  var cursor = startOfDay(now);
  var streak = 0;

  bool hasSessionOn(DateTime day) => completedDates.contains(_dateKey(day));

  // If today already has a completed session, count it and step back
  // from yesterday. If not, today is skipped (not yet failed) and we
  // start checking from yesterday.
  if (hasSessionOn(cursor)) {
    streak++;
  }
  cursor = cursor.subtract(const Duration(days: 1));

  while (hasSessionOn(cursor)) {
    streak++;
    cursor = cursor.subtract(const Duration(days: 1));
  }

  return streak;
}

String _dateKey(DateTime date) {
  final y = date.year.toString().padLeft(4, '0');
  final m = date.month.toString().padLeft(2, '0');
  final d = date.day.toString().padLeft(2, '0');
  return '$y-$m-$d';
}

/// Longest run of consecutive completed-session days on record, e.g.
/// for the "Longest: 18 days" line on the Progress screen.
int longestStreak(Set<String> completedDates) {
  if (completedDates.isEmpty) return 0;

  final sortedDates = completedDates.map((key) {
    final parts = key.split('-');
    return DateTime(
        int.parse(parts[0]), int.parse(parts[1]), int.parse(parts[2]));
  }).toList()
    ..sort();

  var longest = 1;
  var current = 1;
  for (var i = 1; i < sortedDates.length; i++) {
    final gap = sortedDates[i].difference(sortedDates[i - 1]).inDays;
    if (gap == 1) {
      current++;
      longest = current > longest ? current : longest;
    } else if (gap > 1) {
      current = 1;
    }
  }
  return longest;
}
