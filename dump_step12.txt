import 'package:flutter_dotenv/flutter_dotenv.dart';
import 'package:sqflite_common_ffi/sqflite_ffi.dart';
import 'package:supabase_flutter/supabase_flutter.dart';
import 'app/app.dart';
import 'features/auth/auth_service.dart';
import 'services/notification_service.dart';
import 'services/sync_service.dart';

void main() async {
  WidgetsFlutterBinding.ensureInitialized();

  // Plain sqflite only ships a native database engine for Android/iOS.
  // On Windows and Linux desktop there is no such engine bundled, so
  // sqflite_common_ffi's databaseFactory must be installed explicitly
  // before DatabaseService (or anything else) opens a database —
  // otherwise every DB call throws "databaseFactory not initialized."
  if (!kIsWeb && (Platform.isWindows || Platform.isLinux)) {
    sqfliteFfiInit();
    databaseFactory = databaseFactoryFfi;
  }

  // V2: load Supabase credentials from .env and start the client.
  // Wrapped in try/catch so a missing .env or a Supabase problem never
  // stops the local-first app from launching.
  try {
    await dotenv.load(fileName: '.env');
    await Supabase.initialize(
      url: dotenv.env['SUPABASE_URL']!,
      publishableKey: dotenv.env['SUPABASE_ANON_KEY']!,
    );
    // Only flips to true when init fully succeeded. AuthGate reads this
    // so it never touches Supabase.instance when the client isn't ready.
    AuthService.isAvailable = true;
    debugPrint('Supabase initialized OK');

    // V2 Step 4: start local <-> cloud sync (does nothing until signed in).
    SyncService.instance.start();
  } catch (e) {
    AuthService.isAvailable = false;
    debugPrint('Supabase init failed: $e');
  }

  // Safe no-op on Web — see notification_service.dart for why.
  await NotificationService.instance.init();
  runApp(const StudyArchitectApp());
}

import 'dart:async';
import 'package:flutter/material.dart';
import 'data_refresh_bus.dart';

/// Mixin any StatefulWidget's State with this to get:
/// - auto-refresh when relevant data changes elsewhere in the app
/// - pull-to-refresh wired to the same reload function
mixin RefreshableScreenMixin<T extends StatefulWidget> on State<T> {
  StreamSubscription<RefreshScope>? _refreshSub;

  /// Override: which scopes should trigger a refresh on this screen
  List<RefreshScope> get watchedScopes => [RefreshScope.all];

  /// Override: how this screen reloads its data
  Future<void> onRefresh();

  @override
  void initState() {
    super.initState();
    _refreshSub = DataRefreshBus.instance.stream.listen((scope) {
      if (watchedScopes.contains(scope) || scope == RefreshScope.all) {
        onRefresh();
      }
    });
  }

  @override
  void dispose() {
    _refreshSub?.cancel();
    super.dispose();
  }

  Widget buildRefreshable({required Widget child}) {
    return RefreshIndicator(
      onRefresh: onRefresh,
      child: child,
    );
  }
}

import 'dart:async';

enum RefreshScope {
  subjects,
  sessions,
  tasks,
  exams,
  achievements,
  all,
}

class DataRefreshBus {
  DataRefreshBus._internal();
  static final DataRefreshBus instance = DataRefreshBus._internal();

  final StreamController<RefreshScope> _controller =
      StreamController<RefreshScope>.broadcast();

  Stream<RefreshScope> get stream => _controller.stream;

  void notify(RefreshScope scope) {
    _controller.add(scope);
  }

  void dispose() {
    _controller.close();
  }
}

class SessionRepository {
  final _db = DatabaseService.instance;

  /// Throws if the session's end time is not after its start time -
  /// enforces the V1 business rule at the persistence boundary.
  Future<void> create(StudySession session) async {
    if (!session.isTimeRangeValid) {
      throw ArgumentError('Session end time must be after start time.');
    }
    final db = await _db.database;
    await db.insert('study_sessions', session.toMap());
  }

  /// V2 Step 7: saves a whole weekly series in one transaction, so a
  /// failure can never leave half a series behind.
  Future<void> createMany(List<StudySession> sessions) async {
    for (final s in sessions) {
      if (!s.isTimeRangeValid) {
        throw ArgumentError('Session end time must be after start time.');
      }
    }
    final db = await _db.database;
    await db.transaction((txn) async {
      for (final s in sessions) {
        await txn.insert('study_sessions', s.toMap());
      }
    });
  }

  Future<void> update(StudySession session) async {
    if (!session.isTimeRangeValid) {
      throw ArgumentError('Session end time must be after start time.');
    }
    final db = await _db.database;
    await db.update(
      'study_sessions',
      session.toMap(),
      where: 'id = ?',
      whereArgs: [session.id],
    );
  }

  Future<void> delete(String id) async {
    final db = await _db.database;
    await db.delete('study_sessions', where: 'id = ?', whereArgs: [id]);
  }

  /// V2 Step 7: deletes the occurrences of one series. With [fromDate],
  /// only occurrences on or after that day are removed. Returns the
  /// deleted sessions so the caller can cancel their reminders.
  Future<List<StudySession>> deleteSeries(
    String seriesId, {
    DateTime? fromDate,
  }) async {
    final db = await _db.database;
    final String where;
    final List<Object?> args;
    if (fromDate == null) {
      where = 'seriesId = ?';
      args = [seriesId];
    } else {
      final day = DateTime(fromDate.year, fromDate.month, fromDate.day);
      where = 'seriesId = ? AND scheduledDate >= ?';
      args = [seriesId, day.toIso8601String()];
    }
    final rows = await db.query(
      'study_sessions',
      where: where,
      whereArgs: args,
    );
    await db.delete('study_sessions', where: where, whereArgs: args);
    return rows.map(StudySession.fromMap).toList();
  }

  /// Every occurrence of one series, oldest first.
  Future<List<StudySession>> getBySeries(String seriesId) async {
    final db = await _db.database;
    final rows = await db.query(
      'study_sessions',
      where: 'seriesId = ?',
      whereArgs: [seriesId],
      orderBy: 'scheduledDate ASC',
    );
    return rows.map(StudySession.fromMap).toList();
  }

  Future<StudySession?> getById(String id) async {
    final db = await _db.database;
    final rows =
        await db.query('study_sessions', where: 'id = ?', whereArgs: [id]);
    if (rows.isEmpty) return null;
    return StudySession.fromMap(rows.first);
  }

  /// Sessions scheduled on one calendar day - powers Home's "today's
  /// schedule" and the streak calculation.
  Future<List<StudySession>> getForDate(DateTime date) async {
    final db = await _db.database;
    final startOfDay = DateTime(date.year, date.month, date.day);
    final endOfDay = startOfDay.add(const Duration(days: 1));
    final rows = await db.query(
      'study_sessions',
      where: 'scheduledDate >= ? AND scheduledDate < ?',
      whereArgs: [startOfDay.toIso8601String(), endOfDay.toIso8601String()],
      orderBy: 'startTime ASC',
    );
    return rows.map(StudySession.fromMap).toList();
  }

  /// Sessions within an inclusive date range - powers the Timetable
  /// week view and the Progress screen.
  Future<List<StudySession>> getForRange(DateTime start, DateTime end) async {
    final db = await _db.database;
    final rows = await db.query(
      'study_sessions',
      where: 'scheduledDate >= ? AND scheduledDate <= ?',
      whereArgs: [start.toIso8601String(), end.toIso8601String()],
      orderBy: 'startTime ASC',
    );
    return rows.map(StudySession.fromMap).toList();
  }

  Future<List<StudySession>> getForSubject(String subjectId) async {
    final db = await _db.database;
    final rows = await db.query(
      'study_sessions',
      where: 'subjectId = ?',
      whereArgs: [subjectId],
      orderBy: 'scheduledDate DESC',
    );
    return rows.map(StudySession.fromMap).toList();
  }

  /// Distinct calendar dates (as yyyy-MM-dd strings) that have at
  /// least one completed session - the raw material for the streak
  /// calculation in utils/study_stats.dart.
  Future<Set<String>> getCompletedDates() async {
    final db = await _db.database;
    final rows = await db.rawQuery('''
      SELECT DISTINCT substr(scheduledDate, 1, 10) as day
      FROM study_sessions
      WHERE complete = 1
    ''');
    return rows.map((r) => r['day'] as String).toSet();
  }

  /// Every session ever recorded - used by the Progress screen's
  /// "All time" view.
  Future<List<StudySession>> getAll() async {
    final db = await _db.database;
    final rows = await db.query('study_sessions', orderBy: 'scheduledDate ASC');
    return rows.map(StudySession.fromMap).toList();
  }
}

import 'package:sqflite/sqflite.dart';
import 'package:uuid/uuid.dart';

import '../models/exam.dart';
import '../models/exam_topic.dart';
import '../services/database_service.dart';

/// All reads and writes for exams and their topic checklist.
/// Like the other repositories, it does not notify DataRefreshBus —
/// the screens that change data call
/// DataRefreshBus.instance.notify(RefreshScope.exams) afterwards.
class ExamRepository {
  final _db = DatabaseService.instance;
  static const _uuid = Uuid();

  /// Creates the exam and, in the same transaction, its topic checklist.
  Future<void> create(Exam exam, {List<String> topicIds = const []}) async {
    final db = await _db.database;
    await db.transaction((txn) async {
      await txn.insert('exams', exam.toMap());
      for (final topicId in topicIds.toSet()) {
        await txn.insert(
          'exam_topics',
          ExamTopic(id: _uuid.v4(), examId: exam.id, topicId: topicId).toMap(),
        );
      }
    });
  }

  /// Updates the exam. When [topicIds] is given, the checklist is made to
  /// match it: removed topics are deleted, new ones are added, and topics
  /// that stay keep their "prepared" state.
  Future<void> update(Exam exam, {List<String>? topicIds}) async {
    final db = await _db.database;
    await db.transaction((txn) async {
      await txn.update(
        'exams',
        exam.toMap(),
        where: 'id = ?',
        whereArgs: [exam.id],
      );
      if (topicIds != null) {
        await _syncTopicLinks(txn, exam.id, topicIds);
      }
    });
  }

  Future<void> _syncTopicLinks(
    DatabaseExecutor txn,
    String examId,
    List<String> topicIds,
  ) async {
    final wanted = topicIds.toSet();
    final rows = await txn.query(
      'exam_topics',
      where: 'examId = ?',
      whereArgs: [examId],
    );
    final existing = <String, String>{
      for (final r in rows) r['topicId'] as String: r['id'] as String,
    };

    for (final entry in existing.entries) {
      if (!wanted.contains(entry.key)) {
        await txn.delete(
          'exam_topics',
          where: 'id = ?',
          whereArgs: [entry.value],
        );
      }
    }
    for (final topicId in wanted) {
      if (!existing.containsKey(topicId)) {
        await txn.insert(
          'exam_topics',
          ExamTopic(id: _uuid.v4(), examId: examId, topicId: topicId).toMap(),
        );
      }
    }
  }

  /// Deletes the exam. Its checklist rows cascade.
  Future<void> delete(String id) async {
    final db = await _db.database;
    await db.delete('exams', where: 'id = ?', whereArgs: [id]);
  }

  Future<Exam?> getById(String id) async {
    final db = await _db.database;
    final rows = await db.query('exams', where: 'id = ?', whereArgs: [id]);
    if (rows.isEmpty) return null;
    return Exam.fromMap(rows.first);
  }

  /// Every exam, nearest date first (past exams come first).
  Future<List<Exam>> getAll() async {
    final db = await _db.database;
    final rows = await db.query('exams', orderBy: 'examDate ASC');
    return rows.map(Exam.fromMap).toList();
  }

  /// Exams from today onwards, nearest first. Pass [limit] to cap the list
  /// (the Home card asks for the nearest few).
  Future<List<Exam>> getUpcoming({int? limit, DateTime? now}) async {
    final db = await _db.database;
    final today = now ?? DateTime.now();
    final startOfToday = DateTime(today.year, today.month, today.day);
    final rows = await db.query(
      'exams',
      where: 'examDate >= ?',
      whereArgs: [startOfToday.toIso8601String()],
      orderBy: 'examDate ASC',
      limit: limit,
    );
    return rows.map(Exam.fromMap).toList();
  }

  Future<List<Exam>> getForSubject(String subjectId) async {
    final db = await _db.database;
    final rows = await db.query(
      'exams',
      where: 'subjectId = ?',
      whereArgs: [subjectId],
      orderBy: 'examDate ASC',
    );
    return rows.map(Exam.fromMap).toList();
  }

  /// The checklist of one exam, in the subject's roadmap order.
  Future<List<ExamTopic>> getTopicLinks(String examId) async {
    final db = await _db.database;
    final rows = await db.rawQuery(
      'SELECT et.* FROM exam_topics et '
      'JOIN topics t ON t.id = et.topicId '
      'WHERE et.examId = ? '
      'ORDER BY t.sortOrder ASC, t.rowid ASC',
      [examId],
    );
    return rows.map(ExamTopic.fromMap).toList();
  }

  /// Every checklist row of every exam (used by backups).
  Future<List<ExamTopic>> getAllTopicLinks() async {
    final db = await _db.database;
    final rows = await db.query('exam_topics', orderBy: 'rowid ASC');
    return rows.map(ExamTopic.fromMap).toList();
  }

  /// Ticks or unticks one topic of an exam's checklist.
  Future<void> setTopicPrepared(String examTopicId, bool prepared) async {
    final db = await _db.database;
    await db.update(
      'exam_topics',
      {'prepared': prepared ? 1 : 0},
      where: 'id = ?',
      whereArgs: [examTopicId],
    );
  }

  Future<ExamProgress> getProgress(String examId) async {
    final all = await getProgressForAll();
    return all[examId] ?? const ExamProgress.empty();
  }

  /// Progress of every exam in one query, keyed by exam id. Exams without
  /// topics are simply absent from the map.
  Future<Map<String, ExamProgress>> getProgressForAll() async {
    final db = await _db.database;
    final rows = await db.rawQuery(
      'SELECT examId, COUNT(*) AS total, '
      'COALESCE(SUM(prepared), 0) AS prepared '
      'FROM exam_topics GROUP BY examId',
    );
    return {
      for (final r in rows)
        r['examId'] as String: ExamProgress(
          prepared: (r['prepared'] as int?) ?? 0,
          total: (r['total'] as int?) ?? 0,
        ),
    };
  }
}

SessionRecurrence recurrenceFromString(String value) {

class StudySession {
  final String id;
  final String subjectId;
  final String? topicId;
  final DateTime scheduledDate;
  final DateTime startTime;
  final DateTime endTime;
  final int? actualDuration; // minutes, filled in once a session runs
  final SessionStatus status;
  final String notes;
  final int? rating; // 1-5, optional, set at session end
  final bool complete;
  final SessionRecurrence recurrence;
  final DateTime? recurrenceUntil; // last day of the series (inclusive)
  final String? seriesId; // shared by every occurrence of one series

  const StudySession({
    required this.id,
    required this.subjectId,
    this.topicId,
    required this.scheduledDate,
    required this.startTime,
    required this.endTime,
    this.actualDuration,
    this.status = SessionStatus.planned,
    this.notes = '',
    this.rating,
    this.complete = false,
    this.recurrence = SessionRecurrence.none,
    this.recurrenceUntil,
    this.seriesId,
  });

  /// Business rule: a session cannot be saved with an end time earlier
  /// than or equal to its start time.
  bool get isTimeRangeValid => endTime.isAfter(startTime);

  /// True when this session is one occurrence of a repeating series.
  bool get isRecurring => seriesId != null;

  StudySession copyWith({
    String? id,
    String? subjectId,
    String? topicId,
    DateTime? scheduledDate,
    DateTime? startTime,
    DateTime? endTime,
    int? actualDuration,
    SessionStatus? status,
    String? notes,
    int? rating,
    bool? complete,
    SessionRecurrence? recurrence,
    DateTime? recurrenceUntil,
    String? seriesId,
  }) {
    return StudySession(
      id: id ?? this.id,
      subjectId: subjectId ?? this.subjectId,
      topicId: topicId ?? this.topicId,
      scheduledDate: scheduledDate ?? this.scheduledDate,
      startTime: startTime ?? this.startTime,
      endTime: endTime ?? this.endTime,
      actualDuration: actualDuration ?? this.actualDuration,
      status: status ?? this.status,
      notes: notes ?? this.notes,
      rating: rating ?? this.rating,
      complete: complete ?? this.complete,
      recurrence: recurrence ?? this.recurrence,
      recurrenceUntil: recurrenceUntil ?? this.recurrenceUntil,
      seriesId: seriesId ?? this.seriesId,
    );
  }

  Map<String, dynamic> toMap() => {
        'id': id,
        'subjectId': subjectId,
        'topicId': topicId,
        'scheduledDate': scheduledDate.toIso8601String(),
        'startTime': startTime.toIso8601String(),
        'endTime': endTime.toIso8601String(),
        'actualDuration': actualDuration,
        'status': status.name,
        'notes': notes,
        'rating': rating,
        'complete': complete ? 1 : 0,
        'recurrence': recurrence.name,
        'recurrenceUntil': recurrenceUntil?.toIso8601String(),
        'seriesId': seriesId,
      };

  factory StudySession.fromMap(Map<String, dynamic> map) => StudySession(
        id: map['id'] as String,
        subjectId: map['subjectId'] as String,
        topicId: map['topicId'] as String?,

/// An exam the student is preparing for (V2 Step 11).
class Exam {
  final String id;
  final String subjectId;
  final String title;

  /// Date and time of the exam, in the device's local time.
  final DateTime examDate;
  final String location;
  final String notes;
  final DateTime createdAt;

  const Exam({
    required this.id,
    required this.subjectId,
    required this.title,
    required this.examDate,
    this.location = '',
    this.notes = '',
    required this.createdAt,
  });

  Exam copyWith({
    String? id,
    String? subjectId,
    String? title,
    DateTime? examDate,
    String? location,
    String? notes,
    DateTime? createdAt,
  }) {
    return Exam(
      id: id ?? this.id,
      subjectId: subjectId ?? this.subjectId,
      title: title ?? this.title,
      examDate: examDate ?? this.examDate,
      location: location ?? this.location,
      notes: notes ?? this.notes,
      createdAt: createdAt ?? this.createdAt,
    );
  }

  /// Whole calendar days from today to the exam day (0 = today,
  /// negative = already passed). The time of day is ignored.
  int daysLeft([DateTime? now]) {
    final today = now ?? DateTime.now();
    final todayDay = DateTime.utc(today.year, today.month, today.day);
    final examDay = DateTime.utc(examDate.year, examDate.month, examDate.day);
    return examDay.difference(todayDay).inDays;
  }

  /// True once the exam day is behind us.
  bool isPast([DateTime? now]) => daysLeft(now) < 0;

  Map<String, dynamic> toMap() => {
        'id': id,
        'subjectId': subjectId,
        'title': title,
        'examDate': examDate.toLocal().toIso8601String(),
        'location': location,
        'notes': notes,
        'createdAt': createdAt.toIso8601String(),
      };

  factory Exam.fromMap(Map<String, dynamic> map) => Exam(
        id: map['id'] as String,
        subjectId: map['subjectId'] as String,
        title: map['title'] as String,
        examDate: DateTime.parse(map['examDate'] as String),
        location: (map['location'] as String?) ?? '',
        notes: (map['notes'] as String?) ?? '',
        createdAt: DateTime.parse(map['createdAt'] as String),
      );
}

class ExamProgress {

  const ExamProgress.empty()
      : prepared = 0,

/// Small date helpers shared by Home, Timetable, and Progress —
/// keeps "start of day" / "start of week" logic in one place.
DateTime startOfDay(DateTime date) => DateTime(date.year, date.month, date.day);

/// Monday-start week, matching the spec's "Week • September 8–14" style.
DateTime startOfWeek(DateTime date) {
  final day = startOfDay(date);
  return day.subtract(Duration(days: day.weekday - DateTime.monday));
}

DateTime endOfWeek(DateTime date) =>
    startOfWeek(date).add(const Duration(days: 7));

bool isSameDay(DateTime a, DateTime b) =>
    a.year == b.year && a.month == b.month && a.day == b.day;

String formatDuration(int minutes) {
  final h = minutes ~/ 60;
  final m = minutes % 60;
  if (h == 0) return '${m}m';
  if (m == 0) return '${h}h';
  return '${h}h ${m}m';
}

String formatTimeOfDay(DateTime time) {
  final hour = time.hour % 12 == 0 ? 12 : time.hour % 12;
  final minute = time.minute.toString().padLeft(2, '0');
  final period = time.hour >= 12 ? 'PM' : 'AM';
  return '$hour:$minute $period';
}

/// Sum of actualDuration (minutes) across completed sessions whose
/// scheduledDate falls on [date].
int completedMinutesForDate(List<StudySession> sessions, DateTime date) {
  return sessions
      .where((s) => s.complete && isSameDay(s.scheduledDate, date))
      .fold(0, (sum, s) => sum + (s.actualDuration ?? 0));
}

/// Sum of actualDuration across completed sessions within [sessions],
/// with no date filtering — pass in a week's worth of sessions to get
/// a weekly total.
int completedMinutesTotal(List<StudySession> sessions) {
  return sessions
      .where((s) => s.complete)
      .fold(0, (sum, s) => sum + (s.actualDuration ?? 0));
}

/// Current study streak, per the V1 business rule: a day counts only
/// if it has at least one completed session.
///
/// [completedDates] are yyyy-MM-dd strings from
/// SessionRepository.getCompletedDates(). Today not yet having a
/// completed session does not break the streak — it simply isn't
/// counted yet, since the day isn't over.
int currentStreak(Set<String> completedDates, {DateTime? today}) {
  final now = today ?? DateTime.now();
  var cursor = startOfDay(now);
  var streak = 0;

  bool hasSessionOn(DateTime day) => completedDates.contains(_dateKey(day));

  // If today already has a completed session, count it and step back
  // from yesterday. If not, today is skipped (not yet failed) and we
  // start checking from yesterday.
  if (hasSessionOn(cursor)) {
    streak++;
  }
  cursor = cursor.subtract(const Duration(days: 1));

  while (hasSessionOn(cursor)) {
    streak++;
    cursor = cursor.subtract(const Duration(days: 1));
  }

  return streak;
}

String _dateKey(DateTime date) {
  final y = date.year.toString().padLeft(4, '0');
  final m = date.month.toString().padLeft(2, '0');
  final d = date.day.toString().padLeft(2, '0');
  return '$y-$m-$d';
}

/// Longest run of consecutive completed-session days on record, e.g.
/// for the "Longest: 18 days" line on the Progress screen.
int longestStreak(Set<String> completedDates) {
  if (completedDates.isEmpty) return 0;

  final sortedDates = completedDates.map((key) {
    final parts = key.split('-');
    return DateTime(
        int.parse(parts[0]), int.parse(parts[1]), int.parse(parts[2]));
  }).toList()
    ..sort();

  var longest = 1;
  var current = 1;
  for (var i = 1; i < sortedDates.length; i++) {
    final gap = sortedDates[i].difference(sortedDates[i - 1]).inDays;
    if (gap == 1) {
      current++;
      longest = current > longest ? current : longest;
    } else if (gap > 1) {
      current = 1;
    }
  }
  return longest;
}

/// V2 Step 10a: fl_chart bar chart for study time (values are minutes).
///
/// Takes the same [BarChartPoint] list as the old SimpleBarChart, so the
/// existing ProgressViewData does not need to change. Shows grid lines,
/// time labels on the left, rounded bars and a tap tooltip.
/// [highlightIndex] draws one bar darker (for example today).
class StudyBarChart extends StatelessWidget {
  final List<BarChartPoint> points;
  final double height;
  final int? highlightIndex;

  const StudyBarChart({
    super.key,
    required this.points,
    this.height = 200,
    this.highlightIndex,
  });

  /// Grid step candidates in minutes. The smallest one that keeps the
  /// chart to about 5 grid lines is used.
  static const _steps = [30, 60, 120, 180, 300, 600, 1200, 3000, 6000];

  static int _stepFor(int maxMinutes) {
    for (final step in _steps) {
      if (maxMinutes <= step * 5) return step;
    }
    return _steps.last;
  }

  @override
  Widget build(BuildContext context) {
    final maxMinutes = points.fold<int>(
      0,
      (m, p) => p.value > m ? p.value : m,
    );
    final step = _stepFor(maxMinutes);
    // Always leave some headroom above the tallest bar.
    final maxY = (((maxMinutes ~/ step) + 1) * step).toDouble();

    final gridColor = context.secondaryText.withValues(alpha: 0.18);
    final labelStyle = Theme.of(context).textTheme.labelSmall?.copyWith(
          color: context.secondaryText,
        );
    final barWidth = points.length > 8 ? 10.0 : 18.0;

    final groups = List.generate(points.length, (i) {
      final isHighlight = highlightIndex == i;
      return BarChartGroupData(
        x: i,
        barRods: [
          BarChartRodData(
            toY: points[i].value.toDouble(),
            width: barWidth,
            color: isHighlight ? AppColors.primaryDark : AppColors.primary,
            borderRadius: const BorderRadius.vertical(
              top: Radius.circular(5),
            ),
          ),
        ],
      );
    });

    return Semantics(
      label: 'Study time bar chart',
      child: SizedBox(
        height: height,
        child: BarChart(
          BarChartData(
            maxY: maxY,
            minY: 0,
            alignment: BarChartAlignment.spaceAround,
            barGroups: groups,
            borderData: FlBorderData(show: false),
            gridData: FlGridData(
              show: true,
              drawVerticalLine: false,
              horizontalInterval: step.toDouble(),
              getDrawingHorizontalLine: (value) =>
                  FlLine(color: gridColor, strokeWidth: 1),
            ),
            titlesData: FlTitlesData(
              topTitles: const AxisTitles(
                sideTitles: SideTitles(showTitles: false),
              ),
              rightTitles: const AxisTitles(
                sideTitles: SideTitles(showTitles: false),
              ),
              bottomTitles: AxisTitles(
                sideTitles: SideTitles(
                  showTitles: true,
                  reservedSize: 28,
                  getTitlesWidget: (value, meta) {
                    final i = value.toInt();
                    if (i < 0 || i >= points.length) {
                      return const SizedBox.shrink();
                    }
                    return Padding(
                      padding: const EdgeInsets.only(top: 8),
                      child: Text(points[i].label, style: labelStyle),
                    );
                  },
                ),
              ),
              leftTitles: AxisTitles(
                sideTitles: SideTitles(
                  showTitles: true,
                  reservedSize: 40,
                  interval: step.toDouble(),
                  getTitlesWidget: (value, meta) {
                    final text =
                        value == 0 ? '0' : formatDuration(value.round());
                    return Padding(
                      padding: const EdgeInsets.only(right: 6),
                      child: Text(
                        text,
                        textAlign: TextAlign.right,
                        style: labelStyle,
                      ),
                    );
                  },
                ),
              ),
            ),
            barTouchData: BarTouchData(
              enabled: true,
              touchTooltipData: BarTouchTooltipData(
                getTooltipColor: (_) => AppColors.primaryDark,
                getTooltipItem: (group, groupIndex, rod, rodIndex) {
                  return BarTooltipItem(
                    formatDuration(rod.toY.round()),
                    const TextStyle(
                      color: Colors.white,
                      fontWeight: FontWeight.w600,
                      fontSize: 12,
                    ),
                  );
                },
              ),
            ),
          ),
        ),
      ),
    );
  }
